        public CustomTSqlParser(
            CodeComprehensionMetadata projectMetadata,
            string filePath,
            string fileContent,
            ICharStream inputStream,
            ITokenStream tokenStream)
        {
            ProjectMetadata = projectMetadata ?? throw new ArgumentNullException(nameof(projectMetadata));
            Project_id = projectMetadata.ProjectId.ToString();
            Path = filePath ?? throw new ArgumentNullException(nameof(filePath));
            FileContent = fileContent ?? throw new ArgumentNullException(nameof(fileContent));
            InputStream = inputStream ?? throw new ArgumentNullException(nameof(inputStream));
            TokenStream = tokenStream ?? throw new ArgumentNullException(nameof(tokenStream));

            FileName = System.IO.Path.GetFileName(Path);
            string dir = System.IO.Path.GetDirectoryName(Path);
            FolderName = dir != null ? System.IO.Path.GetFileName(dir) : string.Empty;

            // Initialize state
            CurrentFunction = null;
            CurrentIfDepth = 0;
            CurrentCaseDepth = 0;
            CurrentLoopDepth = 0;
            FunctionSnippets = new List<FunctionSnippetSchema>();
            FunctionSnippetStack = new List<FunctionSnippetSchema>();

            // Main program level
            MainProgramDecisionNodes = new List<DecisionNodeSchema>();
            ColumnName = new HashSet<string>();
            TableAlias = new Dictionary<string, string>();
            UniqueFields = new HashSet<(string, string)>();
            MainProgramCallees = new List<FunctionCallee>();
            MainProgramServiceCalls = new List<ServiceCallDataEntity>();
            MainProgramDbcalls = new List<DbCallDataEntity>();
            MainProgramCalculations = new List<CalculationEntity>();
            MainProgramTransactionCommits = new List<TransactionCommitEntity>();
            MainProgramTransactionRollbacks = new List<TransactionRollbackEntity>();
            MainProgramIoCalls = new List<IOCallEntity>();
            MainProgramLoopEntities = new List<LoopEntityNodeSchema>();
            MainProgramDatabaseEntities = new List<DatabaseEntitySchema>();
            MainProgramInputEntities = new List<InputEntitySchema>();
            MainProgramDatabaseFields = new List<DatabaseFieldSchema>();

            // Program level
            Entities = new List<Entity>();
            EntityMetadata = new List<object>();
            Imports = new List<string>();
            Aliases = new List<ParserService.Entities.Schemas.AliasNodeSchema>();

            // Execution orders
            ExecutionOrder = 0;
            CalleeExecutionOrder = 0;
            SnippetExecutionOrder = 0;

            // Return control
            ReturnControlConstructs = new List<object>();

            // DB call
            CurrentDbcallDepth = 0;
            DbCallStack = new List<DbCallDataEntity>();
            DbCallExecutionOrder = 0;

            // Custom extractor
            Cve = new ConstantVariableExtractor();

            // Comment tokens
            CommentTokens = new HashSet<int> { TSqlLexer.COMMENT, TSqlLexer.LINE_COMMENT };
        }

        // Get original text from context (includes everything)
        public string GetOriginalTextFromContext(ParserRuleContext ctx)
        {
            int startIdx = ctx.Start.StartIndex;
            int stopIdx = ctx.Stop.StopIndex;
            string text = InputStream.GetText(new Antlr4.Runtime.Misc.Interval(startIdx, stopIdx));
            return text;
        }

        // Get snippet text without comments
        public string GetSnippetTextFromContext(ParserRuleContext ctx)
        {
            int startIdx = ctx.Start.TokenIndex;
            int stopIdx = ctx.Stop.TokenIndex;

            List<IToken> tokens = ((CommonTokenStream)TokenStream).GetTokens(startIdx, stopIdx).ToList();
            string visibleText = string.Concat(
                tokens
                    .Where(t => !CommentTokens.Contains(t.Type))
                    .Select(t => t.Text)
            );

            return visibleText;
        }

        // Get current context name by traversing up hierarchy
        public string GetCurrentContextName(ParserRuleContext ctx)
        {
            while (ctx != null)
            {
                if (ctx is TSqlParser.Create_or_alter_functionContext funcCtx)
                {
                    return funcCtx.funcName.procedure.GetText().Trim('[', ']');
                }
                if (ctx is TSqlParser.Create_or_alter_procedureContext procCtx)
                {
                    return procCtx.procName.procedure.GetText().Trim('[', ']');
                }
                if (ctx is TSqlParser.Create_or_alter_dml_triggerContext dmlTrigCtx)
                {
                    return dmlTrigCtx.simple_name().name.GetText().Trim('[', ']');
                }
                if (ctx is TSqlParser.Create_or_alter_ddl_triggerContext ddlTrigCtx)
                {
                    return ddlTrigCtx.simple_name().name.GetText().Trim('[', ']');
                }
                ctx = (ParserRuleContext)ctx.Parent;
            }

            return MAIN_SNIPPET;
        }

        // Create variable entity list from variables
        public List<VariableEntity> CreateVariableEntityList(List<dynamic> variables)
        {
            var variableList = new List<VariableEntity>();
            foreach (var variable in variables)
            {
                variableList.Add(new VariableEntity
                {
                    Term = variable.term,
                    Type = variable.type,
                    UsageContext = variable.usage_context,
                    Meaning = variable.meaning,
                    FilePath = variable.file_path
                });
            }
            return variableList;
        }

        // Extract table source information recursively
        public (HashSet<string> tables, HashSet<string> aliases, List<string> filterConditions) ExtractTableSource(
            ParserRuleContext tableSource)
        {
            var tables = new HashSet<string>();
            var aliases = new HashSet<string>();
            var filterConditions = new List<string>();

            if (tableSource == null)
                return (tables, aliases, filterConditions);

            try
            {
                if (tableSource is TSqlParser.Table_sourceContext tableSourceCtx)
                {
                    var tsi = tableSourceCtx.table_source_item();
                    if (tsi?.as_table_alias() != null)
                    {
                        aliases.Add(GetOriginalTextFromContext(tsi.as_table_alias()));
                    }
                    if (tsi?.full_table_name() != null)
                    {
                        tables.Add(GetOriginalTextFromContext(tsi.full_table_name()));
                    }

                    var joinParts = tableSourceCtx.join_part();
                    if (joinParts != null && joinParts.Length > 0)
                    {
                        foreach (var join in joinParts)
                        {
                            if (join.join_on() != null)
                            {
                                var joinOn = join.join_on();
                                if (joinOn.search_condition() != null)
                                {
                                    filterConditions.Add(GetOriginalTextFromContext(joinOn.search_condition()));
                                }
                                var (retTables, retAliases, retFilterConditions) = 
                                    ExtractTableSource(joinOn.table_source());
                                tables.UnionWith(retTables);
                                aliases.UnionWith(retAliases);
                                filterConditions.AddRange(retFilterConditions);
                            }
                        }
                    }
                }
            }
            catch
            {
                // Swallow exceptions for robustness
            }

            return (tables, aliases, filterConditions);
        }

        // Create extracted calculation snippet
        public void CreateExtractedCalculationSnippet(string snippet, string functionName, int funcStartLine)
        {
            var extractData = Cve.ExtractMathematicalCalculations(snippet);
            var extractDataStartLine = Cve.ExtractMathematicalCalculationsStartline(snippet);
            
            for (int cnt = 0; cnt < extractData.Count; cnt++)
            {
                try
                {
                    dynamic lineDict = extractData[cnt];
                    string resultantData = "";
                    string expressionData = "";

                    if (lineDict["match_group"].Contains("="))
                    {
                        var matchData = lineDict["match_group"].Split("=");
                        resultantData = matchData[0];
                        expressionData = matchData[1];
                    }
                    else
                    {
                        expressionData = lineDict["match_group"];
                    }

                    string strMatchedData = string.Join(" ", expressionData);
                    var variablesDetails = Cve.ExtractConstantsVariables(strMatchedData, Path);
                    var variableEntities = CreateVariableEntityList(variablesDetails.Cast<dynamic>().ToList());

                    string matchedData = expressionData.Trim();
                    bool keyword = matchedData.Contains("SELECT", StringComparison.OrdinalIgnoreCase);

                    if (!keyword)
                    {
                        var operands = GetOperands(matchedData);
                        var (startLine, endLine) = FindTargetLines(matchedData, (int)extractDataStartLine[cnt]["start"]);

                        var calculationEntity = new CalculationEntity
                        {
                            Name = functionName,
                            Tech = ProjectMetadata.Language,
                            FileName = FileName,
                            FilePath = Path,
                            StartLineNumber = startLine,
                            EndLineNumber = endLine,
                            ProgramId = FileName,
                            Type = "",
                            Snippet = matchedData,
                            Resultant = resultantData,
                            Operands = operands,
                            Variable = variableEntities
                        };

                        if (CurrentFunction != null)
                        {
                            CurrentFunction.Calculation.Add(calculationEntity);
                        }
                        else
                        {
                            MainProgramCalculations.Add(calculationEntity);
                        }
                    }
                }
                catch (Exception ex)
                {
                    // Log error and continue
                }
            }
        }

        // Create function snippet
        public FunctionSnippetSchema CreateFunctionSnippet(ParserRuleContext ctx)
        {
            ExecutionOrder = 0;
            CalleeExecutionOrder = 0;
            CurrentIfDepth = 0;
            CurrentCaseDepth = 0;
            SnippetExecutionOrder += 1;

            int startLineNumber = ctx.Start.Line;
            int endLineNumber = ctx.Stop.Line;

            string functionName = GetCurrentContextName(ctx);

            string className = FileName;
            if (ctx is TSqlParser.Create_or_alter_functionContext funcCtx)
            {
                className = funcCtx.funcName.GetText();
            }
            else if (ctx is TSqlParser.Create_or_alter_procedureContext procCtx)
            {
                className = procCtx.procName.GetText();
            }
            else if (ctx is TSqlParser.Create_or_alter_dml_triggerContext dmlCtx)
            {
                className = dmlCtx.simple_name().GetText();
            }
            else if (ctx is TSqlParser.Create_or_alter_ddl_triggerContext ddlCtx)
            {
                className = ddlCtx.simple_name().GetText();
            }

            string snippetText = GetOriginalTextFromContext(ctx);

            var variablesDetails = Cve.ExtractConstantsVariables(snippetText, Path);
            var variableEntities = CreateVariableEntityList(variablesDetails.Cast<dynamic>().ToList());

            var functionSnippet = new FunctionSnippetSchema
            {
                FilePath = Path,
                FileName = FileName,
                FunctionName = functionName,
                ClassName = className,
                Snippet = snippetText,
                StartLineNumber = startLineNumber,
                EndLineNumber = endLineNumber,
                Callees = new List<FunctionCallee>(),
                Annotations = new List<string>(),
                ReturnType = "NA",
                DecisionNodeConstructs = new List<DecisionNodeSchema>(),
                DbCalls = new List<object>(),
                ServiceCalls = new List<object>(),
                Calculation = new List<object>(),
                ExecutionOrder = ExecutionOrder,
                SnippetExecutionOrder = SnippetExecutionOrder,
                Inputs = new List<object>(),
                Variables = variableEntities.Cast<object>().ToList()
            };

            FunctionSnippets.Add(functionSnippet);
            return functionSnippet;
        }

        // Visit function snippet
        public void VisitFunctionSnippet(ParserRuleContext ctx)
        {
            var functionSnippet = CreateFunctionSnippet(ctx);
            int functionStartLine = ctx.Start.StartIndex;

            CurrentFunction = functionSnippet;
            FunctionSnippetStack.Add(functionSnippet);

            CreateExtractedCalculationSnippet(
                CurrentFunction.Snippet,
                CurrentFunction.FunctionName,
                functionStartLine
            );

            VisitChildren(ctx);

            CurrentFunction = FunctionSnippetStack.Count > 0 
                ? FunctionSnippetStack[FunctionSnippetStack.Count - 1] 
                : null;
            if (FunctionSnippetStack.Count > 0)
            {
                FunctionSnippetStack.RemoveAt(FunctionSnippetStack.Count - 1);
            }
        }

        // Create rule snippet
        public DecisionNodeSchema CreateRuleSnippet(ParserRuleContext ctx, RuleConstructType constructType)
        {
            int startLineNumber = ctx.Start.Line;
            int endLineNumber = ctx.Stop.Line;

            Regex conditionalKeywords;
            if (constructType == ParserService.Infrastructure.Utils.RuleConstructType.SWITCH_CONSTRUCT)
            {
                conditionalKeywords = new Regex(@"\b(WHEN|ELSE)\b", RegexOptions.IgnoreCase);
            }
            else
            {
                conditionalKeywords = new Regex(@"\b((?<!END-)IF|ELSE|(?<!END-)DECIDE)\b", RegexOptions.IgnoreCase);
            }

            string snippetText = GetOriginalTextFromContext(ctx);
            var matches = conditionalKeywords.Matches(snippetText);
            int numConditions = matches.Count;

            string cleanedText = snippetText.Replace("\r", " ");
            string hostingFunctionName = GetCurrentContextName(ctx);

            var variablesDetails = Cve.ExtractConstantsVariables(cleanedText, Path).Cast<dynamic>().ToList();

            string numericExtractionText = "";
            if (ctx is TSqlParser.If_statementContext ifCtx && ifCtx.children != null)
            {
                var searchCondCtx = ifCtx.search_condition();
                if (searchCondCtx != null)
                {
                    numericExtractionText = GetOriginalTextFromContext(searchCondCtx);
                }
            }
            else
            {
                numericExtractionText = snippetText;
            }

            variablesDetails.AddRange(
                Cve.ExtractNumericConstants(numericExtractionText.Replace("\r", " "), Path).Cast<dynamic>().ToList()
            );
            var variableEntities = CreateVariableEntityList(variablesDetails);

            var decisionNode = new DecisionNodeSchema
            {
                Name = hostingFunctionName,
                ShortSummary = "",
                StartLineNumber = startLineNumber,
                EndLineNumber = endLineNumber,
                Snippet = snippetText,
                NumberOfConditionals = numConditions,
                Complexity = ComplexityHelper.CategorizeComplexity(numConditions).ToString(),
                ConstructType = constructType.ToString(),
                ExecutionOrder = ExecutionOrder,
                Variable = variableEntities.Cast<object>().ToList()
            };

            return decisionNode;
        }

        // Create DB call
        public DbCallDataEntity CreateDbCall(
            ParserRuleContext ctx,
            string query,
            List<string> filterConditions,
            HashSet<string> aliases,
            HashSet<string> tables,
            StringUtils.DbCallDataEntity_TYPE operationType)
        {
            ExecutionOrder += 1;

            string fullText = GetOriginalTextFromContext(ctx);
            string cleanedText = fullText.Replace("\r", " ");

            int startLineNumber = ctx.Start.Line;
            int endLineNumber = ctx.Stop.Line;

            var variablesDetails = Cve.ExtractConstantsVariables(cleanedText, Path).Cast<dynamic>().ToList();
            var variableEntities = CreateVariableEntityList(variablesDetails);

            string hostingFunctionName = GetCurrentContextName(ctx);

            var tableNames = new List<string>();
            foreach (var table in tables)
            {
                string tableName = table.Split('.').Last().Trim('[', ']').ToLower();
                tableNames.Add(tableName);
            }

            var dbCallEntity = new DbCallDataEntity
            {
                Name = hostingFunctionName,
                FilePath = Path,
                FileName = FileName,
                Tech = ProjectMetadata.Language,
                ProgramId = FileName,
                StartLineNumber = startLineNumber,
                EndLineNumber = endLineNumber,
                Snippet = cleanedText,
                OperationType = operationType.ToString(),
                ExecutionOrder = ExecutionOrder,
                Variable = variableEntities,
                Query = query,
                FilterConditions = filterConditions,
                Aliases = aliases.ToList(),
                TableNames = tableNames
            };

            if (CurrentFunction != null)
            {
                CurrentFunction.DbCalls.Add(dbCallEntity);
            }
            else
            {
                MainProgramDbcalls.Add(dbCallEntity);
            }

            // Identify order and levels of DB calls
            if (DbCallStack.Count == 0)
            {
                DbCallExecutionOrder += 1;
                dbCallEntity.Level = "PARENT";
                dbCallEntity.DbCallExecutionOrder = DbCallExecutionOrder.ToString();
                dbCallEntity.ChildCount = 0;
            }
            else
            {
                var parent = DbCallStack[DbCallStack.Count - 1];
                parent.ChildCount += 1;

                dbCallEntity.Level = "CHILD";
                dbCallEntity.DbCallExecutionOrder = parent.DbCallExecutionOrder + "." + parent.ChildCount;
                dbCallEntity.ChildCount = 0;
            }

            return dbCallEntity;
        }

        // Get callee conditional hierarchy
        public Dictionary<string, object> GetCalleeConditionalHierarchy(ParserRuleContext ctx)
        {
            var hierarchy = new Dictionary<string, object>();
            var currentLevel = hierarchy;
            bool isElse = false;

            void UpdateCurrentLevel(string conditionString, int startLineNum, int endLineNum, List<string> siblings)
            {
                conditionString = Regex.Replace(conditionString, @"\n\d{4}\s*$", "");

                currentLevel["condition"] = conditionString;
                currentLevel["start_line_number"] = startLineNum;
                currentLevel["end_line_number"] = endLineNum;
                currentLevel["siblings"] = siblings;
                currentLevel["is_else"] = isElse;
                currentLevel["parent"] = new Dictionary<string, object>();
                currentLevel = (Dictionary<string, object>)currentLevel["parent"];
            }

            while (ctx != null)
            {
                var parentCtx = ctx.Parent;
                if (parentCtx == null)
                    break;

                if (parentCtx is TSqlParser.If_statementContext ifStmtCtx && ifStmtCtx.children != null)
                {
                    var elseSymbol = ifStmtCtx.ELSE();
                    if (elseSymbol != null)
                    {
                        var sym = elseSymbol.Symbol;
                        if (sym.StopIndex < ctx.Stop.StartIndex)
                        {
                            isElse = true;
                        }
                    }

                    var conditionCtx = ifStmtCtx.search_condition();
                    if (conditionCtx != null)
                    {
                        ParserRuleContext parentParserCtx = parentCtx as ParserRuleContext;
                        if (parentParserCtx != null)
                        {
                            UpdateCurrentLevel(
                                GetOriginalTextFromContext(conditionCtx),
                                parentParserCtx.Start.Line,
                                parentParserCtx.Stop.Line,
                                new List<string>()
                            );
                        }
                    }

                    isElse = false;
                }

                ctx = parentCtx as ParserRuleContext;
            }

            return hierarchy;
        }

        // Visit Func_body_returns_scalar
        public override object VisitFunc_body_returns_scalar(TSqlParser.Func_body_returns_scalarContext context)
        {
            if (context.expression() != null)
            {
                var expr = context.expression();
                if (expr.bracket_expression() != null)
                {
                    var bracket = expr.bracket_expression();
                    if (bracket.subquery() != null)
                    {
                        // Process subquery if needed
                    }
                }
            }
            return VisitChildren(context);
        }

        // Parse function/procedure name
        public string ParseFuncProcNameServerDatabaseSchema(TSqlParser.Func_proc_name_server_database_schemaContext ctx)
        {
            IParseTree procedure = null;

            if (ctx.func_proc_name_database_schema() != null)
            {
                if (ctx.func_proc_name_database_schema().func_proc_name_schema() != null)
                {
                    procedure = ctx.func_proc_name_database_schema().func_proc_name_schema().procedure;
                }
                else
                {
                    procedure = ctx.func_proc_name_database_schema().procedure;
                }
            }
            else
            {
                procedure = ctx.procedure;
            }

            return procedure != null 
                ? procedure.GetText().Trim('[', ']') 
                : ctx.GetText();
        }

        // Create service call
        public ServiceCallDataEntity CreateServiceCall(ParserRuleContext ctx, string funcName)
        {
            ExecutionOrder += 1;
            int startLineNumber = ctx.Start.Line;
            int endLineNumber = ctx.Stop.Line;

            string snippetText = GetOriginalTextFromContext(ctx);

            var variablesDetails = Cve.ExtractConstantsVariables(snippetText, Path).Cast<dynamic>().ToList();
            var variableEntities = CreateVariableEntityList(variablesDetails);
            
            var parameters = variablesDetails
                .Where(v => !funcName.Contains(((dynamic)v).term))
                .Select(v => (string)((dynamic)v).term)
                .ToList();

            string hostingFunctionName = GetCurrentContextName(ctx);

            var dataEntity = new ServiceCallDataEntity
            {
                Name = hostingFunctionName,
                Tech = ProjectMetadata.Language,
                FileName = FileName,
                FilePath = Path,
                StartLineNumber = startLineNumber,
                EndLineNumber = endLineNumber,
                ProgramId = FileName,
                Target = funcName,
                ExecutionOrder = ExecutionOrder,
                Snippet = snippetText,
                OperationType = "NA",
                Parameters = parameters,
                Scope = "EXTERNAL",
                Variable = variableEntities
            };

            if (CurrentFunction != null)
            {
                CurrentFunction.ServiceCalls.Add(dataEntity);
            }
            else
            {
                MainProgramServiceCalls.Add(dataEntity);
            }

            return dataEntity;
        }

        // Create callee
        public FunctionCallee CreateCallee(ParserRuleContext ctx, CalleeConstructType constructType)
        {
            FunctionCallee callee = null;

            ExecutionOrder += 1;
            CalleeExecutionOrder += 1;
            int startLineNumber = ctx.Start.Line;
            int endLineNumber = ctx.Stop.Line;

            string snippetText = GetOriginalTextFromContext(ctx);

            bool isConditional = false;
            string condition = "NA";
            var conditionalHierarchy = GetCalleeConditionalHierarchy(ctx);
            if (conditionalHierarchy != null && conditionalHierarchy.Count > 0)
            {
                isConditional = true;
                bool isElseFromHierarchy = (bool)conditionalHierarchy["is_else"];
                string conditionFromHierarchy = (string)conditionalHierarchy["condition"];
                condition = isElseFromHierarchy 
                    ? "ELSE_OF(" + conditionFromHierarchy + ")" 
                    : conditionFromHierarchy;
            }

            string targetFunctionName = null;
            var variableEntities = new List<VariableEntity>();
            var parameters = new List<string>();

            if (constructType == CalleeConstructType.EXECUTE)
            {
                if (ctx is TSqlParser.Execute_statementContext execCtx &&
                    execCtx.execute_body() != null &&
                    execCtx.execute_body().func_proc_name_server_database_schema() != null)
                {
                    targetFunctionName = ParseFuncProcNameServerDatabaseSchema(
                        execCtx.execute_body().func_proc_name_server_database_schema()
                    );

                    var variablesDetails = Cve.ExtractConstantsVariables(snippetText, Path).Cast<dynamic>().ToList();
                    variableEntities = CreateVariableEntityList(variablesDetails);
                    parameters = variablesDetails.Select(v => (string)((dynamic)v).term).ToList();
                }
            }
            else if (constructType == CalleeConstructType.SCALAR_FUNCTION)
            {
                if (ctx is TSqlParser.SCALAR_FUNCTIONContext scalarCtx)
                {
                    if (scalarCtx.scalar_function_name() != null)
                    {
                        var sfnCtx = scalarCtx.scalar_function_name();
                        if (sfnCtx.func_proc_name_server_database_schema() != null)
                        {
                            targetFunctionName = ParseFuncProcNameServerDatabaseSchema(
                                sfnCtx.func_proc_name_server_database_schema()
                            );
                        }
                    }

                    if (scalarCtx.expression_list_() != null)
                    {
                        string variables = GetOriginalTextFromContext(scalarCtx.expression_list_());
                        var variablesDetails = Cve.ExtractConstantsVariables(variables, Path).Cast<dynamic>().ToList();
                        variableEntities = CreateVariableEntityList(variablesDetails);

        public override object VisitSelect_statement(TSqlParser.Select_statementContext context)
        {
            string query = null;
            var aliases = new HashSet<string>();
            var tables = new HashSet<string>();
            var filterConditions = new List<string>();
            StringUtils.DbCallDataEntity_TYPE operationType = StringUtils.DbCallDataEntity_TYPE.READ;

            try
            {
                query = GetOriginalTextFromContext(context).Trim(" \n".ToCharArray());

                if (context.query_expression() != null)
                {
                    var qeCtx = context.query_expression();
                    var qsCtx = qeCtx.query_specification();

                    if (qsCtx != null && qsCtx.table_sources() != null)
                    {
                        var searchConditions = qsCtx.search_condition();
                        if (searchConditions != null)
                        {
                            foreach (var con in searchConditions)
                            {
                                if (con != null)
                                {
                                    filterConditions.Add(GetOriginalTextFromContext(con));
                                }
                            }
                        }

                        var tsList = qsCtx.table_sources().table_source();
                        foreach (var ts in tsList)
                        {
                            var (retT, retAl, retFc) = ExtractTableSource(ts);
                            tables.UnionWith(retT);
                            aliases.UnionWith(retAl);
                            filterConditions.AddRange(retFc);
                        }
                    }

                    if (qsCtx != null && qsCtx.table_name() != null)
                    {
                        tables.Add(GetOriginalTextFromContext(qsCtx.table_name()));
                    }

                    var sqlUnions = qeCtx.sql_union();
                    if (sqlUnions != null)
                    {
                        foreach (var union in sqlUnions)
                        {
                            qsCtx = union.query_specification();
                            if (qsCtx?.table_sources() != null)
                            {
                                var searchConditions = qsCtx.search_condition();
                                if (searchConditions != null)
                                {
                                    foreach (var con in searchConditions)
                                    {
                                        if (con != null)
                                        {
                                            filterConditions.Add(GetOriginalTextFromContext(con));
                                        }
                                    }
                                }

                                var tsList = qsCtx.table_sources().table_source();
                                foreach (var ts in tsList)
                                {
                                    var (retT, retAl, retFc) = ExtractTableSource(ts);
                                    tables.UnionWith(retT);
                                    aliases.UnionWith(retAl);
                                    filterConditions.AddRange(retFc);
                                }
                            }
                        }
                    }
                }
            }
            catch (Exception ex)
            {
                // Log error and continue
            }

            if (tables.Count == 0)
                return null;

            string snippet = GetOriginalTextFromContext(context);
            var tableList = new HashSet<string>();

            foreach (var table in tables)
            {
                if (table.Contains('.'))
                {
                    string tablePart = table.Split('.')[1];
                    tableList.Add(tablePart);
                }
                tableList.Add(table);

                if (!EntityExists(table.Trim()))
                {
                    CreateDatabaseEntity(snippet, table.Trim(), context.Start.Line, context.Stop.Line);
                }
            }

            var dbCall = CreateDbCall(context, query, filterConditions, aliases, tableList, operationType);

            DbCallStack.Add(dbCall);
            CurrentDbcallDepth += 1;
            VisitChildren(context);
            CurrentDbcallDepth -= 1;
            DbCallStack.RemoveAt(DbCallStack.Count - 1);

            CreateDatabaseField();
            return null;
        }

        public List<DatabaseFieldSchema> CreateDatabaseField()
        {
            var fields = ColumnName;
            var databaseFields = new List<DatabaseFieldSchema>();
            string defaultTable = null;

            foreach (var kvp in TableAlias)
            {
                if (kvp.Key == kvp.Value)
                {
                    defaultTable = kvp.Value;
                    break;
                }
            }

            foreach (var fieldData in fields)
            {
                var parts = fieldData.Split('.');
                if (parts.Length == 2)
                {
                    string tablePart = parts[0].Trim('[', ']');
                    string columnPart = parts[1];

                    if (TableAlias.TryGetValue(tablePart, out string actualTable))
                    {
                        string actualTableName = actualTable.Split('.').Last().Trim('[', ']').ToLower();
                        var fieldKey = (columnPart, actualTableName);

                        if (!UniqueFields.Contains(fieldKey))
                        {
                            UniqueFields.Add(fieldKey);
                            var field = new DatabaseFieldSchema
                            {
                                Name = columnPart,
                                TableName = actualTableName,
                                DbTech = ProjectMetadata.Language
                            };
                            databaseFields.Add(field);
                        }
                    }
                }
                else if (parts.Length == 1 && !string.IsNullOrEmpty(defaultTable))
                {
                    string columnPart = parts[0];
                    string defaultTableName = defaultTable.Split('.').Last().Trim('[', ']').ToLower();
                    var fieldKey = (columnPart, defaultTableName);

                    if (!UniqueFields.Contains(fieldKey))
                    {
                        UniqueFields.Add(fieldKey);
                        var field = new DatabaseFieldSchema
                        {
                            Name = columnPart,
                            TableName = defaultTableName,
                            DbTech = ProjectMetadata.Language
                        };
                        databaseFields.Add(field);
                    }
                }
            }

            if (CurrentFunction != null)
            {
                CurrentFunction.DatabaseFields.AddRange(databaseFields);
            }
            else
            {
                MainProgramDatabaseFields.AddRange(databaseFields);
            }

            ColumnName.Clear();
            return databaseFields;
        }

    /// <summary>
    /// C# port of services/codeminer/tsql/visitor/extractors/constant_variable_extractor.ConstantVariableExtractor.
    /// Extracts constants, variables, and mathematical calculations from T-SQL code snippets.
    /// </summary>
    public class ConstantVariableExtractor
    {
        private HashSet<string> keywords = new HashSet<string>();
        private string negative_lookahead = "";
        private string variable_pattern = "";
        private Regex variable_regex = null;
        private Regex keywords_as_vars = null;
        private string string_constant = "";
        private Regex string_constant_regex = null;
        private Regex numeric_constant_regex = null;
        private string calc_pattern = "";
        private Regex calc_regex = null;
        private string pattern_resultant = "";

        public ConstantVariableExtractor()
        {
            keywords = new HashSet<string>();

            // Load all keywords from lexer file
            read_keywords();

            // Build negative lookahead pattern
            negative_lookahead = string.Join("|", keywords.Select(k => Regex.Escape(k)));

            // Variable pattern: uppercase identifiers with optional separators
            variable_pattern = (
                $@"(?<!@)(?!(?:{negative_lookahead})(?=[\s(]|$))" +
                $@"\b[A-Z]+(?:[-._][A-Z0-9]+)*\b"
            );
            variable_regex = new Regex(variable_pattern, RegexOptions.IgnoreCase);

            // Keywords as variables pattern: @VAR_NAME format
            keywords_as_vars = new Regex(
                $@"(?:(?<=\s)|(?<=^))(@[A-Z][A-Z0-9_]*)(?=[\s(]|$)",
                RegexOptions.IgnoreCase
            );

            // String constant pattern: 'string content'
            string_constant = @"'([^']*)'";
            string_constant_regex = new Regex(string_constant);

            // Numeric constant pattern
            numeric_constant_regex = new Regex(@"(?<=\s)\d+(?=\s)|(?<=\s)\d+$");

            // Calculation pattern: expressions with @ variables and operators
            calc_pattern = (
                $@"(?!(?:T|CT|ECT|LECT|ELECT|{negative_lookahead})(?=\s|$))" +
                $@"(?:" +
                $@"(?:SET|SELECT)?\s*" +
                $@"(@[\w]+)?\s*" +
                $@"(?:=\s*)?" +
                $@")?" +
                $@"\(*\s*@[\w.]+(?:\s*[+\-*/]\s*[@\w.()]+)+\s*\)*"
            );
            calc_regex = new Regex(calc_pattern, RegexOptions.IgnoreCase);

            // Pattern for resultant variables
            pattern_resultant = @"SET\s+(@\w+)\s+=";
        }

        /// <summary>
        /// Read keywords from T-SQL lexer grammar file
        /// </summary>
        private void read_keywords()
        {
            // Construct path to TSqlLexer.g4
            string currentDir = Path.GetDirectoryName(System.Reflection.Assembly.GetExecutingAssembly().Location);
            string targetFilePath = Path.Combine(currentDir, "..", "..", "..", "grammer", "TSqlLexer.g4");
            targetFilePath = Path.GetFullPath(targetFilePath);

            string startMarker = "// Keywords Start Here";
            string endMarker = "// Keywords End Here";
            bool capture = false;

            try
            {
                if (File.Exists(targetFilePath))
                {
                    foreach (string line in File.ReadLines(targetFilePath))
                    {
                        string trimmedLine = line.Trim();

                        if (trimmedLine == startMarker)
                        {
                            capture = true;
                            continue;
                        }
                        else if (trimmedLine == endMarker)
                        {
                            capture = false;
                            break;
                        }

                        if (capture && !string.IsNullOrEmpty(trimmedLine) && !trimmedLine.StartsWith("//"))
                        {
                            // Match content between colon and semicolon
                            Match colonMatch = Regex.Match(trimmedLine, @":(.*?);");
                            if (colonMatch.Success)
                            {
                                string afterColon = colonMatch.Groups[1].Value;

                                // Extract all quoted strings
                                MatchCollection keywordMatches = Regex.Matches(afterColon, @"'[^']*'");
                                foreach (Match keywordMatch in keywordMatches)
                                {
                                    string keyword = keywordMatch.Value.Trim(" '".ToCharArray());
                                    keywords.Add(keyword);
                                }
                            }
                        }
                    }
                }
            }
            catch (Exception ex)
            {
                // Swallow exception and continue with empty keywords
            }
        }

        /// <summary>
        /// Extract constants and variables from code snippet
        /// </summary>
        public List<DataDefinitionJSON> extract_constants_variables(string code_snippet, string file_path)
        {
            string[] lines = code_snippet.Split('\n');
            var variable_details = new List<DataDefinitionJSON>();
            var variable_dict = new Dictionary<string, DataDefinitionJSON>();

            foreach (string line in lines)
            {
                // Skip commented lines
                if (line.Trim().StartsWith("--"))
                    continue;

                // Extract string constants
                MatchCollection string_matches = string_constant_regex.Matches(line);
                foreach (Match match in string_matches)
                {
                    string constant = match.Groups[1].Value;
                    if (!variable_dict.ContainsKey(constant) && constant.Trim('\'').Split(' ').Length == 1)
                    {
                        variable_dict[constant] = new DataDefinitionJSON(
                            type: "CONSTANT",
                            term: constant,
                            usage_context: new List<string> { line },
                            file_path: file_path,
                            meaning: ""
                        );
                    }
                    else if (variable_dict.ContainsKey(constant) && constant.Trim('\'').Split(' ').Length == 1)
                    {
                        variable_dict[constant].usage_context.Add(line);
                    }
                }

                // Extract variables (uppercase identifiers)
                string upper_line = line.ToUpper();
                MatchCollection var_matches = variable_regex.Matches(upper_line);
                var variables = new List<string>();
                foreach (Match match in var_matches)
                {
                    variables.Add(line.Substring(match.Index, match.Length));
                }

                foreach (string variable in variables)
                {
                    if (!variable_dict.ContainsKey(variable) && variable.Trim('\'').Split(' ').Length == 1)
                    {
                        variable_dict[variable] = new DataDefinitionJSON(
                            type: "DECLARATION",
                            term: variable,
                            usage_context: new List<string> { line },
                            file_path: file_path,
                            meaning: ""
                        );
                    }
                    else if (variable_dict.ContainsKey(variable) && variable.Trim('\'').Split(' ').Length == 1)
                    {
                        variable_dict[variable].usage_context.Add(line);
                    }
                }

                // Extract keyword variables (@VAR_NAME)
                MatchCollection keyword_var_matches = keywords_as_vars.Matches(upper_line);
                var keyword_vars = new List<string>();
                foreach (Match match in keyword_var_matches)
                {
                    keyword_vars.Add(line.Substring(match.Index, match.Length));
                }

                foreach (string key_var in keyword_vars)
                {
                    if (!variable_dict.ContainsKey(key_var) && key_var.Trim('\'').Split(' ').Length == 1)
                    {
                        variable_dict[key_var] = new DataDefinitionJSON(
                            type: "DECLARATION",
                            term: key_var,
                            usage_context: new List<string> { line },
                            file_path: file_path,
                            meaning: ""
                        );
                    }
                    else if (variable_dict.ContainsKey(key_var) && key_var.Trim('\'').Split(' ').Length == 1)
                    {
                        variable_dict[key_var].usage_context.Add(line);
                    }
                }
            }

            variable_details.AddRange(variable_dict.Values);
            return variable_details;
        }

        private bool IsParseableFile(string filePath)
        {
            var validExtensions = new HashSet<string>(StringComparer.OrdinalIgnoreCase)
            {
                ".sql",
                ".tsql",
                ".sp",
                ".proc"
            };

            var ext = Path.GetExtension(filePath);
            return validExtensions.Contains(ext);
        }

        /// <summary>
        /// Check if file is CREATE TABLE or CREATE VIEW - Python skips variable extraction for these
        /// </summary>
        private bool IsCreateTableOrViewFile(string fileContent)
        {
            // Match CREATE TABLE [dbo].[table_name] pattern (Python line 62)
            var createTableMatch = Regex.Match(fileContent, @"CREATE\s+TABLE\s+\[dbo\]\.\[(\w+)\]", RegexOptions.IgnoreCase);
            if (createTableMatch.Success)
            {
                return true;
            }

            // Match CREATE VIEW pattern (Python line 264)
            var createViewMatch = Regex.Match(fileContent, @"CREATE\s+VIEW\s+(?:\[[^\]]+\]\.)?(?:dbo\.)?\[?([^\s\[\]]+)\]?", RegexOptions.IgnoreCase);
            if (createViewMatch.Success)
            {
                return true;
            }

            return false;
        }

    def parse_view_file(self, content):
        content = content.replace("ï»¿", "")

        # Remove comments
        content = re.sub(r'/\*.*?\*/', '', content, flags=re.DOTALL)
        content = re.sub(r'--.*', '', content)

        # Extract view name
        view_name_match = re.search(r'CREATE\s+VIEW\s+(?:\[[^\]]+\]\.)?(?:dbo\.)?\[?([^\s\[\]]+)\]?', content, re.IGNORECASE)
        if view_name_match is None:
            return None, None, None, None
        view_name = view_name_match.group(1)

        # Extract fields between SELECT and FROM (simplified)
        select_from_match = re.search(r"(?s)(?<=SELECT\s)(.*?)(?=\s+FROM\b)", content, re.IGNORECASE | re.DOTALL)
        select_from_match1 = re.search(r"(?s)(?<=SELECT\s)(.*?)(?=\s+FROM\b)", content)
        fields_str=""
        fields = set()
        if select_from_match:
            fields_str = select_from_match.group(1)
            fields.update(self.get_view_fields(fields_str))
            # Split by commas not inside parentheses, quotes, or CASE ... END
        if select_from_match1:
            fields_str = select_from_match1.group(1)
            fields.update(self.get_view_fields(fields_str))

        # Extract tables from FROM and JOIN clauses
        tables = set()
        # Match table names after FROM or JOIN
        table_pattern = re.compile(r'(?:FROM|JOIN)\s+([^\s,]+)', re.IGNORECASE)
        for match in table_pattern.finditer(content):
            table = match.group(1)
            # Clean aliases and brackets
            table = re.split(r'\s+AS\s+|\s+', table, flags=re.IGNORECASE)[0]
            table_clean = table.replace('[', '').replace(']', '')
            # Skip if it's a subquery or variable
            table_name = table_clean.split('.')[-1]
            if not table_name.startswith('$'):
                tables.add(table_name)  # Get table name after schema.
       
        return  view_name,list(fields),list(tables),fields_str
    def get_view_fields(self,fields_str):
        fields = set()
        sql_reserved_keywords = [
            "SELECT", "FROM", "WHERE", "INSERT", "UPDATE", "DELETE", "JOIN", "INNER", "LEFT", "RIGHT", "FULL",
            "GROUP", "ORDER", "BY", "HAVING", "DISTINCT", "AS", "AND", "OR", "NOT", "NULL", "IS", "IN", "BETWEEN",
            "LIKE", "CREATE", "DROP", "ALTER", "TABLE", "DATABASE", "INDEX", "VIEW", "TRIGGER", "PRIMARY", "KEY",
            "FOREIGN", "CONSTRAINT", "DEFAULT", "CHECK", "EXISTS", "UNION", "ALL", "ANY", "CASE", "WHEN", "THEN",
            "ELSE", "END", "LIMIT", "OFFSET", "VALUES", "SET", "TOP", "ASC", "DESC", "TRANSACTION", "COMMIT",
            "ROLLBACK", "GRANT", "REVOKE", "WITH", "NOLOCK", "ON", "OUTER", "CROSS", "NATURAL", "USING", "INNER JOIN",
        ]
        raw_fields = self.split_fields(fields_str)
        for f in raw_fields:
                # Extract field name: if AS alias, take alias, else the expression cleaned
                # Use regex to find alias after AS (case insensitive)
            alias_matches = re.findall(r'AS\s+([\w_]+)', f, re.IGNORECASE)
                
            if alias_matches:
                    # Handle multiple aliases separated by commas (e.g., "AS Chng, HS_Updt_Dt AS ChngDt, REPLACE(...) AS ChngBy")
                    # Split by comma and take last alias
                last_alias = alias_matches[-1].strip().replace('[', '').replace(']', '')
                    # Fix for known aliases like Chng, ChngDt, ChngBy
                if last_alias.lower() == 'changed':
                    last_alias = 'Chng'
                elif last_alias.lower() == 'chngdt':
                    last_alias = 'ChngDt'
                elif last_alias.lower() == 'chngby':
                    last_alias = 'ChngBy'
                field_name = last_alias
            else:
                    # Clean brackets and take last part if dotted
                field_name = f.replace('[', '').replace(']', '').split('.')[-1].strip()
            for sql_key in sql_reserved_keywords:
                if sql_key+" " in field_name.upper() or " "+sql_key in field_name.upper():
                    field_name = re.sub(sql_key, "", field_name, flags=re.IGNORECASE)  # replace
            fields.add(field_name.strip())
        return fields
        
    def split_fields(self,s):
        fields_list = []
        current = ''
        paren_count = 0
        case_count = 0
        in_quote = False
        i = 0
        while i < len(s):
            char = s[i]
            if char == "'":
                in_quote = not in_quote
                current += char
            elif char == '(' and not in_quote:
                paren_count += 1
                current += char
            elif char == ')' and not in_quote:
                paren_count -= 1
                current += char
            elif s[i:i+4].upper() == 'CASE' and not in_quote:
                case_count += 1
                current += char
            elif s[i:i+3].upper() == 'END' and case_count > 0 and not in_quote:
                case_count -= 1
                current += char
            elif char == ',' and paren_count == 0 and case_count == 0 and not in_quote:
                fields_list.append(current.strip())
                current = ''
            else:
                current += char
            i += 1
        if current.strip():
            fields_list.append(current.strip())
        return fields_list

    def create_rule_snippet(self, ctx, construct_type):
        start_line_number = ctx.start.line
        end_line_number = ctx.stop.line

        # Pattern for both IF statements and CASE expressions
        if construct_type == RuleConstructType.SWITCH_CONSTRUCT:
            # For CASE expressions, count WHEN clauses as they represent the actual conditions
            conditional_keywords = re.compile(r"\b(WHEN|ELSE)\b", re.IGNORECASE)
        else:
            # For IF statements, use the original pattern
            conditional_keywords = re.compile(r"\b((?<!END-)IF|ELSE|(?<!END-)DECIDE)\b", re.IGNORECASE)

        snippet_text = self.get_original_text_from_context(ctx)

        # Find all matches in the snippet, and ...
        matches = conditional_keywords.findall(snippet_text)
        num_conditions = len(matches)

        # Sometimes, main section will have IFs
        cleaned_text = snippet_text.replace('\r', ' ')
        hosting_function_name = self.get_current_context_name(ctx)

        # Extract variables and constants
        variables_details = self.cve.extract_constants_variables(cleaned_text, self.path)

        # Extract numeric constants
        numeric_extraction_text = ''
        if isinstance(ctx, TSqlParser.If_statementContext) and ctx.children:
            # Fetch the condition
            condition_ctx = ctx.search_condition() if hasattr(ctx,
                                                              'search_condition') and ctx.search_condition() else None
            # Now you can work with conditionCtx
            if condition_ctx:
                # Process the condition context
                numeric_extraction_text = self.get_original_text_from_context(condition_ctx)
        else:
            numeric_extraction_text = snippet_text

        variables_details += self.cve.extract_numeric_constants(numeric_extraction_text.replace('\r', ' '), self.path)
        variable_entities = self.create_variable_entity_list(variables_details)

        decision_node = DecisionNodeSchema(
            name=hosting_function_name,
            short_summary="",
            start_line_number=start_line_number,
            end_line_number=end_line_number,
            snippet=snippet_text,
            number_of_conditionals=num_conditions,
            complexity=Complexity.categorize_complexity(num_conditions),
            construct_type=construct_type,
            execution_order=self.execution_order,
            # variables=variables_details,
            variable=variable_entities
        )

        return decision_node

    def create_db_call(
            self, ctx, query: str | None, filter_conditions: list[str], aliases: set[str], tables: set[str],
            operation_type: DbCallDataEntity_TYPE,
    ):
        # First and foremost increment the execution_order
        self.execution_order += 1

        full_text = self.get_original_text_from_context(ctx)
        # In many systems, a carriage return character can cause the cursor to move to the beginning of the line,
        # and subsequent characters will overwrite the characters that were already there.
        cleaned_text = full_text.replace('\r', ' ')

        start_line_number = ctx.start.line
        end_line_number = ctx.stop.line

        # Extract variables and constants
        variables_details = self.cve.extract_constants_variables(cleaned_text, self.path)
        variable_entities = self.create_variable_entity_list(variables_details)

        hosting_function_name = self.get_current_context_name(ctx)

        db_call_entity = DbCallDataEntity(
            name=hosting_function_name,
            file_path=self.path,
            file_name=self.file_name,
            tech=self.projectMetadata.technology,
            program_id=self.file_name,
            start_line_number=start_line_number,
            end_line_number=end_line_number,
            snippet=cleaned_text,
            operation_type=operation_type,
            execution_order=self.execution_order,
            # variables=variables_details,
            variable=variable_entities,
            query=query,
            filter_conditions=filter_conditions,
            aliases=aliases,
            table_names=[table.split('.')[-1].strip('[]').lower() for table in tables]
        )

        # Append the created db call object to its respective function snippet
        if self.current_function:
            self.current_function.db_calls.append(db_call_entity)
        else:
            self.main_program_dbcalls.append(db_call_entity)

        # Below code is to identify order and levels of the db calls
        if len(self.db_call_stack) == 0:
            # Process Parent
            self.db_call_execution_order += 1
            db_call_entity.level = DbCallLevel.PARENT
            db_call_entity.db_call_execution_order = str(self.db_call_execution_order)
            db_call_entity.child_count = 0

        else:
            # I know the last added object is my parent
            parent = self.db_call_stack[-1]
            parent.child_count += 1

            # Process Child
            db_call_entity.level = DbCallLevel.CHILD
            db_call_entity.db_call_execution_order = parent.db_call_execution_order + '.' + str(parent.child_count)
            db_call_entity.child_count = 0

        return db_call_entity

    def get_callee_conditional_hierarchy(self, ctx):

        hierarchy = {}  # Immediate hierarchy
        current_level = hierarchy  # Pointer to the current level in the hierarchy
        is_else = False

        def update_current_level(condition_string, start_line_num, end_line_num, siblings):
            nonlocal current_level, is_else

            condition_string = re.sub(r"\n\d{4}\s*$", '', condition_string)

            # Add current level conditions and prepare for parent
            current_level["condition"] = condition_string
            current_level["start_line_number"] = start_line_num
            current_level["end_line_number"] = end_line_num
            current_level["siblings"] = siblings  # Add string sibling conditions here
            current_level["is_else"] = is_else  # Indicator to suggest whether condition belongs to else part
            current_level["parent"] = {}  # Create a placeholder for parent
            current_level = current_level["parent"]  # Move to the next level

        # Traverse up the parent hierarchy
        while ctx is not None:
            parent_ctx = ctx.parentCtx  # Get the parent context
            if parent_ctx is None:
                break

            # Check if the parent context is an instance of Parser.IfStatement
            if isinstance(parent_ctx, TSqlParser.If_statementContext) and parent_ctx.children:
                if parent_ctx.ELSE():
                    sym = parent_ctx.ELSE().getSymbol()
                    # If stop index of ELSE is coming before start index of context then it belongs to ELSE part
                    if sym.stop < ctx.stop.start:
                        is_else = True

                # Fetch the condition
                condition_ctx = parent_ctx.search_condition() \
                    if hasattr(parent_ctx, 'search_condition') and parent_ctx.search_condition() else None

                if condition_ctx:
                    update_current_level(self.get_original_text_from_context(condition_ctx), parent_ctx.start.line,
                                         parent_ctx.stop.line, [])

                is_else = False

            # Move to the next parent context
            ctx = parent_ctx

        return hierarchy

    def create_callee(self, ctx, construct_type: CalleeConstructType):

        callee = None

        # updating execution orders and line numbers
        self.execution_order = self.execution_order + 1
        self.callee_execution_order = self.callee_execution_order + 1
        start_line_number = ctx.start.line
        end_line_number = ctx.stop.line

        # get the snippet text of callee statement
        snippet_text = self.get_original_text_from_context(ctx)

        # get the conditional hierarchy and other related attributes for the calle
        is_conditional = False
        condition = 'NA'
        conditional_hierarchy = self.get_callee_conditional_hierarchy(ctx)
        if conditional_hierarchy:
            is_conditional = True
            condition = 'ELSE_OF(' + conditional_hierarchy['condition'] + ')' \
                if conditional_hierarchy['is_else'] else conditional_hierarchy['condition']

        target_function_name = None
        variable_entities = []
        parameters = []

        # fetch the target function name, parameters and variables as per the callee construct type
        if construct_type == self.CalleeConstructType.EXECUTE:
            if ctx.execute_body() and ctx.execute_body().func_proc_name_server_database_schema():
                target_function_name = self.parse_func_proc_name_server_database_schema(
                    ctx.execute_body().func_proc_name_server_database_schema()
                )

                # Extract variables and constants
                variables_details = self.cve.extract_constants_variables(snippet_text, self.path)
                variable_entities = self.create_variable_entity_list(variables_details)
                parameters = [var.term for var in variables_details]

        elif construct_type == self.CalleeConstructType.SCALAR_FUNCTION:
            # extract target function name
            if ctx.scalar_function_name():
                sfn_ctx = ctx.scalar_function_name()
                if sfn_ctx.func_proc_name_server_database_schema():
                    target_function_name = self.parse_func_proc_name_server_database_schema(
                        sfn_ctx.func_proc_name_server_database_schema()
                    )

            # extract variable entities and parameters from the expression list
            if ctx.expression_list_():
                variables = self.get_original_text_from_context(ctx.expression_list_())
                # Extract variables and constants
                variables_details = self.cve.extract_constants_variables(variables, self.path)
                variable_entities = self.create_variable_entity_list(variables_details)
                parameters = [var.term for var in variables_details]

        # only create a callee node if value of target function is fetched
        if target_function_name:
            callee = FunctionCallee(
                execution_order=self.execution_order,
                callee_execution_order=self.callee_execution_order,
                start_line_number=start_line_number,
                end_line_number=end_line_number,
                tech=self.projectMetadata.technology,
                caller_function=self.get_current_context_name(ctx),
                callee_function=target_function_name,
                callee_class=self.file_name,
                callee_file_name=self.file_name,
                callee_file_path=self.path,
                snippet=snippet_text,
                callee_type=construct_type.value,
                type_of_construct='External',
                module_name=self.folder_name,
                behavior=CalleeBehavior.RETURNS.value,
                condition=condition,
                variable=variable_entities,
                parameters=parameters,
                is_conditional=is_conditional,
                conditional_hierarchy=conditional_hierarchy
            )

        # Append the created db call object to its respective function snippet
        if callee:
            if self.current_function:
                self.current_function.callees.append(callee)
            else:
                self.main_program_callees.append(callee)

        return callee

    def create_database_entity(self, snippet, table_name, start_line, end_line):
        
        server, database, schema, table = '', '', '', ''

        # table name with dots in it gives different elements, destructuring and assigning them as per the structure
        table_split = table_name.split('.')
        if len(table_split) == 4:
            server, database, schema, table = table_split  # three dots means four elements
        elif len(table_split) == 3:
            database, schema, table = table_split  # two dots means three elements
        elif len(table_split) == 2:
            schema, table = table_split  # one dot means two elements
        else:
            table = table_name  # in case of no dot or more than three dots, use the table name as is

        db_entity = DatabaseEntitySchema(
            name=table.strip('[]').lower(),
            business_aliases=[],
            file_name=self.file_name,
            file_path=self.path,
            module_name=self.folder_name,
            start_line_number=start_line,
            end_line_number=end_line,
            program_id=self.file_name,
            type='TABLE',
            snippet=snippet,
            schema_table=table_name.lower(),
            structure_schema='NA',
            fields=[],
            sample_record='NA',
            complexity='NA',
            volume_class='NA',
            main_role='NA',
            relationship_density='NA',
            db_type='Relational',
            description='NA',
            db_tech=self.projectMetadata.technology,
            hosted_at='NA',
            records_type='NA',
            created_from='Program',
            category='NA',
            servers=([server.strip('[]').lower()] if server else []),
            databases=([database.strip('[]').lower()] if database else []),
            schemas=([schema.strip('[]').lower()] if schema else [])
        )

        # Append the created db call object to its respective function snippet
        if self.current_function:
            self.current_function.database_entities.append(db_entity)
        else:
            self.main_program_database_entities.append(db_entity)

    def visitSelect_statement(self, ctx: TSqlParser.Select_statementContext):
        query = None
        aliases = set()
        tables = set()
        filter_conditions = []
        operation_type = DbCallDataEntity_TYPE.SELECT.value

        try:
            # Fetch the query from the ccontext
            query = self.get_original_text_from_context(ctx).strip(" \n")

            # Fetch the filter condition, table names, aliases from the context
            if ctx.query_expression():
                qe_ctx = ctx.query_expression()

                qs_ctx = qe_ctx.query_specification()
                if qs_ctx.table_sources():
                    for con in qs_ctx.search_condition():
                        if con:
                            filter_conditions.append(self.get_original_text_from_context(con))

                    ts_list = qs_ctx.table_sources().table_source()
                    for ts in ts_list:
                        ret_t, ret_al, ret_fc = self.extract_table_source(ts)
                        tables.update(ret_t)
                        aliases.update(ret_al)
                        filter_conditions.extend(ret_fc)

                if qs_ctx.table_name():
                    tables.add(self.get_original_text_from_context(qs_ctx.table_name()))

                if qe_ctx.sql_union():
                    for union in qe_ctx.sql_union():
                        qs_ctx = union.query_specification()
                        if qs_ctx.table_sources():
                            for con in qs_ctx.search_condition():
                                if con:
                                    filter_conditions.append(self.get_original_text_from_context(con))

                            ts_list = qs_ctx.table_sources().table_source()
                            for ts in ts_list:
                                ret_t, ret_al, ret_fc = self.extract_table_source(ts)
                                tables.update(ret_t)
                                aliases.update(ret_al)
                                filter_conditions.extend(ret_fc)

        except Exception as e:
            logger.error(f"Error while extracting select statement fields: {e}")

        if len(tables) == 0:
            return None
        snippet = self.get_original_text_from_context(ctx)
        table_list=set()
        for table in tables:
            if '.' in table:
                table=table.split('.')[1]
                table_list.add(table)

            table_list.add(table)

            if not self.entity_exists(table.strip()):
                self.create_database_entity(snippet, table.strip(), ctx.start.line, ctx.stop.line)

        db_call = self.create_db_call(ctx, query, filter_conditions, aliases, table_list, operation_type)

        # Adding current dbcall object to stack
        self.db_call_stack.append(db_call)

        self.current_dbcall_depth += 1
        self.visitChildren(ctx)
        self.current_dbcall_depth -= 1

        # Pop the last dbcall object from the stack
        self.db_call_stack.pop()
        fields = self.create_database_field()
        return None

    def create_database_field(self):
        fields = self.column_name
        database_fields = []
        default_table = None
        for key, value in self.table_alias.items():
            if key == value:
                default_table = value
                break
        
        for field_data in fields:
            parts = field_data.split('.')
            if len(parts) == 2:
                table_part = parts[0].strip('[]')
                column_part = parts[1]
                actual_table = self.table_alias.get(table_part)
                if actual_table:
                    actual_table_name = actual_table.split('.')[-1].strip('[]').lower()
                    field_key = (column_part, actual_table_name)
                    if field_key not in self.unique_fields:
                        self.unique_fields.add(field_key)
                        field = DatabaseFieldSchema(
                            name=column_part,
                            table_name=actual_table_name,
                            db_tech=self.projectMetadata.technology
                        )
                        database_fields.append(field)
            elif len(parts) == 1 and default_table:
                column_part = parts[0]
                default_table_name = default_table.split('.')[-1].strip('[]').lower()
                field_key = (column_part, default_table_name)
                if field_key not in self.unique_fields:
                    self.unique_fields.add(field_key)
                    field = DatabaseFieldSchema(
                        name=column_part,
                        table_name=default_table_name,
                        db_tech=self.projectMetadata.technology
                    )
                    database_fields.append(field)
        if self.current_function:
            self.current_function.database_fields.extend(database_fields)
        else:
            self.main_program_database_fields.extend(database_fields)
        self.column_name.clear()
        return database_fields

    def visitInsert_statement(self, ctx: TSqlParser.Insert_statementContext):
        aliases = set()
        tables = set()
        filter_conditions = []
        operation_type = DbCallDataEntity_TYPE.INSERT.value

        query = self.get_original_text_from_context(ctx)

        if ctx.ddl_object():
            tables.add(self.get_original_text_from_context(ctx.ddl_object()))
            for table in tables:
                if not self.entity_exists(table.strip()):
                    self.create_database_entity(query, table.strip(), ctx.start.line, ctx.stop.line)

        elif ctx.rowset_function_limited():
            tables.add(self.get_original_text_from_context(ctx.rowset_function_limited()))

        if len(tables) == 0:
            return None

        db_call = self.create_db_call(ctx, query, filter_conditions, aliases, tables, operation_type)

        # Adding current dbcall object to stack
        self.db_call_stack.append(db_call)

        self.current_dbcall_depth += 1
        self.visitChildren(ctx)
        self.current_dbcall_depth -= 1

        # Pop the last dbcall object from the stack
        self.db_call_stack.pop()
        fields = self.create_database_field()
        return None

    def visitExpression(self, ctx: TSqlParser.ExpressionContext):
        if ctx.full_column_name():
            full_name = ctx.full_column_name().getText()
            # Split the full name which could be in format: [schema].[table].[column] or [table].[column]
            parts = [part.strip('[]') for part in full_name.split('.')]

            if len(parts) >= 2:  # If we have at least table.column
                table_name = parts[-2]  # Second last part is table name
                column_name = parts[-1]  # Last part is always column name
                self.column_name.add(f"{table_name}.{column_name}")
            else:  # If we only have column name
                self.column_name.add(parts[0])
        return self.visitChildren(ctx)

        tables = set()
        filter_conditions = []
        operation_type = DbCallDataEntity_TYPE.UPDATE.value

        query = self.get_original_text_from_context(ctx)

        if ctx.ddl_object():
            tables.add(self.get_original_text_from_context(ctx.ddl_object()))
            for table in tables:
                if not self.entity_exists(table.strip()):
                    self.create_database_entity(query, table.strip(), ctx.start.line, ctx.stop.line)

        elif ctx.rowset_function_limited():
            tables.add(self.get_original_text_from_context(ctx.rowset_function_limited()))

        if ctx.table_sources():
            for ts in ctx.table_sources().table_source():
                ret_t, ret_al, ret_fc = self.extract_table_source(ts)
                tables.update(ret_t)
                aliases.update(ret_al)
                filter_conditions.extend(ret_fc)

        if len(tables) == 0:
            return None

        for table in tables:
            if not self.entity_exists(table.strip()):
                self.create_database_entity(query, table.strip(), ctx.start.line, ctx.stop.line)

        db_call = self.create_db_call(ctx, query, filter_conditions, aliases, tables, operation_type)

        # Adding current dbcall object to stack
        self.db_call_stack.append(db_call)

        self.current_dbcall_depth += 1
        self.visitChildren(ctx)
        self.current_dbcall_depth -= 1

        # Pop the last dbcall object from the stack
        self.db_call_stack.pop()
        fields = self.create_database_field()
        return None

    def visitDelete_statement(self, ctx: TSqlParser.Delete_statementContext):
        aliases = set()

class ConstantVariableExtractor:
    def __init__(self):
        self.keywords = set()

        # load all the keywords of natural from lexer file
        self.read_keywords()
        self.negative_lookahead = '|'.join(re.escape(keyword) for keyword in self.keywords)
        self.variable_pattern = (rf"(?<!@)(?!(?:{self.negative_lookahead})(?=[\s(]|$))" +
                                 rf"\b[A-Z]+(?:[-._][A-Z0-9]+)*\b")
        # rf"|(?:(?<=\s)|(?<=^))[\*#]*\b(?!{self.negative_lookahead}\b)[A-Z]+(?:-[A-Z0-9]+)*\b"
        self.variable_regex = re.compile(self.variable_pattern)
        self.keywords_as_vars = re.compile(rf"(?:(?<=\s)|(?<=^))(@[A-Z][A-Z0-9_]*)(?=[\s(]|$)")
        self.string_constant = r"'([^']*)'"
        self.string_constant_regex = re.compile(self.string_constant)
        self.numeric_constant_regex = re.compile(r"(?<=\s)\d+(?=\s)|(?<=\s)\d+$")
        self.calc_pattern = (
            rf"(?!(?:T|CT|ECT|LECT|ELECT|{self.negative_lookahead})(?=\s|$))"
            # Optional SET or SELECT and variable assignment
            rf"(?:"
            rf"(?:SET|SELECT)?\s*"
            rf"(@[\w]+)?\s*"
            rf"(?:=\s*)?"
            r")?"
            # Require at least one @ as an operand (not just anywhere)
            rf"\(*\s*@[\w.]+(?:\s*[+\-*/]\s*[@\w.()]+)+\s*\)*"
        )
        self.calc_regex = re.compile(self.calc_pattern)
        # Regex pattern to match resultant variables (e.g., @x, @y)
        self.pattern_resultant = r"SET\s+(@\w+)\s+="

    def read_keywords(self):
        # Construct the lexer file path
        current_dir = os.path.dirname(os.path.abspath(__file__))
        target_file_path = os.path.join(current_dir, '..', '..', 'grammer', 'TSqlLexer.g4')
        target_file_path = os.path.abspath(target_file_path)
        logger.debug('Reading the Lexer File: ' + target_file_path)

        start_marker = "// Keywords Start Here"
        end_marker = "// Keywords End Here"
        capture = False

        try:
            with open(target_file_path, 'r') as file:
                for line in file:
                    line = line.strip()
                    if line == start_marker:
                        capture = True
                        continue
                    elif line == end_marker:
                        capture = False
                        break

                    if capture and line and not line.startswith('//'):
                        # Match only part after colon and before semicolon
                        colon_part = re.search(r":(.*?);", line)
                        if colon_part:
                            after_colon = colon_part.group(1)
                            # Extract all quoted strings from after colon part
                            keyword_list = re.findall(r"'[^']*'", after_colon)

                            self.keywords.update([k.strip(" '") for k in keyword_list])
        except Exception as e:
            print("Error while extracting the variables", e)

    def extract_constants_variables(self, code_snippet: str, file_path: str):
        lines = code_snippet.split('\n')
        variable_details = []
        variable_dict = {}

        # Construct the regex pattern with a negative lookahead for each keyword
        for line in lines:
            # Skip commented lines
            if line.strip().startswith('--'):
                continue
            # Extract constants (strings within single quotes) and their corresponding lines
            constants = self.string_constant_regex.findall(line)
            for constant in constants:
                if constant not in variable_dict and len(constant.strip("'").split()) == 1:
                    variable_dict[constant] = DataDefinitionJSON(type='CONSTANT', term=constant,
                                                                 usage_context=[line],
                                                                 file_path=file_path,
                                                                 meaning="")
                elif constant in variable_dict and len(constant.strip("'").split()) == 1:
                    variable_dict[constant].usage_context.append(line)

            # Extract variables (starting with #) and their corresponding lines
            upper_line = line.upper()
            matches = self.variable_regex.finditer(upper_line)
            # Then extract the original form of those matches from the original line
            variables = [line[m.start():m.end()] for m in matches]
            for variable in variables:
                if variable not in variable_dict and len(variable.strip("'").split()) == 1:
                    variable_dict[variable] = DataDefinitionJSON(type='DECLARATION', term=variable,
                                                                 usage_context=[line],
                                                                 file_path=file_path,
                                                                 meaning="")
                elif variable in variable_dict and len(variable.strip("'").split()) == 1:
                    variable_dict[variable].usage_context.append(line)

            # Extract variables (starting with #) which are there in lexer tokens AS keywords
            # First find matches by checking against uppercase version of line
            matches = self.keywords_as_vars.finditer(upper_line)
            # Then extract the original form of those matches from the original line
            keyword_vars = [line[m.start():m.end()] for m in matches]
            for key_var in keyword_vars:
                if key_var not in variable_dict and len(key_var.strip("'").split()) == 1:
                    variable_dict[key_var] = DataDefinitionJSON(type='DECLARATION', term=key_var,
                                                                usage_context=[line],
                                                                file_path=file_path,
                                                                meaning="")
                elif key_var in variable_dict and len(key_var.strip("'").split()) == 1:
                    variable_dict[key_var].usage_context.append(line)

        variable_details += list(variable_dict.values())

        return variable_details
