    def remove_subsections(
            file_content: str, subsections: List[FunctionSnippetSchema]

    def parse_view_file(self, content):
        content = content.replace("ï»¿", "")

        # Remove comments
        content = re.sub(r'/\*.*?\*/', '', content, flags=re.DOTALL)
        content = re.sub(r'--.*', '', content)

        # Extract view name
        view_name_match = re.search(r'CREATE\s+VIEW\s+(?:\[[^\]]+\]\.)?(?:dbo\.)?\[?([^\s\[\]]+)\]?', content, re.IGNORECASE)
        if view_name_match is None:
            return None, None, None, None
        view_name = view_name_match.group(1)

        # Extract fields between SELECT and FROM (simplified)
        select_from_match = re.search(r"(?s)(?<=SELECT\s)(.*?)(?=\s+FROM\b)", content, re.IGNORECASE | re.DOTALL)
        select_from_match1 = re.search(r"(?s)(?<=SELECT\s)(.*?)(?=\s+FROM\b)", content)
        fields_str=""
        fields = set()
        if select_from_match:
            fields_str = select_from_match.group(1)
            fields.update(self.get_view_fields(fields_str))
            # Split by commas not inside parentheses, quotes, or CASE ... END
        if select_from_match1:
            fields_str = select_from_match1.group(1)
            fields.update(self.get_view_fields(fields_str))

        # Extract tables from FROM and JOIN clauses
        tables = set()
        # Match table names after FROM or JOIN
        table_pattern = re.compile(r'(?:FROM|JOIN)\s+([^\s,]+)', re.IGNORECASE)
        for match in table_pattern.finditer(content):
            table = match.group(1)
            # Clean aliases and brackets
            table = re.split(r'\s+AS\s+|\s+', table, flags=re.IGNORECASE)[0]
            table_clean = table.replace('[', '').replace(']', '')
            # Skip if it's a subquery or variable
            table_name = table_clean.split('.')[-1]
            if not table_name.startswith('$'):
                tables.add(table_name)  # Get table name after schema.
       
        return  view_name,list(fields),list(tables),fields_str
    def get_view_fields(self,fields_str):
        fields = set()
        sql_reserved_keywords = [
            "SELECT", "FROM", "WHERE", "INSERT", "UPDATE", "DELETE", "JOIN", "INNER", "LEFT", "RIGHT", "FULL",
            "GROUP", "ORDER", "BY", "HAVING", "DISTINCT", "AS", "AND", "OR", "NOT", "NULL", "IS", "IN", "BETWEEN",
            "LIKE", "CREATE", "DROP", "ALTER", "TABLE", "DATABASE", "INDEX", "VIEW", "TRIGGER", "PRIMARY", "KEY",
            "FOREIGN", "CONSTRAINT", "DEFAULT", "CHECK", "EXISTS", "UNION", "ALL", "ANY", "CASE", "WHEN", "THEN",
            "ELSE", "END", "LIMIT", "OFFSET", "VALUES", "SET", "TOP", "ASC", "DESC", "TRANSACTION", "COMMIT",
            "ROLLBACK", "GRANT", "REVOKE", "WITH", "NOLOCK", "ON", "OUTER", "CROSS", "NATURAL", "USING", "INNER JOIN",
        ]
        raw_fields = self.split_fields(fields_str)
        for f in raw_fields:
                # Extract field name: if AS alias, take alias, else the expression cleaned
                # Use regex to find alias after AS (case insensitive)
            alias_matches = re.findall(r'AS\s+([\w_]+)', f, re.IGNORECASE)
                
            if alias_matches:
                    # Handle multiple aliases separated by commas (e.g., "AS Chng, HS_Updt_Dt AS ChngDt, REPLACE(...) AS ChngBy")
                    # Split by comma and take last alias
                last_alias = alias_matches[-1].strip().replace('[', '').replace(']', '')
                    # Fix for known aliases like Chng, ChngDt, ChngBy
                if last_alias.lower() == 'changed':
                    last_alias = 'Chng'
                elif last_alias.lower() == 'chngdt':
                    last_alias = 'ChngDt'
                elif last_alias.lower() == 'chngby':
                    last_alias = 'ChngBy'
                field_name = last_alias
            else:
                    # Clean brackets and take last part if dotted
                field_name = f.replace('[', '').replace(']', '').split('.')[-1].strip()
            for sql_key in sql_reserved_keywords:
                if sql_key+" " in field_name.upper() or " "+sql_key in field_name.upper():
                    field_name = re.sub(sql_key, "", field_name, flags=re.IGNORECASE)  # replace
            fields.add(field_name.strip())
        return fields
        
    def split_fields(self,s):
        fields_list = []
        current = ''
        paren_count = 0
        case_count = 0
        in_quote = False
        i = 0
        while i < len(s):
            char = s[i]
            if char == "'":
                in_quote = not in_quote
                current += char
            elif char == '(' and not in_quote:
                paren_count += 1
                current += char
            elif char == ')' and not in_quote:
                paren_count -= 1
                current += char
            elif s[i:i+4].upper() == 'CASE' and not in_quote:
                case_count += 1
                current += char
            elif s[i:i+3].upper() == 'END' and case_count > 0 and not in_quote:
                case_count -= 1
                current += char
            elif char == ',' and paren_count == 0 and case_count == 0 and not in_quote:
                fields_list.append(current.strip())
                current = ''
            else:
                current += char
            i += 1
        if current.strip():
            fields_list.append(current.strip())
        return fields_list
  

    def delete_entities_if_found(self,name, project_id, run_id):
        DELETE_QUERY = """
                    MATCH (e:DatabaseEntity)
                    WHERE toLower(e.name) = toLower($name) 
                    AND e.project_id = $project_id 
                    AND e.run_id = $run_id
                    OPTIONAL MATCH (e)-[:HAS_FIELD]->(f:DatabaseField)
                    DETACH DELETE e, f
                    """

        # DELETE_QUERY = """
        #     MATCH (e:DatabaseEntity {name: $name, project_id: $project_id, run_id: $run_id})
        #     OPTIONAL MATCH (e)-[:HAS_FIELD]->(f:DatabaseField)
        #     DETACH DELETE e, f
        #     """
        db.cypher_query(DELETE_QUERY,
        {'name': name, 'project_id': project_id, 'run_id': run_id})
    
    def create_node_database_entity(self, table_name, field_list, fields, project_id, run_id, file_path, snippet,db_tech,source_tables=[],type="TABLE"):
        self.delete_entities_if_found(table_name,project_id,run_id)
        query = """
            CREATE (d:DatabaseEntity {key:$key,
                analysis_status:$analysis_status,
                name: $name,
                business_aliases: $business_aliases,
                project_id: $project_id,
                run_id: $run_id,
                file_name: $file_name,
                file_path: $file_path,
                schema_table: $schema_table,
                snippet: $snippet,
                structure_schema: $structure_schema,
                fields: $fields,
                type: $type,
                sample_record: $sample_record,
                complexity: $complexity,
                volume_class: $volume_class,
                main_role: $main_role,
                relationship_density: $relationship_density,
                db_type: $db_type,
                description: $description,
                db_tech: $db_tech,
                hosted_at: $hosted_at,
                records_type: $records_type,
                category: $category,
                source_tables: $source_tables

                """

        db.cypher_query(query, {'key': str(uuid.uuid4()),
                                'analysis_status': "NOT_STARTED",

                                'project_id': project_id,
                                'run_id': run_id,
                                'file_name': os.path.splitext(os.path.basename(file_path))[0],
                                'file_path': file_path,

                                'snippet': snippet,
                                'structure_schema': fields,
                                'fields': field_list,
                                'type': type,
                                'sample_record': "NA",
                                'complexity': "NA",
                                'volume_class': "NA",
                                'main_role': "NA",
                                'relationship_density': "NA",
                                'db_type': "Relational",
                                'description': "NA",
                                'db_tech': db_tech,
                                'hosted_at': "NA",
                                'records_type': "NA",
                                'category': "NA",
                                "source_tables": source_tables})
        for db_field in field_list:
            self.generate_database_field(db_field, table_name, project_id, run_id,db_tech)
            
    def generate_database_field(self, db_field, table_name, project_id, run_id,db_tech):
        DatabaseField(
            key=str(uuid.uuid4()),
            name=db_field,
            table_name=table_name,
            description="",
            project_id=project_id,
            run_id=run_id,
            db_tech=db_tech
        ).save()

    def create_has_field_relationships(self, project_id, run_id):
        query = """
        MATCH (de:DatabaseEntity {run_id: $run_id, project_id: $project_id})
        MATCH (df:DatabaseField {run_id: $run_id, project_id: $project_id})
        WHERE df.table_name = de.name
        MERGE (de)-[:HAS_FIELD]->(df)
        """
        db.cypher_query(query, {'run_id': run_id, 'project_id': project_id})
