    /// <summary>
    /// C# port of services/codeminer/tsql/visitor/extractors/constant_variable_extractor.ConstantVariableExtractor.
    /// Extracts constants, variables, and mathematical calculations from T-SQL code snippets.
    /// </summary>
    public class ConstantVariableExtractor
    {
        // Static cache to avoid re-reading/compiling per-file
        private static readonly HashSet<string> keywords = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
        private static readonly string negative_lookahead;
        private static readonly string variable_pattern;
        private static readonly Regex variable_regex;
        private static readonly Regex keywords_as_vars;
        private static readonly Regex string_constant_regex;
        private static readonly Regex numeric_constant_regex;
        private static readonly Regex calc_regex;
        private static readonly string pattern_resultant;
        private static readonly object keywordsLock = new object();
        private static bool keywordsInitialized = false;

        static ConstantVariableExtractor()
        {
            // Static constructor runs once per app domain
            lock (keywordsLock)
            {
                if (!keywordsInitialized)
                {
                    LoadKeywordsAndCompileRegexes();
                    keywordsInitialized = true;
                }
            }
        }

        private static void LoadKeywordsAndCompileRegexes()
        {
            try
            {
                // Read keywords from lexer file once
                string currentDir = Path.GetDirectoryName(System.Reflection.Assembly.GetExecutingAssembly().Location);
                string targetFilePath = Path.Combine(currentDir, "..", "..", "..", "grammer", "TSqlLexer.g4");
                targetFilePath = Path.GetFullPath(targetFilePath);

                string startMarker = "// Keywords Start Here";
                string endMarker = "// Keywords End Here";
                bool capture = false;

                if (File.Exists(targetFilePath))
                {
                    foreach (string line in File.ReadLines(targetFilePath))
                    {
                        string trimmedLine = line.Trim();

                        if (trimmedLine == startMarker)
                        {
                            capture = true;
                            continue;
                        }
                        else if (trimmedLine == endMarker)
                        {
                            capture = false;
                            break;
                        }

                        if (capture && !string.IsNullOrEmpty(trimmedLine) && !trimmedLine.StartsWith("//"))
                        {
                            Match colonMatch = Regex.Match(trimmedLine, @":(.* ?);");
                            if (colonMatch.Success)
                            {
                                string afterColon = colonMatch.Groups[1].Value;
                                MatchCollection keywordMatches = Regex.Matches(afterColon, @"'[^']*'");
                                foreach (Match keywordMatch in keywordMatches)
                                {
                                    string keyword = keywordMatch.Value.Trim(" '" .ToCharArray());
                                    keywords.Add(keyword);
                                }
                            }
                        }
                    }
                }

                // Build negative lookahead pattern
                negative_lookahead = string.Join("|", keywords.Select(k => Regex.Escape(k)));

                // Compile all regexes once with RegexOptions.Compiled for performance
                variable_pattern = $@"(?<!@)(?!(?:{negative_lookahead})(?=[\s(]|$))\b[A-Z]+(?:[-._][A-Z0-9]+)*\b";
                variable_regex = new Regex(variable_pattern, RegexOptions.IgnoreCase | RegexOptions.Compiled);

                keywords_as_vars = new Regex(
                    $@"(?:(?<=\s)|(?<=^))(@[A-Z][A-Z0-9_]*)(?=[\s(]|$)",
                    RegexOptions.IgnoreCase | RegexOptions.Compiled
                );

                string_constant_regex = new Regex(@"'([^']*)'", RegexOptions.Compiled);

                numeric_constant_regex = new Regex(@"(?<=\s)\d+(?=\s)|(?<=\s)\d+$", RegexOptions.Compiled);

                string calcPattern = (
                    $@"(?!(?:T|CT|ECT|LECT|ELECT|{negative_lookahead})(?=\s|$))" +
                    $@"(?:" +
                    $@"(?:SET|SELECT)?\s*" +
                    $@"(@[\w]+)?\s*" +
                    $@"(?:=\s*)?" +
                    $@")?" +
                    $@"\(*\s*@[\w.]+(?:\s*[+\-*/]\s*[@\w.()]+)+\s*\)*"
                );
                calc_regex = new Regex(calcPattern, RegexOptions.IgnoreCase | RegexOptions.Compiled);

                pattern_resultant = @"SET\s+(@\w+)\s+=";
            }
            catch
            {
                // Swallow exception - non-fatal
            }
        }

        /// <summary>
        /// Extract mathematical calculations from code snippet
        /// </summary>
        public List<Dictionary<string, object>> extract_mathematical_calculations(string code_snippet)
        {
            var matched_list = new List<Dictionary<string, object>>();
            try
            {
                foreach (Match match in calc_regex.Matches(code_snippet))
                {
                    var data_dict = new Dictionary<string, object>
                    {
                        { "match_group", match.Value },
                        { "start", match.Index },
                        { "end", match.Index + match.Length }
                    };
                    matched_list.Add(data_dict);
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine("Error in extract_mathematical_calculations: " + ex.Message);
            }
            return matched_list;
        }

        /// <summary>
        /// Extract mathematical calculations with start line numbers
        /// </summary>
        public List<Dictionary<string, object>> extract_mathematical_calculations_startline(string code_snippet)
        {
            var matched_list = new List<Dictionary<string, object>>();
            string[] lines = code_snippet.Split('\n');  // Match Python: split('\n') after normalization
            try
            {
                for (int i = 0; i < lines.Length; i++)
                {
                    string line = lines[i];
                    foreach (Match match in calc_regex.Matches(line))
                    {
                        var data_dict = new Dictionary<string, object>
                        {
                            { "match_group", match.Value },
                            { "start", i + 1 }  // 1-based line number
                        };
                        matched_list.Add(data_dict);
                    }
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine("Error in extract_mathematical_calculations: " + ex.Message);
            }
            return matched_list;
        }

        /// <summary>
        /// Extract constants and variables from code snippet
        /// </summary>
        public List<DataDefinitionJSON> extract_constants_variables(string code_snippet, string file_path)
        {
            string[] lines = code_snippet.Split('\n');
            var variable_details = new List<DataDefinitionJSON>();
            // Use case-insensitive keys to collapse VAR/Var/var duplicates
            var variable_dict = new Dictionary<string, DataDefinitionJSON>(StringComparer.OrdinalIgnoreCase);

            foreach (string line in lines)
            {
                // Skip commented lines
                if (line.Trim().StartsWith("--"))
                    continue;

                // Extract string constants
                MatchCollection string_matches = string_constant_regex.Matches(line);
                foreach (Match match in string_matches)
                {
                    string constant = match.Groups[1].Value;
                    string key = constant; // case-insensitive dict will normalize keys
                    if (!variable_dict.ContainsKey(key) && constant.Trim('\'').Split(' ').Length == 1)
                    {
                        variable_dict[key] = new DataDefinitionJSON(
                            type: "CONSTANT",
                            term: constant,
                            usage_context: new List<string> { line },
                            file_path: file_path,
                            meaning: ""
                        );
                    }
                    else if (variable_dict.ContainsKey(key))
                    {
                        variable_dict[key].usage_context.Add(line);
                    }
                }

                // Extract variables (uppercase identifiers) - use regex directly on line (IgnoreCase handles matching)
                MatchCollection var_matches = variable_regex.Matches(line);
                var variables = new List<string>();
                foreach (Match match in var_matches)
                {
                    // Use the matched value exactly (preserves case), but dictionary is case-insensitive
                    variables.Add(match.Value);
                }

                foreach (string variable in variables)
                {
                    string key = variable;
                    if (!variable_dict.ContainsKey(key) && variable.Trim('\'').Split(' ').Length == 1)
                    {
                        variable_dict[key] = new DataDefinitionJSON(
                            type: "DECLARATION",
                            term: variable,
                            usage_context: new List<string> { line },
                            file_path: file_path,
                            meaning: ""
                        );
                    }
                    else if (variable_dict.ContainsKey(key))
                    {
                        variable_dict[key].usage_context.Add(line);
                    }
                }

                // Extract keyword variables (@VAR_NAME) - use regex directly on line
                MatchCollection keyword_var_matches = keywords_as_vars.Matches(line);
                var keyword_vars = new List<string>();
                foreach (Match match in keyword_var_matches)
                {
                    keyword_vars.Add(match.Value);
                }

                foreach (string key_var in keyword_vars)
                {
                    string key = key_var;
                    if (!variable_dict.ContainsKey(key) && key_var.Trim('\'').Split(' ').Length == 1)
                    {
                        variable_dict[key] = new DataDefinitionJSON(
                            type: "DECLARATION",
                            term: key_var,
                            usage_context: new List<string> { line },
                            file_path: file_path,
                            meaning: ""
                        );
                    }
                    else if (variable_dict.ContainsKey(key))
                    {
                        variable_dict[key].usage_context.Add(line);
                    }
                }
            }

            variable_details.AddRange(variable_dict.Values);
            return variable_details;
        }

        /// <summary>
        /// Extract numeric constants from code snippet
        /// </summary>
        public List<DataDefinitionJSON> extract_numeric_constants(string code_snippet, string file_path)
        {
            string[] lines = code_snippet.Split('\n');
            var variable_details = new List<DataDefinitionJSON>();
            // Use case-insensitive keys
            var variable_dict = new Dictionary<string, DataDefinitionJSON>(StringComparer.OrdinalIgnoreCase);

            foreach (string line in lines)
            {
                MatchCollection constant_matches = numeric_constant_regex.Matches(line);
                foreach (Match match in constant_matches)
                {
                    string constant = match.Value;
                    string key = constant;
                    if (!variable_dict.ContainsKey(key) && constant.Trim('\'').Split(' ').Length == 1)
                    {
                        variable_dict[key] = new DataDefinitionJSON(
                            type: "CONSTANT",
                            term: constant,
                            usage_context: new List<string> { line },
                            file_path: file_path,
                            meaning: ""
                        );
                    }
                    else if (variable_dict.ContainsKey(key))
                    {
                        variable_dict[key].usage_context.Add(line);
                    }
                }
            }

            variable_details.AddRange(variable_dict.Values);
            return variable_details;
        }

    /// <summary>
    /// Data definition JSON schema for variables/constants
    /// </summary>
    public class DataDefinitionJSON
    {
        public string Type { get; set; }
        public string Term { get; set; }
        public List<string> UsageContext { get; set; }
        public string FilePath { get; set; }
        public string Meaning { get; set; }

        public DataDefinitionJSON(string type, string term, List<string> usage_context, string file_path, string meaning)
        {
            Type = type;
            Term = term;
            UsageContext = usage_context ?? new List<string>();
            FilePath = file_path;
            Meaning = meaning;
        }

        // Property accessors for compatibility (lowercase properties)
        public string type
        {
            get => Type;
            set => Type = value;
        }

        public string term
        {
            get => Term;
            set => Term = value;
        }

        public List<string> usage_context
        {
            get => UsageContext;
            set => UsageContext = value;
        }

        public string file_path
        {
            get => FilePath;
            set => FilePath = value;
        }

        public string meaning
        {
            get => Meaning;
            set => Meaning = value;
        }
    }
