using System;
using System.Collections;
using System.Collections.Generic;
using System.IO;
using Newtonsoft.Json;
using ClosedXML.Excel;

namespace TsqlParserProject
{
    public static class ParserUtils
    {
        public static void WriteFunctionSnippetsToExcelWithCounts(
            CodeContext output,
            string inputFilePath,
            string saveDirectory)
        {
            if (output == null)
                throw new ArgumentNullException(nameof(output));

            if (inputFilePath == null)
                throw new ArgumentNullException(nameof(inputFilePath));

            if (saveDirectory == null)
                throw new ArgumentNullException(nameof(saveDirectory));

            // ---- EXACT PYTHON LOGIC ----

            string baseFilename = Path.GetFileNameWithoutExtension(inputFilePath);
            string timestamp = DateTime.Now.ToString("ddMMyyyy_HHmmss");
            string excelFilename = $"{baseFilename}_Parser_Results_{timestamp}.xlsx";
            string excelPath = Path.Combine(saveDirectory, excelFilename);

            var workbook = new XLWorkbook();

            // ---- EXACT PYTHON FIELD STRUCTURE (snake_case) ----

            var fields = new Dictionary<string, List<string>>
            {
                { "decision_node_constructs", new List<string>{ "file_name", "name", "start_line_number", "end_line_number" } },
                { "loop_entity", new List<string>{ "file_name", "name", "start_line_number", "end_line_number" } },
                { "db_calls", new List<string>{ "file_name", "function_name", "start_line_number", "end_line_number", "operation_type", "aliases", "table_names", "filter_conditions" } },
                { "database_entities", new List<string>{ "name", "file_name", "start_line_number", "end_line_number" } },
                { "calculation", new List<string>{ "file_name", "name", "start_line_number", "end_line_number", "snippet", "operands", "variable" } },
                { "callees", new List<string>{ "callee_file_name", "caller_function", "snippet", "start_line_number", "end_line_number", "callee_function", "condition", "conditional_hierarchy", "is_conditional", "parameters" } },
                { "service_calls", new List<string>{ "file_name", "name", "start_line_number", "end_line_number", "snippet", "target", "parameters" } },
                { "transaction_commits", new List<string>{ "file_name", "name", "start_line_number", "end_line_number", "snippet" } },
                { "transaction_rollbacks", new List<string>{ "file_name", "name", "start_line_number", "end_line_number", "snippet" } }
            };

            // ---- WRITE EACH FIELD SHEET ----

            foreach (var fieldEntry in fields)
            {
                string fieldName = fieldEntry.Key;
                var properties = fieldEntry.Value;

                var sheet = workbook.Worksheets.Add(fieldName);

                // Header row (row 1)
                for (int i = 0; i < properties.Count; i++)
                {
                    sheet.Cell(1, i + 1).Value = properties[i];
                }

                int rowIndex = 2;

                foreach (var functionSnippet in output.FunctionSnippets)
                {
                    var fieldObjects = GetPropertyValue(functionSnippet, fieldName) as IEnumerable;

                    if (fieldObjects == null)
                        continue;

                    foreach (var obj in fieldObjects)
                    {
                        for (int colIndex = 0; colIndex < properties.Count; colIndex++)
                        {
                            string prop = properties[colIndex];
                            var value = GetPropertyValue(obj, prop);

                            if (prop == "conditional_hierarchy")
                            {
                                sheet.Cell(rowIndex, colIndex + 1).Value =
                                    JsonConvert.SerializeObject(value ?? new object(), Formatting.Indented);
                            }
                            else
                            {
                                sheet.Cell(rowIndex, colIndex + 1).Value =
                                    value?.ToString() ?? "";
                            }
                        }

                        rowIndex++;
                    }
                }
            }

            // ---- SUMMARY SHEET (EXACT PYTHON STRUCTURE) ----

            var summarySheet = workbook.Worksheets.Add("Summary");

            var summaryHeaders = new List<string>
            {
                "file_name", "function_name", "start_line_number", "end_line_number",
                "decision_node_constructs_count", "loop_entity_count", "db_calls_count",
                "calculation_count", "callees_count"
            };

            for (int i = 0; i < summaryHeaders.Count; i++)
            {
                summarySheet.Cell(1, i + 1).Value = summaryHeaders[i];
            }

            int summaryRowIndex = 2;

            foreach (var functionSnippet in output.FunctionSnippets)
            {
                var fileName = GetPropertyValue(functionSnippet, "file_name");
                var functionName = GetPropertyValue(functionSnippet, "function_name");
                var startLineNumber = GetPropertyValue(functionSnippet, "start_line_number");
                var endLineNumber = GetPropertyValue(functionSnippet, "end_line_number");

                int decisionCount = GetCount(functionSnippet, "decision_node_constructs");
                int loopCount = GetCount(functionSnippet, "loop_entity");
                int dbCallCount = GetCount(functionSnippet, "db_calls");
                int calcCount = GetCount(functionSnippet, "calculation");
                int calleeCount = GetCount(functionSnippet, "callees");

                summarySheet.Cell(summaryRowIndex, 1).Value = fileName?.ToString() ?? "";
                summarySheet.Cell(summaryRowIndex, 2).Value = functionName?.ToString() ?? "";
                summarySheet.Cell(summaryRowIndex, 3).Value = startLineNumber?.ToString() ?? "";
                summarySheet.Cell(summaryRowIndex, 4).Value = endLineNumber?.ToString() ?? "";
                summarySheet.Cell(summaryRowIndex, 5).Value = decisionCount;
                summarySheet.Cell(summaryRowIndex, 6).Value = loopCount;
                summarySheet.Cell(summaryRowIndex, 7).Value = dbCallCount;
                summarySheet.Cell(summaryRowIndex, 8).Value = calcCount;
                summarySheet.Cell(summaryRowIndex, 9).Value = calleeCount;

                summaryRowIndex++;
            }

            // ---- REMOVE DEFAULT SHEET (MATCH PYTHON LOGIC) ----

            if (workbook.Worksheets.Contains("Sheet"))
            {
                workbook.Worksheet("Sheet").Delete();
            }

            workbook.SaveAs(excelPath);
        }

        private static object GetPropertyValue(object obj, string propertyName)
        {
            if (obj == null)
                return null;

            var prop = obj.GetType().GetProperty(propertyName);
            if (prop != null)
                return prop.GetValue(obj);

            var field = obj.GetType().GetField(propertyName);
            if (field != null)
                return field.GetValue(obj);

            return null;
        }

        private static int GetCount(object obj, string propertyName)
        {
            var value = GetPropertyValue(obj, propertyName) as IEnumerable;
            if (value == null)
                return 0;

            int count = 0;
            foreach (var _ in value)
                count++;

            return count;
        }
    }
}
