class ParserStandardizer:
    """Python version of ParserStandardizer for verification"""
    
    @staticmethod
    def standardize_file_path(file_path: str) -> str:
        if not file_path:
            return file_path
        
        standardized = file_path.replace("\\", "/")
        
        if re.match(r'^[A-Za-z]:', standardized):
            drive = standardized[0].lower()
            path_parts = standardized.split('/')
            if len(path_parts) > 2:
                for part in path_parts:
                    if '_' in part and len(part) > 10:
                        standardized = f"/mnt/storage/{part}" + standardized[standardized.find(part) + len(part):]
                        break
                else:
                    standardized = f"/mnt/storage/{drive}" + standardized[2:]
            else:
                standardized = f"/mnt/storage/{drive}" + standardized[2:]
        
        return standardized
    
    @staticmethod
    def generate_deterministic_key(file_path: str, function_name: str, class_name: str, 
                                 start_line: int, end_line: int, node_type: str = "") -> str:
        hash_data = {
            'file_path': ParserStandardizer.standardize_file_path(file_path),
            'function_name': function_name or "",
            'class_name': class_name or "",
            'start_line': start_line,
            'end_line': end_line,
            'node_type': node_type
        }
        
        hash_string = json.dumps(hash_data, sort_keys=True)
        hash_bytes = hashlib.sha256(hash_string.encode()).digest()
        return str(uuid.UUID(bytes=hash_bytes[:16]))
    
    @staticmethod
    def generate_deterministic_timestamp(file_path: str, start_line: int, end_line: int) -> str:
        hash_data = f"{ParserStandardizer.standardize_file_path(file_path)}{start_line}{end_line}"
        hash_value = hashlib.md5(hash_data.encode()).hexdigest()
        base_timestamp = 1704067200.0  # 2024-01-01 00:00:00 UTC
        hash_offset = int(hash_value[:8], 16) % 86400
        timestamp = base_timestamp + hash_offset
        return datetime.fromtimestamp(timestamp).strftime("%Y-%m-%d %H:%M:%S")
    
    @staticmethod
    def get_standardized_project_id(project_id: int = None) -> int:
        return project_id if project_id is not None else None
    
    @staticmethod
    def generate_deterministic_run_file_id(file_path: str, project_id: int) -> int:
        hash_data = f"{ParserStandardizer.standardize_file_path(file_path)}{project_id}"
        hash_value = hashlib.md5(hash_data.encode()).hexdigest()
        base_id = 93000
        hash_offset = int(hash_value[:6], 16) % 200
        return base_id + hash_offset

class Neo4jMultiNodeComparator:
    def __init__(self, uri, username, password):
        self.driver = GraphDatabase.driver(uri, auth=(username, password))
    
    def close(self):
        self.driver.close()
    
    def get_nodes_by_run_id_and_type(self, run_id, node_type):
        """Get all nodes for a specific run_id and node type"""
        query = f"""
        MATCH (n:{node_type}) 
        WHERE n.run_id = $run_id 
        RETURN n, labels(n) as node_labels, id(n) as node_id
        """
        
        with self.driver.session() as session:
            result = session.run(query, run_id=run_id)
            nodes = []
            for record in result:
                node_data = dict(record['n'])
                node_data['_labels'] = record['node_labels']
                node_data['_node_id'] = record['node_id']
                nodes.append(node_data)
            return nodes
    
    def normalize_node_for_comparison(self, node):
        """Normalize node data for comparison, excluding run_id"""
        normalized = {}
        for key, value in node.items():
            if key not in ['run_id', '_node_id']:
                if isinstance(value, (list, dict)):
                    normalized[key] = json.dumps(value, sort_keys=True)
                else:
                    # Don't truncate 'key' property values
                    if key.lower() == 'key':
                        normalized[key] = str(value) if value is not None else ''
                    else:
                        normalized[key] = str(value) if value is not None else ''
        return normalized
    
    def find_node_matches_and_mismatches(self, nodes1, nodes2, node_type):
        """Find exact node matches and mismatches based on key properties"""
        matched_nodes = []
        mismatched_nodes = []
        missing_in_run2 = []
        missing_in_run1 = []
        
        # Create lookup dictionaries based on key properties
        nodes1_lookup = {}
        nodes2_lookup = {}
        
        # Define key properties for matching based on node type
        key_props = ['file_path', 'function_name', 'class_name', 'start_line_number', 'end_line_number', 'name']
        
        for node in nodes1:
            key = self._create_node_key(node, key_props)
            nodes1_lookup[key] = node
        
        for node in nodes2:
            key = self._create_node_key(node, key_props)
            nodes2_lookup[key] = node
        
        # Find matches and mismatches
        for key, node1 in nodes1_lookup.items():
            if key in nodes2_lookup:
                node2 = nodes2_lookup[key]
                # Compare all properties
                norm1 = self.normalize_node_for_comparison(node1)
                norm2 = self.normalize_node_for_comparison(node2)
                
                if norm1 == norm2:
                    matched_nodes.append({'node1': node1, 'node2': node2, 'match_key': key})
                else:
                    # Find specific property differences
                    prop_diffs = {}
                    all_props = set(norm1.keys()) | set(norm2.keys())
                    for prop in all_props:
                        val1 = norm1.get(prop, 'MISSING')
                        val2 = norm2.get(prop, 'MISSING')
                        if val1 != val2:
                            prop_diffs[prop] = {'run1_value': val1, 'run2_value': val2}
                    
                    mismatched_nodes.append({
                        'node1': node1, 
                        'node2': node2, 
                        'match_key': key,
                        'property_differences': prop_diffs
                    })
            else:
                missing_in_run2.append(node1)
        
        # Find nodes only in run2
        for key, node2 in nodes2_lookup.items():
            if key not in nodes1_lookup:
                missing_in_run1.append(node2)
        
        return {
            'matched_nodes': matched_nodes,
            'mismatched_nodes': mismatched_nodes,
            'missing_in_run2': missing_in_run2,
            'missing_in_run1': missing_in_run1
        }
    
    def _create_node_key(self, node, key_props):
        """Create a unique key for node matching"""
        key_parts = []
        for prop in key_props:
            value = node.get(prop, '')
            if value is None:
                value = ''
            key_parts.append(str(value))
        return '|'.join(key_parts)

    def compare_node_type(self, run_id1, run_id2, node_type):
        """Compare nodes of a specific type between two run IDs"""
        print(f"Comparing {node_type} nodes...")
        
        nodes1 = self.get_nodes_by_run_id_and_type(run_id1, node_type)
        nodes2 = self.get_nodes_by_run_id_and_type(run_id2, node_type)
        
        print(f"  Run {run_id1}: {len(nodes1)} {node_type} nodes")
        print(f"  Run {run_id2}: {len(nodes2)} {node_type} nodes")
        
        # Find detailed node matches and mismatches
        node_analysis = self.find_node_matches_and_mismatches(nodes1, nodes2, node_type)
        
        # Get all properties and their values from both runs
        all_props_run1 = set()
        all_props_run2 = set()
        prop_values_run1 = {}
        prop_values_run2 = {}
        
        for node in nodes1:
            normalized = self.normalize_node_for_comparison(node)
            all_props_run1.update(normalized.keys())
            for prop, value in normalized.items():
                if prop not in prop_values_run1:
                    prop_values_run1[prop] = set()
                # Don't truncate 'key' property values
                if prop.lower() == 'key':
                    prop_values_run1[prop].add(str(value))
                else:
                    prop_values_run1[prop].add(value[:100] if len(str(value)) > 100 else str(value))
        
        for node in nodes2:
            normalized = self.normalize_node_for_comparison(node)
            all_props_run2.update(normalized.keys())
            for prop, value in normalized.items():
                if prop not in prop_values_run2:
                    prop_values_run2[prop] = set()
                # Don't truncate 'key' property values
                if prop.lower() == 'key':
                    prop_values_run2[prop].add(str(value))
                else:
                    prop_values_run2[prop].add(value[:100] if len(str(value)) > 100 else str(value))
        
        # Find property differences
        missing_in_run2 = all_props_run1 - all_props_run2
        missing_in_run1 = all_props_run2 - all_props_run1
        common_props = all_props_run1 & all_props_run2
        
        # Find value differences in common properties
        value_differences = {}
        for prop in common_props:
            values1 = prop_values_run1.get(prop, set())
            values2 = prop_values_run2.get(prop, set())
            if values1 != values2:
                value_differences[prop] = {
                    'run1_values': list(values1),
                    'run2_values': list(values2)
                }
        
        return {
            'node_type': node_type,
            'count_run1': len(nodes1),
            'count_run2': len(nodes2),
            'total_props_run1': len(all_props_run1),
            'total_props_run2': len(all_props_run2),
            'missing_in_run2': list(missing_in_run2),
            'missing_in_run1': list(missing_in_run1),
            'common_props': list(common_props),
            'missing_count_run2': len(missing_in_run2),
            'missing_count_run1': len(missing_in_run1),
            'prop_values_run1': {k: list(v) for k, v in prop_values_run1.items()},
            'prop_values_run2': {k: list(v) for k, v in prop_values_run2.items()},
            'value_differences': value_differences,
            'node_analysis': node_analysis
        }
    
    def compare_all_node_types(self, run_id1, run_id2, node_types):
        """Compare all specified node types"""
        results = []
        
        for node_type in node_types:
            try:
                result = self.compare_node_type(run_id1, run_id2, node_type)
                results.append(result)
            except Exception as e:
                print(f"Error comparing {node_type}: {str(e)}")
                results.append({
                    'node_type': node_type,
                    'error': str(e),
                    'count_run1': 0,
                    'count_run2': 0,
                    'missing_in_run2': [],
                    'missing_in_run1': []
                })
        
        return results

            if 'error' not in result and 'node_analysis' in result:
                node_analysis = result['node_analysis']
                
                # Count differences - Extra nodes in each parser
                if result['count_run1'] != result['count_run2']:
                    # Extra nodes in DotNet parser (run1)
                    for extra_node in node_analysis['missing_in_run2']:

                            'Node_Type': result['node_type'],
                            'Extra_In_Parser': 'DotNet Parser',

                            'File_Path': extra_node.get('file_path', 'N/A'),
                            'Function_Name': extra_node.get('function_name', 'N/A'),
                            'Class_Name': extra_node.get('class_name', 'N/A'),
                            'Start_Line': extra_node.get('start_line_number', 'N/A'),
                            'End_Line': extra_node.get('end_line_number', 'N/A'),
                            'Node_Name': extra_node.get('name', 'N/A'),

                for mismatch in node_analysis['mismatched_nodes']:

        print("\nSheets created:")
        print("- Matched_Values: Parser standardization verification with Python vs .NET values")
        print("- Comparison_Info: General comparison information")
        print("- Missing_Properties: Properties missing in either parser")
        print("- Value_Differences: Properties with different values")
        print("- Node_Mismatches: Detailed node-level property mismatches with exact locations")
        print("- Missing_Nodes: Individual nodes missing in either parser with full details")
        print("- Count_Differences: Extra nodes in each parser explaining count mismatches")
        print("- Summary: High-level comparison summary")
