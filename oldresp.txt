import os
import json
import openpyxl
from datetime import datetime


def write_function_snippets_to_excel_with_counts(output, input_file_path, save_directory):
    # Extract the base name of the input file (without extension)
    base_filename = os.path.splitext(os.path.basename(input_file_path))[0]

    # Generate a timestamp for the filename
    timestamp = datetime.now().strftime("%d%m%Y_%H%M%S")

    # Construct the Excel file name
    excel_filename = f"{base_filename}_Parser_Results_{timestamp}.xlsx"
    excel_path = os.path.join(save_directory, excel_filename)
    # Create a new workbook
    workbook = openpyxl.Workbook()

    # Define the fields and their additional properties
    fields = {
        "decision_node_constructs": ["file_name", "name", "start_line_number", "end_line_number"],
        "loop_entity": ["file_name", "name", "start_line_number", "end_line_number"],
        "db_calls": ["file_name", "function_name", "start_line_number", "end_line_number", "operation_type", "aliases",
                     "table_names", "filter_conditions"],
        "database_entities": ["name", "file_name", "start_line_number", "end_line_number"],
        "calculation": ["file_name", "name", "start_line_number", "end_line_number", "snippet", "operands", "variable"],
        "callees": [
            "callee_file_name", "caller_function", "snippet", "start_line_number", "end_line_number",
            "callee_function", "condition", "conditional_hierarchy", "is_conditional", "parameters"
        ],
        "service_calls": [
            "file_name", "name", "start_line_number", "end_line_number", "snippet", "target", "parameters"
        ],
        "transaction_commits": ["file_name", "name", "start_line_number", "end_line_number", "snippet"],
        "transaction_rollbacks": ["file_name", "name", "start_line_number", "end_line_number", "snippet"]
    }

    # Write each field's data to separate tabs
    for field_name, properties in fields.items():
        sheet = workbook.create_sheet(title=field_name)
        sheet.append(properties)  # Write header row

        for function_snippet in output.function_snippets:
            field_objects = getattr(function_snippet, field_name, [])
            for obj in field_objects:
                row = []
                for prop in properties:
                    # Explicit handling for a property
                    if prop == 'conditional_hierarchy':
                        row.append(json.dumps(getattr(obj, prop, {}), indent=4))
                    else:
                        row.append(str(getattr(obj, prop, '')))
                sheet.append(row)

    # Create a summary sheet for counts
    summary_sheet = workbook.create_sheet(title="Summary")
    summary_headers = [
        "file_name", "function_name", "start_line_number", "end_line_number",
        "decision_node_constructs_count", "loop_entity_count", "db_calls_count",
        "calculation_count", "callees_count"
    ]
    summary_sheet.append(summary_headers)

    for function_snippet in output.function_snippets:
        file_name = getattr(function_snippet, "file_name", None)
        function_name = getattr(function_snippet, "function_name", None)
        start_line_number = getattr(function_snippet, "start_line_number", None)
        end_line_number = getattr(function_snippet, "end_line_number", None)

        counts = [
            len(getattr(function_snippet, "decision_node_constructs", [])),
            len(getattr(function_snippet, "loop_entity", [])),
            len(getattr(function_snippet, "db_calls", [])),
            len(getattr(function_snippet, "calculation", [])),
            len(getattr(function_snippet, "callees", []))
        ]

        summary_row = [file_name, function_name, start_line_number, end_line_number] + counts
        summary_sheet.append(summary_row)

    # Remove the default sheet created by openpyxl
    if "Sheet" in workbook.sheetnames:
        workbook.remove(workbook["Sheet"])

    # Save the workbook to the specified path
    workbook.save(excel_path)

# Example usage:
# Assume `output` is a dictionary containing `function_snippets` list of objects
# Assume `excel_path` is the path to save the Excel file
