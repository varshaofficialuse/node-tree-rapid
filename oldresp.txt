import re


class TSqlPreprocessor:
    def __init__(self):
        # Patterns for single-line and multi-line comments
        # self.single_line_comment_pattern = r"--.*"
        # Match -- comments that are not between quotes
        # Handle both -- and --- comments, but preserve when inside quotes
        self.single_line_comment_pattern = r"(?:^|\s)(?:---|--)[^\n]*"
        self.multi_line_comment_pattern = r"/\*[\s\S]*?\*/"

        self.unwanted_line_patterns = [
            r'^[ \t]*Text\s+--.*$', r'^[ \t]*Completion\stime:.*$'
        ]

    def remove_unwanted_lines(self, file_content: str) -> str:
        for pattern in self.unwanted_line_patterns:
            # file_content = re.sub(pattern, '', file_content, flags=re.MULTILINE)
            # Replace matched patterns with empty strings while preserving line breaks
            file_content = re.sub(pattern, lambda match: "\n" * match.group(0).count("\n"), file_content,
                                  flags=re.MULTILINE)

        return file_content

    def extract_comments(self, file_content: str) -> str:
        """
        Extract single-line and multi-line comments from the text.
        """
        # Extract single-line comments
        single_line_comments = re.findall(self.single_line_comment_pattern, file_content, flags=re.MULTILINE)

        # Extract multi-line comments
        multi_line_comments = re.findall(self.multi_line_comment_pattern, file_content, flags=re.MULTILINE)

        # Combine all extracted comments
        comments = single_line_comments + multi_line_comments

        return '\n'.join(comments)

    def remove_comments(self, file_content: str) -> str:
        """
        Remove single-line and multi-line comments from the text while preserving line breaks.
        """
        # Remove single-line comments
        text_without_comments = re.sub(self.single_line_comment_pattern, "", file_content, flags=re.MULTILINE)

        # Remove multi-line comments, preserving line breaks
        text_without_comments = re.sub(
            self.multi_line_comment_pattern,
            lambda match: "\n" * match.group(0).count("\n"),
            text_without_comments,
            flags=re.MULTILINE
        )

        return text_without_comments

using System;
using System.Collections.Generic;
using System.Linq;
using System.Text.RegularExpressions;

namespace TsqlParserProject
{
    public class TSqlPreprocessor
    {
        private string singleLineCommentPattern = @"(?:^|\s)(?:---|--)[^\n]*";
        private string multiLineCommentPattern = @"/\*[\s\S]*?\*/";
        private List<string> unwantedLinePatterns = new()
        {
            @"^[ \t]*Text\s+--.*$",
            @"^[ \t]*Completion\stime:.*$"
        };

        public string RemoveUnwantedLines(string fileContent)
        {
            foreach (var pattern in unwantedLinePatterns)
            {
                fileContent = Regex.Replace(fileContent, pattern, match =>
                {
                    int lineCount = match.Value.Count(c => c == '\n');
                    return new string('\n', lineCount);
                }, RegexOptions.Multiline);
            }
            return fileContent;
        }

        public string ExtractComments(string fileContent)
        {
            var singleLineComments = Regex.Matches(fileContent, singleLineCommentPattern, RegexOptions.Multiline)
                                          .Cast<Match>().Select(m => m.Value);
            var multiLineComments = Regex.Matches(fileContent, multiLineCommentPattern, RegexOptions.Multiline)
                                         .Cast<Match>().Select(m => m.Value);
            
            return string.Join("\n", singleLineComments.Concat(multiLineComments));
        }

        public string RemoveComments(string fileContent)
        {
            string textWithoutComments = Regex.Replace(fileContent, singleLineCommentPattern, "", RegexOptions.Multiline);
            
            textWithoutComments = Regex.Replace(textWithoutComments, multiLineCommentPattern, match =>
            {
                int lineCount = match.Value.Count(c => c == '\n');
                return new string('\n', lineCount);
            }, RegexOptions.Multiline);

            return textWithoutComments;
        }
    }
}
