        public CustomTSqlParser(CodeComprehensionMetadata projectMetadata, string filePath, string file_content, ICharStream input_stream, CommonTokenStream token_stream)
        {
            this.projectMetadata = projectMetadata;
            this.project_id = projectMetadata.ProjectId.ToString();
            this.path = filePath;
            this.file_content = file_content;
            this.input_stream = input_stream;
            this.token_stream = token_stream;
            this.file_name = Path.GetFileName(this.path);
            this.folder_name = Path.GetFileName(Path.GetDirectoryName(filePath));
        }

        public string GetOriginalTextFromContext(dynamic ctx)
        {
            return input_stream.GetText(new Interval(ctx.Start.StartIndex, ctx.Stop.StopIndex));
        }

        public string GetSnippetTextFromContext(dynamic ctx)
        {
            int startIdx = ctx.Start.TokenIndex;
            int stopIdx = ctx.Stop.TokenIndex;
            var tokens = token_stream.GetTokens(startIdx, stopIdx + 1);
            return string.Join("", tokens.Where(t => !comment_tokens.Contains(t.Type)).Select(t => t.Text));
        }

        public string GetCurrentContextName(dynamic ctx)
        {
            var p = ctx;
            while (p != null)
            {
                string typeName = p.GetType().Name;
                if (typeName == "Create_or_alter_functionContext") return ((string)p.funcName.procedure.GetText()).Trim('[', ']');
                if (typeName == "Create_or_alter_procedureContext") return ((string)p.procName.procedure.GetText()).Trim('[', ']');
                if (typeName == "Create_or_alter_dml_triggerContext" || typeName == "Create_or_alter_ddl_triggerContext")
                    return ((string)p.simple_name().name.GetText()).Trim('[', ']');
                p = p.Parent;
            }
            return MAIN_SNIPPET;
        }

        public List<VariableEntity> CreateVariableEntityList(List<DataDefinitionJSON> variables)
        {
            return variables.Select(v => new VariableEntity
            {
                Term = v.Term,
                Type = v.Type,
                UsageContext = v.UsageContext,
                Meaning = v.Meaning,
                FilePath = v.FilePath
            }).ToList();
        }

        public (HashSet<string>, HashSet<string>, List<string>) ExtractTableSource(TSqlParser.Table_sourceContext table_source)
        {
            var tables = new HashSet<string>();
            var aliases = new HashSet<string>();
            var filter_conditions = new List<string>();

            if (table_source != null)
            {
                var tsi = table_source.table_source_item();
                if (tsi.as_table_alias() != null) aliases.Add(GetOriginalTextFromContext(tsi.as_table_alias()));
                if (tsi.full_table_name() != null) tables.Add(GetOriginalTextFromContext(tsi.full_table_name()));

                if (table_source.join_part() != null)
                {
                    foreach (var join in table_source.join_part())
                    {
                        if (join.join_on() != null)
                        {
                            var join_on = join.join_on();
                            if (join_on.search_condition() != null)
                                filter_conditions.Add(GetOriginalTextFromContext(join_on.search_condition()));
                            
                            var result = ExtractTableSource(join_on.table_source());
                            foreach (var t in result.Item1) tables.Add(t);
                            foreach (var a in result.Item2) aliases.Add(a);
                            filter_conditions.AddRange(result.Item3);
                        }
                    }
                }
            }
            return (tables, aliases, filter_conditions);
        }

        public void CreateExtractedCalculationSnippet(string snippet, string function_name, int func_start_line)
        {
            var extractData = cve.ExtractMathematicalCalculations(snippet);
            var extractDataStartLine = cve.ExtractMathematicalCalculationsStartLine(snippet);

            for (int cnt = 0; cnt < extractData.Count; cnt++)
            {
                try
                {
                    var lineDict = extractData[cnt];
                    string resultantData = "";
                    string expressionData = "";
                    string matchGroup = (string)lineDict["match_group"];

                    if (matchGroup.Contains("="))
                    {
                        var matchData = matchGroup.Split('=');
                        resultantData = matchData[0];
                        expressionData = matchData[1];
                    }
                    else
                    {
                        expressionData = matchGroup;
                    }

                    var variablesDetails = cve.ExtractConstantsVariables(string.Join(" ", expressionData.ToCharArray()), this.path);
                    var variableEntities = CreateVariableEntityList(variablesDetails);

                    string matchedData = expressionData.Trim();
                    bool keyword = matchedData.Contains("SELECT", StringComparison.OrdinalIgnoreCase);

                    if (!keyword)
                    {
                        var operands = GetOperands(matchedData);
                        var linesResult = FindTargetLines(matchedData, (int)extractDataStartLine[cnt]["start"]);

                        var calculationEntity = new CalculationEntity
                        {
                            Name = function_name,
                            Tech = projectMetadata.Language,
                            FileName = file_name,
                            FilePath = path,
                            StartLineNumber = linesResult.Item1,
                            EndLineNumber = linesResult.Item2,
                            ProgramId = file_name,
                            Snippet = matchedData,
                            Resultant = resultantData,
                            Operands = operands,
                            Variables = variableEntities.Select(v => new DataDefinitionJSON { Term = v.Term, Type = v.Type, FilePath = v.FilePath, Meaning = v.Meaning, UsageContext = v.UsageContext }).ToList()
                        };

                        if (current_function != null) current_function.Calculation.Add(calculationEntity);
                        else main_program_calculations.Add(calculationEntity);
                    }
                }
                catch (Exception) { }
            }
        }

        public FunctionSnippetSchema CreateFunctionSnippet(dynamic ctx)
        {
            execution_order = 0;
            callee_execution_order = 0;
            current_if_depth = 0;
            current_case_depth = 0;
            snippet_execution_order++;

            int start_line_number = ctx.Start.Line;
            int end_line_number = ctx.Stop.Line;
            string function_name = GetCurrentContextName(ctx);

            string class_name = file_name;
            string typeName = ctx.GetType().Name;
            if (typeName == "Create_or_alter_functionContext") class_name = (string)ctx.funcName.GetText();
            if (typeName == "Create_or_alter_procedureContext") class_name = (string)ctx.procName.GetText();
            if (typeName == "Create_or_alter_dml_triggerContext" || typeName == "Create_or_alter_ddl_triggerContext")
                class_name = (string)ctx.simple_name().GetText();

            string snippet_text = GetOriginalTextFromContext(ctx);
            var variablesDetails = cve.ExtractConstantsVariables(snippet_text, this.path);
            var variable_entities = CreateVariableEntityList(variablesDetails);

            var function_snippet = new FunctionSnippetSchema
            {
                FilePath = path,
                FileName = file_name,
                FunctionName = function_name,
                ClassName = class_name,
                Snippet = snippet_text,
                StartLineNumber = start_line_number,
                EndLineNumber = end_line_number,
                ExecutionOrder = execution_order,
                SnippetExecutionOrder = snippet_execution_order,
                ReturnType = "NA",
                Variable = variable_entities.Select(v => new DataDefinitionJSON { Term = v.Term, Type = v.Type, FilePath = v.FilePath, Meaning = v.Meaning, UsageContext = v.UsageContext }).Cast<object>().ToList()
            };

            function_snippets.Add(function_snippet);
            return function_snippet;
        }

        public void VisitFunctionSnippet(dynamic ctx)
        {
            var functionSnippet = CreateFunctionSnippet(ctx);
            int functionStartLine = ctx.Start.StartIndex;

            current_function = functionSnippet;
            function_snippet_stack.Push(functionSnippet);

            CreateExtractedCalculationSnippet(current_function.Snippet, current_function.FunctionName, functionStartLine);

            VisitChildren(ctx);

            function_snippet_stack.Pop();
            current_function = function_snippet_stack.Count > 0 ? function_snippet_stack.Peek() : null;
        }

        public DecisionNodeSchema CreateRuleSnippet(dynamic ctx, RuleConstructType constructType)
        {
            int start_line_number = ctx.Start.Line;
            int end_line_number = ctx.Stop.Line;

            Regex conditionalKeywords;
            if (constructType == RuleConstructType.SWITCH_CONSTRUCT)
                conditionalKeywords = new Regex(@"\b(WHEN|ELSE)\b", RegexOptions.IgnoreCase);
            else
                conditionalKeywords = new Regex(@"\b((?<!END-)IF|ELSE|(?<!END-)DECIDE)\b", RegexOptions.IgnoreCase);

            string snippet_text = GetOriginalTextFromContext(ctx);
            int num_conditions = conditionalKeywords.Matches(snippet_text).Count;

            string cleaned_text = snippet_text.Replace("\r", " ");
            string hosting_function_name = GetCurrentContextName(ctx);

            var variablesDetails = cve.ExtractConstantsVariables(cleaned_text, this.path);

            string numericExtractionText = "";
            if (ctx.GetType().Name == "If_statementContext" && ctx.children != null)
            {
                var conditionCtx = ctx.search_condition();
                if (conditionCtx != null)
                {
                    numericExtractionText = GetOriginalTextFromContext(conditionCtx);
                }
            }
            else
            {
                numericExtractionText = snippet_text;
            }

            variablesDetails.AddRange(cve.ExtractNumericConstants(numericExtractionText.Replace("\r", " "), this.path));
            var variableEntities = CreateVariableEntityList(variablesDetails);

            return new DecisionNodeSchema
            {
                Name = hosting_function_name,
                ShortSummary = "",
                StartLineNumber = start_line_number,
                EndLineNumber = end_line_number,
                Snippet = snippet_text,
                NumberOfConditionals = num_conditions,
                Complexity = ComplexityHelper.CategorizeComplexity(num_conditions).ToString(),
                ConstructType = constructType.ToString(),
                ExecutionOrder = execution_order,
                Variable = variableEntities.Cast<object>().ToList()
            };
        }

        public DbCallDataEntity CreateDbCall(dynamic ctx, string query, List<string> filter_conditions, HashSet<string> aliases, HashSet<string> tables, string operation_type)
        {
            execution_order++;
            string full_text = GetOriginalTextFromContext(ctx);
            string cleaned_text = full_text.Replace("\r", " ");

            int start_line_number = ctx.Start.Line;
            int end_line_number = ctx.Stop.Line;

            var variablesDetails = cve.ExtractConstantsVariables(cleaned_text, this.path);
            var variableEntities = CreateVariableEntityList(variablesDetails);

            string hosting_function_name = GetCurrentContextName(ctx);

            var db_call_entity = new DbCallDataEntity
            {
                Name = hosting_function_name,
                FilePath = path,
                FileName = file_name,
                Tech = projectMetadata.Language,
                ProgramId = file_name,
                StartLineNumber = start_line_number,
                EndLineNumber = end_line_number,
                Snippet = cleaned_text,
                OperationType = operation_type,
                ExecutionOrder = execution_order,
                Variable = variableEntities,
                Query = query,
                FilterConditions = filter_conditions,
                Aliases = aliases.ToList(),
                TableNames = tables.Select(t => t.Split('.').Last().Trim('[', ']').ToLower()).ToList()
            };

            if (current_function != null) current_function.DbCalls.Add(db_call_entity);
            else main_program_dbcalls.Add(db_call_entity);

            if (db_call_stack.Count == 0)
            {
                db_call_execution_order++;
                db_call_entity.Level = DbCallLevel.PARENT.ToString();
                db_call_entity.DbCallExecutionOrder = db_call_execution_order.ToString();
                db_call_entity.ChildCount = 0;
            }
            else
            {
                var parent = db_call_stack.Peek();
                parent.ChildCount++;
                db_call_entity.Level = DbCallLevel.CHILD.ToString();
                db_call_entity.DbCallExecutionOrder = parent.DbCallExecutionOrder + "." + parent.ChildCount;
                db_call_entity.ChildCount = 0;
            }

            return db_call_entity;
        }

        public FunctionCallee CreateCallee(dynamic ctx, CalleeConstructType constructType)
        {
            execution_order++;
            callee_execution_order++;
            int start_line_number = ctx.Start.Line;
            int end_line_number = ctx.Stop.Line;
            string snippet_text = GetOriginalTextFromContext(ctx);

            var conditionalHierarchy = GetCalleeConditionalHierarchy(ctx);
            bool isConditional = conditionalHierarchy.Count > 0;
            string condition = "NA";
            if (isConditional)
            {
                condition = (bool)conditionalHierarchy["is_else"] ? "ELSE_OF(" + (string)conditionalHierarchy["condition"] + ")" : (string)conditionalHierarchy["condition"];
            }

            string targetFunctionName = null;
            var variableEntities = new List<VariableEntity>();
            var parameters = new List<string>();

            if (constructType == CalleeConstructType.EXECUTE)
            {
                if (ctx.execute_body() != null && ctx.execute_body().func_proc_name_server_database_schema() != null)
                {
                    targetFunctionName = ParseFuncProcNameServerDatabaseSchema(ctx.execute_body().func_proc_name_server_database_schema());
                    var variablesDetails = cve.ExtractConstantsVariables(snippet_text, this.path);
                    variableEntities = CreateVariableEntityList(variablesDetails);
                    parameters = variablesDetails.Select(v => v.Term).ToList();
                }
            }
            else if (constructType == CalleeConstructType.SCALAR_FUNCTION)
            {
                if (ctx.scalar_function_name() != null && ctx.scalar_function_name().func_proc_name_server_database_schema() != null)
                {
                    targetFunctionName = ParseFuncProcNameServerDatabaseSchema(ctx.scalar_function_name().func_proc_name_server_database_schema());
                }
                if (ctx.expression_list_() != null)
                {
                    string variables = GetOriginalTextFromContext(ctx.expression_list_());
                    var variablesDetails = cve.ExtractConstantsVariables(variables, this.path);
                    variableEntities = CreateVariableEntityList(variablesDetails);
                    parameters = variablesDetails.Select(v => v.Term).ToList();
                }
            }

            if (targetFunctionName != null)
            {
                var callee = new FunctionCallee
                {
                    ExecutionOrder = execution_order,
                    CalleeExecutionOrder = callee_execution_order,
                    StartLineNumber = start_line_number,
                    EndLineNumber = end_line_number,
                    Tech = projectMetadata.Language,
                    CallerFunction = GetCurrentContextName(ctx),
                    CalleeFunction = targetFunctionName,
                    CalleeClass = file_name,
                    CalleeFileName = file_name,
                    CalleeFilePath = path,
                    Snippet = snippet_text,
                    CalleeType = constructType.ToString(),
                    TypeOfConstruct = "External",
                    ModuleName = folder_name,
                    Behavior = CalleeBehavior.RETURNS.ToString(),
                    Condition = condition,
                    Variable = variableEntities.Cast<object>().ToList(),
                    Parameters = parameters,
                    IsConditional = isConditional,
                    ConditionalHierarchy = conditionalHierarchy
                };

                if (current_function != null) current_function.Callees.Add(callee);
                else main_program_callees.Add(callee);
                return callee;
            }
            return null;
        }

        public void CreateDatabaseEntity(string snippet, string tableName, int startLine, int endLine)
        {
            string server = "", database = "", schema = "", table = "";
            var parts = tableName.Split('.');
            if (parts.Length == 4) { server = parts[0]; database = parts[1]; schema = parts[2]; table = parts[3]; }
            else if (parts.Length == 3) { database = parts[0]; schema = parts[1]; table = parts[2]; }
            else if (parts.Length == 2) { schema = parts[0]; table = parts[1]; }
            else table = tableName;

            var db_entity = new DatabaseEntitySchema
            {
                Name = table.Trim('[', ']').ToLower(),
                BusinessAliases = new List<string>(),
                FileName = file_name,
                FilePath = path,
                StartLineNumber = startLine,
                EndLineNumber = endLine,
                ProgramId = file_name,
                Type = "TABLE",
                Snippet = snippet,
                SchemaTable = tableName.ToLower(),
                StructureSchema = "NA",
                Fields = new List<string>(),
                SampleRecord = "NA",
                Complexity = "NA",
                VolumeClass = "NA",
                MainRole = "NA",
                RelationshipDensity = "NA",
                DbType = "Relational",
                Description = "NA",
                DbTech = projectMetadata.Language,
                HostedAt = "NA",
                RecordsType = "NA",
                CreatedFrom = "Program",
                Category = "NA",
                Servers = !string.IsNullOrEmpty(server) ? new List<string> { server.Trim('[', ']').ToLower() } : new List<string>(),
                Databases = !string.IsNullOrEmpty(database) ? new List<string> { database.Trim('[', ']').ToLower() } : new List<string>(),
                Schemas = !string.IsNullOrEmpty(schema) ? new List<string> { schema.Trim('[', ']').ToLower() } : new List<string>()
            };

            if (current_function != null) current_function.DatabaseEntities.Add(db_entity);
            else main_program_database_entities.Add(db_entity);
        }

        public override object VisitSelect_statement(TSqlParser.Select_statementContext ctx)
        {
            string query = null;
            var aliases = new HashSet<string>();
            var tables = new HashSet<string>();
            var filter_conditions = new List<string>();
            string operation_type = DbCallDataEntity_TYPE.SELECT.ToString();

            try
            {
                query = GetOriginalTextFromContext(ctx).Trim(' ', '\n');
                if (ctx.query_expression() != null)
                {
                    var qe_ctx = ctx.query_expression();
                    var qs_ctx = qe_ctx.query_specification();
                    if (qs_ctx.table_sources() != null)
                    {
                        foreach (var con in qs_ctx.search_condition())
                            if (con != null) filter_conditions.Add(GetOriginalTextFromContext(con));

                        foreach (var ts in qs_ctx.table_sources().table_source())
                        {
                            var result = ExtractTableSource(ts);
                            foreach (var t in result.Item1) tables.Add(t);
                            foreach (var a in result.Item2) aliases.Add(a);
                            filter_conditions.AddRange(result.Item3);
                        }
                    }
                    if (qs_ctx.table_name() != null) tables.Add(GetOriginalTextFromContext(qs_ctx.table_name()));

                    if (qe_ctx.sql_union() != null)
                    {
                        foreach (var union in qe_ctx.sql_union())
                        {
                            var uqs_ctx = union.query_specification();
                            if (uqs_ctx.table_sources() != null)
                            {
                                foreach (var con in uqs_ctx.search_condition())
                                    if (con != null) filter_conditions.Add(GetOriginalTextFromContext(con));
                                foreach (var ts in uqs_ctx.table_sources().table_source())
                                {
                                    var result = ExtractTableSource(ts);
                                    foreach (var t in result.Item1) tables.Add(t);
                                    foreach (var a in result.Item2) aliases.Add(a);
                                    filter_conditions.AddRange(result.Item3);
                                }
                            }
                        }
                    }
                }
            }
            catch { }

            if (tables.Count == 0) return null;
            string snippet = GetOriginalTextFromContext(ctx);
            var table_list = new HashSet<string>();
            foreach (var table in tables)
            {
                string t = table;
                if (t.Contains('.')) t = t.Split('.')[1];
                table_list.Add(t);
                if (!EntityExists(table.Trim())) CreateDatabaseEntity(snippet, table.Trim(), ctx.Start.Line, ctx.Stop.Line);
            }

            var db_call = CreateDbCall(ctx, query, filter_conditions, aliases, table_list, operation_type);
            db_call_stack.Push(db_call);
            current_dbcall_depth++;
            VisitChildren(ctx);
            current_dbcall_depth--;
            db_call_stack.Pop();
            CreateDatabaseField();
            return null;
        }

        public override object VisitInsert_statement(TSqlParser.Insert_statementContext ctx)
        {
            var aliases = new HashSet<string>();
            var tables = new HashSet<string>();
            var filter_conditions = new List<string>();
            string operation_type = DbCallDataEntity_TYPE.INSERT.ToString();
            string query = GetOriginalTextFromContext(ctx);

            if (ctx.ddl_object() != null)
            {
                string table = GetOriginalTextFromContext(ctx.ddl_object());
                tables.Add(table);
                if (!EntityExists(table.Trim())) CreateDatabaseEntity(query, table.Trim(), ctx.Start.Line, ctx.Stop.Line);
            }
            else if (ctx.rowset_function_limited() != null)
                tables.Add(GetOriginalTextFromContext(ctx.rowset_function_limited()));

            if (tables.Count == 0) return null;

            var db_call = CreateDbCall(ctx, query, filter_conditions, aliases, tables, operation_type);
            db_call_stack.Push(db_call);
            current_dbcall_depth++;
            VisitChildren(ctx);
            current_dbcall_depth--;
            db_call_stack.Pop();
            CreateDatabaseField();
            return null;
        }

        public Dictionary<string, object> GetCalleeConditionalHierarchy(dynamic ctx)
        {
            var hierarchy = new Dictionary<string, object>();
            var currentLevel = hierarchy;
            bool isElse = false;

            while (ctx != null)
            {
                var parentCtx = ctx.Parent;
                if (parentCtx == null) break;

                if (parentCtx.GetType().Name == "If_statementContext" && parentCtx.children != null)
                {
                    if (parentCtx.ELSE() != null)
                    {
                        var sym = parentCtx.ELSE().Symbol;
                        if (sym.StopIndex < ctx.Stop.StartIndex) isElse = true;
                    }

                    try
                    {
                        var conditionCtx = parentCtx.search_condition();
                        if (conditionCtx != null)
                        {
                            string conditionString = GetOriginalTextFromContext(conditionCtx);
                            conditionString = Regex.Replace(conditionString, @"\n\d{4}\s*$", "");
                            currentLevel["condition"] = conditionString;
                            currentLevel["start_line_number"] = parentCtx.Start.Line;
                            currentLevel["end_line_number"] = parentCtx.Stop.Line;
                            currentLevel["siblings"] = new List<string>();
                            currentLevel["is_else"] = isElse;
                            var nextParent = new Dictionary<string, object>();
                            currentLevel["parent"] = nextParent;
                            currentLevel = nextParent;
                        }
                    }
                    catch { }
                    isElse = false;
                }
                ctx = parentCtx;
            }
            return hierarchy;
        }

            }
            else procedure = ctx.procedure;

            return procedure != null ? ((string)procedure.GetText()).Trim('[', ']') : (string)ctx.GetText();
        }

        public List<DatabaseFieldSchema> CreateDatabaseField()
        {
            var database_fields = new List<DatabaseFieldSchema>();
            string default_table = table_alias.FirstOrDefault(kv => kv.Key == kv.Value).Value;

            foreach (var field_data in column_name)
            {
                var parts = field_data.Split('.');
                if (parts.Length == 2)
                {
                    string table_part = parts[0].Trim('[', ']');
                    string column_part = parts[1];
                    if (table_alias.TryGetValue(table_part, out var actual_table))
                    {
                        string actual_table_name = actual_table.Split('.').Last().Trim('[', ']').ToLower();
                        var field_key = (column_part, actual_table_name);
                        if (!unique_fields.Contains(field_key))
                        {
                            unique_fields.Add(field_key);
                            database_fields.Add(new DatabaseFieldSchema { Name = column_part, TableName = actual_table_name, DbTech = projectMetadata.Language });
                        }
                    }
                }
                else if (parts.Length == 1 && default_table != null)
                {
                    string column_part = parts[0];
                    string default_table_name = default_table.Split('.').Last().Trim('[', ']').ToLower();
                    var field_key = (column_part, default_table_name);
                    if (!unique_fields.Contains(field_key))
                    {
                        unique_fields.Add(field_key);
                        database_fields.Add(new DatabaseFieldSchema { Name = column_part, TableName = default_table_name, DbTech = projectMetadata.Language });
                    }
                }
            }
            if (current_function != null) current_function.DatabaseFields.AddRange(database_fields);
            else main_program_database_fields.AddRange(database_fields);
            column_name.Clear();
            return database_fields;
        }

        public override object VisitUpdate_statement(TSqlParser.Update_statementContext ctx)
        {
            var aliases = new HashSet<string>();
            var tables = new HashSet<string>();
            var filter_conditions = new List<string>();
            string operation_type = DbCallDataEntity_TYPE.UPDATE.ToString();
            string query = GetOriginalTextFromContext(ctx);

            if (ctx.ddl_object() != null)
            {
                string table = GetOriginalTextFromContext(ctx.ddl_object());
                tables.Add(table);
                if (!EntityExists(table.Trim())) CreateDatabaseEntity(query, table.Trim(), ctx.Start.Line, ctx.Stop.Line);
            }
            else if (ctx.rowset_function_limited() != null)
                tables.Add(GetOriginalTextFromContext(ctx.rowset_function_limited()));

            if (ctx.table_sources() != null)
            {
                foreach (var ts in ctx.table_sources().table_source())
                {
                    var result = ExtractTableSource(ts);
                    foreach (var t in result.Item1) tables.Add(t);
                    foreach (var a in result.Item2) aliases.Add(a);
                    filter_conditions.AddRange(result.Item3);
                }
            }
            if (tables.Count == 0) return null;
            foreach (var table in tables) if (!EntityExists(table.Trim())) CreateDatabaseEntity(query, table.Trim(), ctx.Start.Line, ctx.Stop.Line);

            var db_call = CreateDbCall(ctx, query, filter_conditions, aliases, tables, operation_type);
            db_call_stack.Push(db_call);
            current_dbcall_depth++;
            VisitChildren(ctx);
            current_dbcall_depth--;
            db_call_stack.Pop();
            CreateDatabaseField();
            return null;
        }

        public override object VisitDelete_statement(TSqlParser.Delete_statementContext ctx)
        {
            var aliases = new HashSet<string>();
            var tables = new HashSet<string>();
            var filter_conditions = new List<string>();
            string operation_type = DbCallDataEntity_TYPE.DELETE.ToString();
            string query = GetOriginalTextFromContext(ctx);

            if (ctx.delete_statement_from() != null) tables.Add(GetOriginalTextFromContext(ctx.delete_statement_from()));
            if (ctx.search_condition() != null) filter_conditions.Add(GetOriginalTextFromContext(ctx.search_condition()));
            if (ctx.table_sources() != null)
            {
                foreach (var ts in ctx.table_sources().table_source())
                {
                    var result = ExtractTableSource(ts);
                    foreach (var t in result.Item1) tables.Add(t);
                    foreach (var a in result.Item2) aliases.Add(a);
                    filter_conditions.AddRange(result.Item3);
                }
            }
            if (tables.Count == 0) return null;

            var db_call = CreateDbCall(ctx, query, filter_conditions, aliases, tables, operation_type);
            db_call_stack.Push(db_call);
            current_dbcall_depth++;
            VisitChildren(ctx);
            current_dbcall_depth--;
            db_call_stack.Pop();
            CreateDatabaseField();
            return null;
        }

        public override object VisitTable_source_item(TSqlParser.Table_source_itemContext ctx)
        {
            if ((ctx.full_table_name() != null || ctx.LOCAL_ID() != null) && ctx.as_table_alias() != null)
            {
                string alias = (string)ctx.as_table_alias().table_alias().GetText();
                if (ctx.full_table_name() != null) table_alias[alias] = (string)ctx.full_table_name().GetText();
                else if (ctx.LOCAL_ID() != null) table_alias[alias] = (string)ctx.LOCAL_ID().GetText();
            }
            else if (ctx.full_table_name() != null)
            {
                table_alias[(string)ctx.full_table_name().GetText()] = (string)ctx.full_table_name().GetText();
            }
            return null;
        }

        public override object VisitExecute_statement(TSqlParser.Execute_statementContext ctx)
        {
            string targetFunctionName = null;
            if (ctx.execute_body() != null && ctx.execute_body().func_proc_name_server_database_schema() != null)
                targetFunctionName = ParseFuncProcNameServerDatabaseSchema(ctx.execute_body().func_proc_name_server_database_schema());

            if (target_functions.Contains(targetFunctionName)) CreateServiceCall(ctx, targetFunctionName);
            else CreateCallee(ctx, CalleeConstructType.EXECUTE);

            return VisitChildren(ctx);
        }

        public override object VisitRowset_function(TSqlParser.Rowset_functionContext ctx)
        {
            string snippet_text = GetOriginalTextFromContext(ctx);
            if (snippet_text.ToUpperInvariant().Contains("OPENROWSET")) CreateServiceCall(ctx, "OPENROWSET");
            return VisitChildren(ctx);
        }

        public override object VisitWhile_statement(TSqlParser.While_statementContext ctx)
        {
            execution_order++;
            execution_order = execution_order + 1;

            string full_text = GetOriginalTextFromContext(ctx);
            string cleaned_text = full_text.Replace("\r", " ");
            int start_line_number = ctx.Start.Line;
            int end_line_number = ctx.Stop.Line;
            string hosting_function_name = GetCurrentContextName(ctx);

            var loop_entity = new LoopEntityNodeSchema
            {
                Name = hosting_function_name,
                Tech = projectMetadata.Language,
                FileName = file_name,
                FilePath = path,
                StartLineNumber = start_line_number,
                EndLineNumber = end_line_number,
                ProgramId = file_name.Split('.')[0],
                Type = "WHILE",
                Snippet = cleaned_text,
                ExecutionOrder = execution_order,
                ModuleName = folder_name
            };

            if (current_function != null) current_function.LoopEntity.Add(loop_entity);
            else main_program_loop_entities.Add(loop_entity);

            return VisitChildren(ctx);
        }

        public override object VisitExpression(TSqlParser.ExpressionContext ctx)
        {
            if (ctx.full_column_name() != null)
            {
                string full_name = (string)ctx.full_column_name().GetText();
                var parts = full_name.Split('.').Select(p => p.Trim('[', ']')).ToList();
                if (parts.Count >= 2) column_name.Add($"{parts[parts.Count - 2]}.{parts.Last()}");
                else column_name.Add(parts[0]);
            }
            return VisitChildren(ctx);
        }

        public List<string> GetOperands(string data)
        {
            return Regex.Matches(data, @"[+\-*/]").Cast<Match>().Select(m => m.Value).ToList();
        }

        public (int, int) FindTargetLines(string target_data, int start_line)
        {
            try
            {
                var target_data_line = target_data.Split(new[] { "\r\n", "\r", "\n" }, StringSplitOptions.None);
                if (target_data_line.Length > 1) return (start_line, start_line + target_data_line.Length - 1);
                if (target_data_line.Length == 1) return (start_line, start_line);
            }
            catch (Exception e)
            {
                Console.WriteLine($"Error in find_target_lines: {e}");
            }
            return (0, 0);
        }
    }

    # This function gives you original
    def get_original_text_from_context(self, ctx):
        start_idx = ctx.start.start  # Start char index
        stop_idx = ctx.stop.stop  # End char index

        # Get the text from the input stream
        text = self.input_stream.getText(start_idx, stop_idx)

        return text

    # This function gives you text that does not contains any comments, etc.
    def get_snippet_text_from_context(self, ctx):
        start_idx = ctx.start.tokenIndex  # Start char index
        stop_idx = ctx.stop.tokenIndex  # End char index

        tokens = self.token_stream.getTokens(start_idx, stop_idx + 1)
        visible_text = ''.join(t.text for t in tokens if t.type not in self.comment_tokens)

        return visible_text

    def get_current_context_name(self, ctx):
        while ctx:
            if isinstance(ctx, TSqlParser.Create_or_alter_functionContext):
                return ctx.funcName.procedure.getText().strip('[]')
            if isinstance(ctx, TSqlParser.Create_or_alter_procedureContext):
                return ctx.procName.procedure.getText().strip('[]')
            if isinstance(ctx, TSqlParser.Create_or_alter_dml_triggerContext) \
                    or isinstance(ctx, TSqlParser.Create_or_alter_ddl_triggerContext):
                return ctx.simple_name().name.getText().strip('[]')
            ctx = ctx.parentCtx

        return CustomTSqlParser.MAIN_SNIPPET

    def create_variable_entity_list(self, variables) -> List[VariableEntity]:
        variable_list = []
        for variable in variables:
            variable_list.append(VariableEntity(
                term=variable.term,
                type=variable.type,
                usage_context=variable.usage_context,
                meaning=variable.meaning,
                file_path=variable.file_path
            ))

        return variable_list

    def extract_table_source(self, table_source: TSqlParser.Table_sourceContext):
        tables = set()
        aliases = set()
        filter_conditions = []
        if table_source:
            tsi = table_source.table_source_item()
            if tsi.as_table_alias():
                aliases.add(self.get_original_text_from_context(tsi.as_table_alias()))
            if tsi.full_table_name():
                tables.add(self.get_original_text_from_context(tsi.full_table_name()))

            if table_source.join_part():
                joins = table_source.join_part()
                for join in joins:
                    if join.join_on():  # Check if join_on is not None
                        join_on = join.join_on()
                        if join_on.search_condition():
                            filter_conditions.append(self.get_original_text_from_context(join_on.search_condition()))
                        ret_tables, ret_aliases, ret_filter_conditions = self.extract_table_source(
                            join_on.table_source())
                        tables.update(ret_tables)
                        aliases.update(ret_aliases)
                        filter_conditions.extend(ret_filter_conditions)
                    elif join.apply_():
                        pass

        return tables, aliases, filter_conditions

    def create_extracted_calculation_snippet(self, snippet, function_name, func_start_line):

        extract_data = self.cve.extract_mathematical_calculations(snippet)
        extract_data_start_line = self.cve.extract_mathematical_calculations_startline(snippet)
        resultant_data = ""
        expression_data = ""
        for cnt, line_dict in enumerate(extract_data, start=0):
            try:
                if "=" in line_dict["match_group"]:
                    match_data = line_dict["match_group"].split("=")
                    resultant_data = match_data[0]
                    expression_data = match_data[1]
                else:
                    expression_data = line_dict["match_group"]

                # print(line_dict["match_group"], line_dict["start"], line_dict["end"])
                str_Matched_Data = ' '.join(expression_data)
                variables_details = self.cve.extract_constants_variables(str_Matched_Data, self.path)
                variable_entities = self.create_variable_entity_list(variables_details)
                keyword = False

                matched_data = expression_data.strip()
                if "SELECT" in matched_data:
                    keyword = True
                else:
                    keyword = False

                if not keyword:
                    operands = self.get_operands(matched_data)
                    start_line, end_line = self.find_target_lines(matched_data,extract_data_start_line[cnt]["start"])

                    calculation_entity = CalculationEntity(
                        name=str(function_name),
                        tech=self.projectMetadata.technology,
                        file_name=self.file_name,
                        file_path=self.path,
                        start_line_number=start_line,
                        end_line_number=end_line,
                        program_id=self.file_name,
                        type='',
                        snippet=matched_data,
                        resultant=resultant_data,
                        operands=operands,
                        # variables = [],
                        variable=variable_entities
                    )

                    if self.current_function:
                        self.current_function.calculation.append(calculation_entity)
                    else:
                        self.main_program_calculations.append(calculation_entity)
            except Exception as e:
                logger.error(f"Error processing calculation entity for function '{function_name}': {e}",
                             exc_info=True)

        return None

    def create_function_snippet(self, ctx):
        self.execution_order = 0
        self.callee_execution_order = 0
        self.current_if_depth = 0
        self.current_case_depth = 0
        self.snippet_execution_order += 1

        # Fetch the start and end line number from context
        start_line_number = ctx.start.line
        end_line_number = ctx.stop.line

        # Fetch the name of the function and class
        function_name = self.get_current_context_name(ctx)

        class_name = self.file_name
        if isinstance(ctx, TSqlParser.Create_or_alter_functionContext):
            class_name = ctx.funcName.getText()
        if isinstance(ctx, TSqlParser.Create_or_alter_procedureContext):
            class_name = ctx.procName.getText()
        if isinstance(ctx, TSqlParser.Create_or_alter_dml_triggerContext) \
                or isinstance(ctx, TSqlParser.Create_or_alter_ddl_triggerContext):
            class_name = ctx.simple_name().getText()

        # Get the snippet text that will have tabs and spaces in place, but comments removed
        snippet_text = self.get_original_text_from_context(ctx)

        # Extract variables and constants
        variables_details = self.cve.extract_constants_variables(snippet_text, self.path)
        variable_entities = self.create_variable_entity_list(variables_details)

        # Create the function snippet object
        function_snippet = FunctionSnippetSchema(
            file_path=self.path,
            file_name=self.file_name,
            function_name=str(function_name),  # Only function name without any sq brackets or schema name
            class_name=class_name,  # Actual name as written including schema name (e.g. [dbo].[someProcName])
            snippet=str(snippet_text),
            start_line_number=start_line_number,
            end_line_number=end_line_number,
            callees=[],
            annotations=[],
            return_type="NA",
            decision_node_constructs=[],
            db_calls=[],
            service_calls=[],
            calculation=[],
            execution_order=self.execution_order,
            snippet_execution_order=self.snippet_execution_order,
            inputs=[],
            # variables=variables_details,
            variable=variable_entities
        )

        # Update the function snippets list
        self.function_snippets.append(function_snippet)

        return function_snippet

    def visit_function_snippet(self, ctx):
        # Create function snippet
        function_snippet = self.create_function_snippet(ctx)
        function_start_line = ctx.start.start

        # Assign the current function object for further references
        self.current_function = function_snippet
        self.function_snippet_stack.append(function_snippet)

        # This is for calculation call to capture all calculations related information
        self.create_extracted_calculation_snippet(
            self.current_function.snippet,
            self.current_function.function_name,
            function_start_line
        )

        # Visit function children
        self.visitChildren(ctx)

        # Update the current function reference
        self.current_function = self.function_snippet_stack.pop() if self.function_snippet_stack else None

    def create_rule_snippet(self, ctx, construct_type):
        start_line_number = ctx.start.line
        end_line_number = ctx.stop.line

        # Pattern for both IF statements and CASE expressions
        if construct_type == RuleConstructType.SWITCH_CONSTRUCT:
            # For CASE expressions, count WHEN clauses as they represent the actual conditions
            conditional_keywords = re.compile(r"\b(WHEN|ELSE)\b", re.IGNORECASE)
        else:
            # For IF statements, use the original pattern
            conditional_keywords = re.compile(r"\b((?<!END-)IF|ELSE|(?<!END-)DECIDE)\b", re.IGNORECASE)

        snippet_text = self.get_original_text_from_context(ctx)

        # Find all matches in the snippet, and ...
        matches = conditional_keywords.findall(snippet_text)
        num_conditions = len(matches)

        # Sometimes, main section will have IFs
        cleaned_text = snippet_text.replace('\r', ' ')
        hosting_function_name = self.get_current_context_name(ctx)

        # Extract variables and constants
        variables_details = self.cve.extract_constants_variables(cleaned_text, self.path)

        # Extract numeric constants
        numeric_extraction_text = ''
        if isinstance(ctx, TSqlParser.If_statementContext) and ctx.children:
            # Fetch the condition
            condition_ctx = ctx.search_condition() if hasattr(ctx,
                                                              'search_condition') and ctx.search_condition() else None
            # Now you can work with conditionCtx
            if condition_ctx:
                # Process the condition context
                numeric_extraction_text = self.get_original_text_from_context(condition_ctx)
        else:
            numeric_extraction_text = snippet_text

        variables_details += self.cve.extract_numeric_constants(numeric_extraction_text.replace('\r', ' '), self.path)
        variable_entities = self.create_variable_entity_list(variables_details)

        decision_node = DecisionNodeSchema(
            name=hosting_function_name,
            short_summary="",
            start_line_number=start_line_number,
            end_line_number=end_line_number,
            snippet=snippet_text,
            number_of_conditionals=num_conditions,
            complexity=Complexity.categorize_complexity(num_conditions),
            construct_type=construct_type,
            execution_order=self.execution_order,
            # variables=variables_details,
            variable=variable_entities
        )

        return decision_node

    def create_db_call(
            self, ctx, query: str | None, filter_conditions: list[str], aliases: set[str], tables: set[str],
            operation_type: DbCallDataEntity_TYPE,
    ):
        # First and foremost increment the execution_order
        self.execution_order += 1

        full_text = self.get_original_text_from_context(ctx)
        # In many systems, a carriage return character can cause the cursor to move to the beginning of the line,
        # and subsequent characters will overwrite the characters that were already there.
        cleaned_text = full_text.replace('\r', ' ')

        start_line_number = ctx.start.line
        end_line_number = ctx.stop.line

        # Extract variables and constants
        variables_details = self.cve.extract_constants_variables(cleaned_text, self.path)
        variable_entities = self.create_variable_entity_list(variables_details)

        hosting_function_name = self.get_current_context_name(ctx)

        db_call_entity = DbCallDataEntity(
            name=hosting_function_name,
            file_path=self.path,
            file_name=self.file_name,
            tech=self.projectMetadata.technology,
            program_id=self.file_name,
            start_line_number=start_line_number,
            end_line_number=end_line_number,
            snippet=cleaned_text,
            operation_type=operation_type,
            execution_order=self.execution_order,
            # variables=variables_details,
            variable=variable_entities,
            query=query,
            filter_conditions=filter_conditions,
            aliases=aliases,
            table_names=[table.split('.')[-1].strip('[]').lower() for table in tables]
        )

        # Append the created db call object to its respective function snippet
        if self.current_function:
            self.current_function.db_calls.append(db_call_entity)
        else:
            self.main_program_dbcalls.append(db_call_entity)

        # Below code is to identify order and levels of the db calls
        if len(self.db_call_stack) == 0:
            # Process Parent
            self.db_call_execution_order += 1
            db_call_entity.level = DbCallLevel.PARENT
            db_call_entity.db_call_execution_order = str(self.db_call_execution_order)
            db_call_entity.child_count = 0

        else:
            # I know the last added object is my parent
            parent = self.db_call_stack[-1]
            parent.child_count += 1

            # Process Child
            db_call_entity.level = DbCallLevel.CHILD
            db_call_entity.db_call_execution_order = parent.db_call_execution_order + '.' + str(parent.child_count)
            db_call_entity.child_count = 0

        return db_call_entity

    def get_callee_conditional_hierarchy(self, ctx):

        hierarchy = {}  # Immediate hierarchy
        current_level = hierarchy  # Pointer to the current level in the hierarchy
        is_else = False

        def update_current_level(condition_string, start_line_num, end_line_num, siblings):
            nonlocal current_level, is_else

            condition_string = re.sub(r"\n\d{4}\s*$", '', condition_string)

            # Add current level conditions and prepare for parent
            current_level["condition"] = condition_string
            current_level["start_line_number"] = start_line_num
            current_level["end_line_number"] = end_line_num
            current_level["siblings"] = siblings  # Add string sibling conditions here
            current_level["is_else"] = is_else  # Indicator to suggest whether condition belongs to else part
            current_level["parent"] = {}  # Create a placeholder for parent
            current_level = current_level["parent"]  # Move to the next level

        # Traverse up the parent hierarchy
        while ctx is not None:
            parent_ctx = ctx.parentCtx  # Get the parent context
            if parent_ctx is None:
                break

            # Check if the parent context is an instance of Parser.IfStatement
            if isinstance(parent_ctx, TSqlParser.If_statementContext) and parent_ctx.children:
                if parent_ctx.ELSE():
                    sym = parent_ctx.ELSE().getSymbol()
                    # If stop index of ELSE is coming before start index of context then it belongs to ELSE part
                    if sym.stop < ctx.stop.start:
                        is_else = True

                # Fetch the condition
                condition_ctx = parent_ctx.search_condition() \
                    if hasattr(parent_ctx, 'search_condition') and parent_ctx.search_condition() else None

                if condition_ctx:
                    update_current_level(self.get_original_text_from_context(condition_ctx), parent_ctx.start.line,
                                         parent_ctx.stop.line, [])

                is_else = False

            # Move to the next parent context
            ctx = parent_ctx

        return hierarchy

    def create_callee(self, ctx, construct_type: CalleeConstructType):

        callee = None

        # updating execution orders and line numbers
        self.execution_order = self.execution_order + 1
        self.callee_execution_order = self.callee_execution_order + 1
        start_line_number = ctx.start.line
        end_line_number = ctx.stop.line

        # get the snippet text of callee statement
        snippet_text = self.get_original_text_from_context(ctx)

        # get the conditional hierarchy and other related attributes for the calle
        is_conditional = False
        condition = 'NA'
        conditional_hierarchy = self.get_callee_conditional_hierarchy(ctx)
        if conditional_hierarchy:
            is_conditional = True
            condition = 'ELSE_OF(' + conditional_hierarchy['condition'] + ')' \
                if conditional_hierarchy['is_else'] else conditional_hierarchy['condition']

        target_function_name = None
        variable_entities = []
        parameters = []

        # fetch the target function name, parameters and variables as per the callee construct type
        if construct_type == self.CalleeConstructType.EXECUTE:
            if ctx.execute_body() and ctx.execute_body().func_proc_name_server_database_schema():
                target_function_name = self.parse_func_proc_name_server_database_schema(
                    ctx.execute_body().func_proc_name_server_database_schema()
                )

                # Extract variables and constants
                variables_details = self.cve.extract_constants_variables(snippet_text, self.path)
                variable_entities = self.create_variable_entity_list(variables_details)
                parameters = [var.term for var in variables_details]

        elif construct_type == self.CalleeConstructType.SCALAR_FUNCTION:
            # extract target function name
            if ctx.scalar_function_name():
                sfn_ctx = ctx.scalar_function_name()
                if sfn_ctx.func_proc_name_server_database_schema():
                    target_function_name = self.parse_func_proc_name_server_database_schema(
                        sfn_ctx.func_proc_name_server_database_schema()
                    )

            # extract variable entities and parameters from the expression list
            if ctx.expression_list_():
                variables = self.get_original_text_from_context(ctx.expression_list_())
                # Extract variables and constants
                variables_details = self.cve.extract_constants_variables(variables, self.path)
                variable_entities = self.create_variable_entity_list(variables_details)
                parameters = [var.term for var in variables_details]

        # only create a callee node if value of target function is fetched
        if target_function_name:
            callee = FunctionCallee(
                execution_order=self.execution_order,
                callee_execution_order=self.callee_execution_order,
                start_line_number=start_line_number,
                end_line_number=end_line_number,
                tech=self.projectMetadata.technology,
                caller_function=self.get_current_context_name(ctx),
                callee_function=target_function_name,
                callee_class=self.file_name,
                callee_file_name=self.file_name,
                callee_file_path=self.path,
                snippet=snippet_text,
                callee_type=construct_type.value,
                type_of_construct='External',
                module_name=self.folder_name,
                behavior=CalleeBehavior.RETURNS.value,
                condition=condition,
                variable=variable_entities,
                parameters=parameters,
                is_conditional=is_conditional,
                conditional_hierarchy=conditional_hierarchy
            )

        # Append the created db call object to its respective function snippet
        if callee:
            if self.current_function:
                self.current_function.callees.append(callee)
            else:
                self.main_program_callees.append(callee)

        return callee

            database, schema, table = table_split  # two dots means three elements

            business_aliases=[],

            end_line_number=end_line,
            program_id=self.file_name,
            type='TABLE',
            snippet=snippet,
            schema_table=table_name.lower(),
            structure_schema='NA',
            fields=[],
            sample_record='NA',
            complexity='NA',
            volume_class='NA',
            main_role='NA',
            relationship_density='NA',
            db_type='Relational',
            description='NA',
            db_tech=self.projectMetadata.technology,
            hosted_at='NA',
            records_type='NA',
            created_from='Program',
            category='NA',
            servers=([server.strip('[]').lower()] if server else []),
            databases=([database.strip('[]').lower()] if database else []),
            schemas=([schema.strip('[]').lower()] if schema else [])
        )

        # Append the created db call object to its respective function snippet

    def visitSelect_statement(self, ctx: TSqlParser.Select_statementContext):
        query = None
        aliases = set()
        tables = set()
        filter_conditions = []
        operation_type = DbCallDataEntity_TYPE.SELECT.value

        try:
            # Fetch the query from the ccontext
            query = self.get_original_text_from_context(ctx).strip(" \n")

            # Fetch the filter condition, table names, aliases from the context
            if ctx.query_expression():
                qe_ctx = ctx.query_expression()

                qs_ctx = qe_ctx.query_specification()
                if qs_ctx.table_sources():
                    for con in qs_ctx.search_condition():
                        if con:
                            filter_conditions.append(self.get_original_text_from_context(con))

                    ts_list = qs_ctx.table_sources().table_source()
                    for ts in ts_list:
                        ret_t, ret_al, ret_fc = self.extract_table_source(ts)
                        tables.update(ret_t)
                        aliases.update(ret_al)
                        filter_conditions.extend(ret_fc)

                if qs_ctx.table_name():
                    tables.add(self.get_original_text_from_context(qs_ctx.table_name()))

                if qe_ctx.sql_union():
                    for union in qe_ctx.sql_union():
                        qs_ctx = union.query_specification()
                        if qs_ctx.table_sources():
                            for con in qs_ctx.search_condition():
                                if con:
                                    filter_conditions.append(self.get_original_text_from_context(con))

                            ts_list = qs_ctx.table_sources().table_source()
                            for ts in ts_list:
                                ret_t, ret_al, ret_fc = self.extract_table_source(ts)
                                tables.update(ret_t)
                                aliases.update(ret_al)
                                filter_conditions.extend(ret_fc)

        except Exception as e:
            logger.error(f"Error while extracting select statement fields: {e}")

        if len(tables) == 0:
            return None
        snippet = self.get_original_text_from_context(ctx)
        table_list=set()
        for table in tables:
            if '.' in table:
                table=table.split('.')[1]
                table_list.add(table)

            table_list.add(table)

            if not self.entity_exists(table.strip()):
                self.create_database_entity(snippet, table.strip(), ctx.start.line, ctx.stop.line)

        db_call = self.create_db_call(ctx, query, filter_conditions, aliases, table_list, operation_type)

        # Adding current dbcall object to stack
        self.db_call_stack.append(db_call)

        self.current_dbcall_depth += 1
        self.visitChildren(ctx)
        self.current_dbcall_depth -= 1

        # Pop the last dbcall object from the stack
        self.db_call_stack.pop()
        fields = self.create_database_field()
        return None

    def create_database_field(self):
        fields = self.column_name
        database_fields = []
        default_table = None
        for key, value in self.table_alias.items():
            if key == value:
                default_table = value
                break
        
        for field_data in fields:
            parts = field_data.split('.')
            if len(parts) == 2:
                table_part = parts[0].strip('[]')
                column_part = parts[1]
                actual_table = self.table_alias.get(table_part)
                if actual_table:
                    actual_table_name = actual_table.split('.')[-1].strip('[]').lower()
                    field_key = (column_part, actual_table_name)
                    if field_key not in self.unique_fields:
                        self.unique_fields.add(field_key)
                        field = DatabaseFieldSchema(
                            name=column_part,
                            table_name=actual_table_name,
                            db_tech=self.projectMetadata.technology
                        )
                        database_fields.append(field)
            elif len(parts) == 1 and default_table:
                column_part = parts[0]
                default_table_name = default_table.split('.')[-1].strip('[]').lower()
                field_key = (column_part, default_table_name)
                if field_key not in self.unique_fields:
                    self.unique_fields.add(field_key)
                    field = DatabaseFieldSchema(
                        name=column_part,
                        table_name=default_table_name,
                        db_tech=self.projectMetadata.technology
                    )
                    database_fields.append(field)
        if self.current_function:
            self.current_function.database_fields.extend(database_fields)
        else:
            self.main_program_database_fields.extend(database_fields)
        self.column_name.clear()
        return database_fields

        fields = self.create_database_field()
        return None

    def visitExpression(self, ctx: TSqlParser.ExpressionContext):
        if ctx.full_column_name():
            full_name = ctx.full_column_name().getText()
            # Split the full name which could be in format: [schema].[table].[column] or [table].[column]
            parts = [part.strip('[]') for part in full_name.split('.')]

            if len(parts) >= 2:  # If we have at least table.column
                table_name = parts[-2]  # Second last part is table name
                column_name = parts[-1]  # Last part is always column name
                self.column_name.add(f"{table_name}.{column_name}")
            else:  # If we only have column name
                self.column_name.add(parts[0])
        return self.visitChildren(ctx)

            for table in tables:
                if not self.entity_exists(table.strip()):
                    self.create_database_entity(query, table.strip(), ctx.start.line, ctx.stop.line)

        elif ctx.rowset_function_limited():
            tables.add(self.get_original_text_from_context(ctx.rowset_function_limited()))

        if ctx.table_sources():
            for ts in ctx.table_sources().table_source():
                ret_t, ret_al, ret_fc = self.extract_table_source(ts)
                tables.update(ret_t)
                aliases.update(ret_al)
                filter_conditions.extend(ret_fc)

        if len(tables) == 0:
            return None

        for table in tables:
            if not self.entity_exists(table.strip()):
                self.create_database_entity(query, table.strip(), ctx.start.line, ctx.stop.line)

        db_call = self.create_db_call(ctx, query, filter_conditions, aliases, tables, operation_type)

        # Adding current dbcall object to stack
        self.db_call_stack.append(db_call)

        self.current_dbcall_depth += 1
        self.visitChildren(ctx)
        self.current_dbcall_depth -= 1

        # Pop the last dbcall object from the stack
        self.db_call_stack.pop()
        fields = self.create_database_field()
        return None

    def visitDelete_statement(self, ctx: TSqlParser.Delete_statementContext):
        aliases = set()
        tables = set()
        filter_conditions = []
        operation_type = DbCallDataEntity_TYPE.DELETE.value

        query = self.get_original_text_from_context(ctx)

        if ctx.delete_statement_from():
            tables.add(self.get_original_text_from_context(ctx.delete_statement_from()))

        if ctx.search_condition():
            filter_conditions.append(self.get_original_text_from_context(ctx.search_condition()))

        if ctx.table_sources():
            for ts in ctx.table_sources().table_source():
                ret_t, ret_al, ret_fc = self.extract_table_source(ts)
                tables.update(ret_t)
                aliases.update(ret_al)
                filter_conditions.extend(ret_fc)

        if len(tables) == 0:
            return None

        db_call = self.create_db_call(ctx, query, filter_conditions, aliases, tables, operation_type)

        # Adding current dbcall object to stack
        self.db_call_stack.append(db_call)

        self.current_dbcall_depth += 1
        self.visitChildren(ctx)
        self.current_dbcall_depth -= 1

        # Pop the last dbcall object from the stack
        self.db_call_stack.pop()
        fields = self.create_database_field()
        return None

    def visitTable_source_item(self, ctx:TSqlParser.Table_source_itemContext):
        if (ctx.full_table_name()  or ctx.LOCAL_ID()) and ctx.as_table_alias():
            if ctx.full_table_name():
                self.table_alias[ctx.as_table_alias().table_alias().getText()] = ctx.full_table_name().getText()
            elif ctx.LOCAL_ID():
                self.table_alias[ctx.as_table_alias().table_alias().getText()] = ctx.LOCAL_ID().getText()

        elif ctx.full_table_name():
            self.table_alias[ctx.full_table_name().getText()] = ctx.full_table_name().getText()


    def visitExecute_statement(self, ctx: TSqlParser.Execute_statementContext):
        target_function_name = None

        if ctx.execute_body() and ctx.execute_body().func_proc_name_server_database_schema():
            target_function_name = self.parse_func_proc_name_server_database_schema(
                ctx.execute_body().func_proc_name_server_database_schema()
            )

        if target_function_name in self.target_functions:
            self.create_service_call(ctx, target_function_name)
        else:
            self.create_callee(ctx, self.CalleeConstructType.EXECUTE)

        return self.visitChildren(ctx)

    def visitRowset_function(self, ctx:TSqlParser.Rowset_functionContext):
        snippet_text = self.get_original_text_from_context(ctx)
        if 'OPENROWSET' in snippet_text.upper():
            target_function_name = 'OPENROWSET'

            self.create_service_call(ctx, target_function_name)

        return self.visitChildren(ctx)

    def visitWhile_statement(self, ctx: TSqlParser.While_statementContext):
        self.execution_order += 1

        # First and foremost increment the execution_order
        self.execution_order = self.execution_order + 1

        full_text = self.get_original_text_from_context(ctx)
        # In many systems, a carriage return character can cause the cursor to move to the beginning of the line,
        # and subsequent characters will overwrite the characters that were already there.
        cleaned_text = full_text.replace('\r', ' ')

        start_line_number = ctx.start.line
        end_line_number = ctx.stop.line

        # Extract variables and constants
        # variables_details = self.cve.extract_constants_variables(cleaned_text, self.path)
        # variable_entities = self.create_variable_entity_list(variables_details)

        hosting_function_name = self.get_current_context_name(ctx)

        loop_entity = LoopEntityNodeSchema(
            name=hosting_function_name,
            tech=self.projectMetadata.technology,
            file_name=self.file_name,
            file_path=self.path,
            start_line_number=start_line_number,
            end_line_number=end_line_number,
            program_id=self.file_name.split('.')[0],
            type=self.LoopEntityType.WHILE.value,
            snippet=cleaned_text,
            execution_order=self.execution_order,
            module_name=self.folder_name
        )

        # Append the created db call object to its respective function snippet
        if self.current_function:
            self.current_function.loop_entity.append(loop_entity)
        else:
            self.main_program_loop_entities.append(loop_entity)

        return self.visitChildren(ctx)

    def visitTransaction_statement(self, ctx: TSqlParser.Transaction_statementContext):
        if ctx.COMMIT():
            self.create_transaction_commit(ctx)

        elif ctx.ROLLBACK():
            self.create_transaction_rollback(ctx)

        return self.visitChildren(ctx)

    def visitSCALAR_FUNCTION(self, ctx: TSqlParser.SCALAR_FUNCTIONContext):
        self.create_callee(ctx, self.CalleeConstructType.SCALAR_FUNCTION)
        return self.visitChildren(ctx)

    def get_operands(self, data):
        # Use regex to find all operators in the string
        operands = re.findall(r'[+\-*/]', data)
        return operands

    def find_target_lines(self, target_data,start_line):
        try:
            target_data_line=target_data.splitlines()
            
            if len(target_data_line) > 1:
                # If target_data is a multi-line string, find the line number range
                start_line = start_line
                end_line = start_line + len(target_data_line) - 1
                return start_line,end_line
            # If target_data is a single line, find the line number
            if len(target_data_line) == 1:
                return start_line, start_line
            # If no match is found, return empty lines
            return [], []
        except Exception as e:
            logger.error(f"Error in find_target_lines: {e}", exc_info=True)
            return []
