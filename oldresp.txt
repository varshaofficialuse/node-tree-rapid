using System;
using System.IO;
using System.Text;
using System.Linq;
using System.Text.RegularExpressions;
using System.Collections.Generic;
using Antlr4.Runtime;
using Antlr4.Runtime.Tree;
using Neo4j.Driver;

public class TSqlCodeParser : BaseParser
{
    public static string RemoveSubsections(
        string FileContent,
        List<FunctionSnippetSchema> Subsequences)
    {
        foreach (var Subsection in Subsequences)
        {
            FileContent = FileContent.Replace(
                Subsection.Snippet,
                string.Join("\n", new string[Subsection.Snippet.Split('\n').Length])
            );
        }

        return FileContent;
    }

    public CodeContext ParseCodebase(
        int RunId,
        int RunFileId,
        CodeComprehensionMetadata CcMetadata,
        string FilePath)
    {
        if (FilePath.Contains(".DS_Store"))
            return null;

        var TsqlPreprocessor = new TSqlPreprocessor();

        var RelPath = Path.GetRelativePath(Directory.GetCurrentDirectory(), FilePath);
        var Actual = RelPath.TrimStart('.', '\\');

        Console.WriteLine($"parsing {FilePath}");

        Dictionary<string, List<string>> Result = new();

        string FileContent = File.ReadAllText(FilePath);
        FileContent = FileContent.Replace("ï»¿", "");
        FileContent = FileContent.Replace("\r\n", "\n").Replace("\r", "\n");

        Match Match = Regex.Match(
            FileContent,
            @"CREATE TABLE \[dbo\]\.\[(\w+)\] \((.*?)\);",
            RegexOptions.Singleline);

        if (Match.Success)
        {
            string Snippet = Match.Value;
            string TableName = Match.Groups[1].Value;
            string FieldsBlock = Match.Groups[2].Value;

            Result[TableName] = new List<string>();

            var Lines = FieldsBlock.Split('\n');
            foreach (var LineRaw in Lines)
            {
                var Line = LineRaw.Trim();
                if (!string.IsNullOrEmpty(Line)
                    && !Line.StartsWith("--")
                    && !Line.StartsWith("GO"))
                {
                    var MatchField = Regex.Match(
                        Line,
                        @"\[(\w+)\]\s+([^,\s]+(?:\s*\([^)]*\))?)");

                    if (MatchField.Success)
                    {
                        string FieldName = MatchField.Groups[1].Value;
                        Result[TableName].Add(FieldName);
                    }
                }
            }

            CreateNodeDatabaseEntity(
                TableName,
                Result[TableName],
                FieldsBlock.TrimStart('\n'),
                CcMetadata.ProjectId,
                RunId,
                FilePath,
                Snippet,
                CcMetadata.Technology);

            CreateHasFieldRelationships(CcMetadata.ProjectId, RunId);

            return new CodeContext { FunctionSnippets = new List<FunctionSnippetSchema>() };
        }

        // -------- VIEW Parsing --------

        var ViewParse = ParseViewFile(FileContent);

        if (ViewParse.ViewName != null)
        {
            CreateNodeDatabaseEntity(
                ViewParse.ViewName,
                ViewParse.Fields,
                ViewParse.FieldsBlock.Trim('\n'),
                CcMetadata.ProjectId,
                RunId,
                FilePath,
                FileContent,
                CcMetadata.Technology,
                ViewParse.Tables,
                "VIEW");

            CreateHasFieldRelationships(CcMetadata.ProjectId, RunId);

            return new CodeContext { FunctionSnippets = new List<FunctionSnippetSchema>() };
        }

        // -------- ANTLR Parsing --------

        FileContent = TsqlPreprocessor.RemoveUnwantedLines(FileContent);
        var RawComments = TsqlPreprocessor.ExtractComments(FileContent);
        var CommentsRemovedCode = TsqlPreprocessor.RemoveComments(FileContent);
        CommentsRemovedCode = CommentsRemovedCode.Replace("ï»¿", "");

        var InputStream = new AntlrInputStream(CommentsRemovedCode);
        var Lexer = new TSqlLexer(InputStream);
        var Stream = new CommonTokenStream(Lexer);

        Console.WriteLine("Starting parsing...");

        var Tree = SaTsql._cpp_parse(InputStream, "tsql_file");

        var Visitor = new CustomTSqlParser(
            projectMetadata: CcMetadata,
            filePath: Actual,
            file_content: FileContent,
            input_stream: InputStream,
            token_stream: Stream
        );

        Console.WriteLine("Visiting tree...");
        Visitor.Visit(Tree);

        foreach (var Snippet in Visitor.FunctionSnippets)
        {
            Snippet.Imports = Visitor.Imports;
            Snippet.RawComments = RawComments;
        }

        return new CodeContext
        {
            FunctionSnippets = Visitor.FunctionSnippets,
            Comments = RawComments.Split('\n').ToList(),
            EntityMetadata = Visitor.EntityMetadata,
            Entities = Visitor.Entities,
            Aliases = Visitor.Aliases,
            InputInterfaces = new List<InputInterfaceSchema>(),
            Miscellaneous = new List<object>()
        };
    }

    // ---------------- VIEW PARSER ----------------

    public (string ViewName, List<string> Fields, List<string> Tables, string FieldsBlock)
        ParseViewFile(string Content)
    {
        Content = Content.Replace("ï»¿", "");
        Content = Regex.Replace(Content, @"/\*.*?\*/", "", RegexOptions.Singleline);
        Content = Regex.Replace(Content, @"--.*", "");

        var ViewMatch = Regex.Match(
            Content,
            @"CREATE\s+VIEW\s+(?:\[[^\]]+\]\.)?(?:dbo\.)?\[?([^\s\[\]]+)\]?",
            RegexOptions.IgnoreCase);

        if (!ViewMatch.Success)
            return (null, null, null, null);

        string ViewName = ViewMatch.Groups[1].Value;

        var SelectMatch = Regex.Match(
            Content,
            @"(?s)(?<=SELECT\s)(.*?)(?=\s+FROM\b)",
            RegexOptions.IgnoreCase | RegexOptions.Singleline);

        string FieldsStr = "";
        HashSet<string> Fields = new();

        if (SelectMatch.Success)
        {
            FieldsStr = SelectMatch.Groups[1].Value;
            Fields.UnionWith(GetViewFields(FieldsStr));
        }

        HashSet<string> Tables = new();
        var TablePattern = new Regex(@"(?:FROM|JOIN)\s+([^\s,]+)", RegexOptions.IgnoreCase);

        foreach (Match M in TablePattern.Matches(Content))
        {
            string Table = M.Groups[1].Value;
            Table = Regex.Split(Table, @"\s+AS\s+|\s+", RegexOptions.IgnoreCase)[0];
            Table = Table.Replace("[", "").Replace("]", "");
            string TableName = Table.Split('.').Last();

            if (!TableName.StartsWith("$"))
                Tables.Add(TableName);
        }

        return (ViewName, Fields.ToList(), Tables.ToList(), FieldsStr);
    }

    public HashSet<string> GetViewFields(string FieldsStr)
    {
        var Fields = new HashSet<string>();
        var RawFields = SplitFields(FieldsStr);

        foreach (var F in RawFields)
        {
            var AliasMatches = Regex.Matches(F, @"AS\s+([\w_]+)", RegexOptions.IgnoreCase);

            string FieldName;

            if (AliasMatches.Count > 0)
            {
                FieldName = AliasMatches[^1].Groups[1].Value
                    .Replace("[", "")
                    .Replace("]", "");
            }
            else
            {
                FieldName = F.Replace("[", "")
                    .Replace("]", "")
                    .Split('.').Last().Trim();
            }

            Fields.Add(FieldName.Trim());
        }

        return Fields;
    }

    public List<string> SplitFields(string S)
    {
        List<string> FieldsList = new();
        string Current = "";
        int ParenCount = 0;
        int CaseCount = 0;
        bool InQuote = false;

        for (int i = 0; i < S.Length; i++)
        {
            char Char = S[i];

            if (Char == '\'')
            {
                InQuote = !InQuote;
                Current += Char;
            }
            else if (Char == '(' && !InQuote)
            {
                ParenCount++;
                Current += Char;
            }
            else if (Char == ')' && !InQuote)
            {
                ParenCount--;
                Current += Char;
            }
            else if (i + 4 <= S.Length && S.Substring(i, 4).ToUpper() == "CASE" && !InQuote)
            {
                CaseCount++;
                Current += Char;
            }
            else if (i + 3 <= S.Length && S.Substring(i, 3).ToUpper() == "END" && CaseCount > 0 && !InQuote)
            {
                CaseCount--;
                Current += Char;
            }
            else if (Char == ',' && ParenCount == 0 && CaseCount == 0 && !InQuote)
            {
                FieldsList.Add(Current.Trim());
                Current = "";
            }
            else
            {
                Current += Char;
            }
        }

        if (!string.IsNullOrWhiteSpace(Current))
            FieldsList.Add(Current.Trim());

        return FieldsList;
    }
}
