        public static string RemoveSubsections(string fileContent, List<FunctionSnippetSchema> subsections)
        {
            foreach (var subsection in subsections)
            {
                int lineCount = subsection.Snippet.Split(new[] { "\r\n", "\r", "\n" }, StringSplitOptions.None).Length;
                fileContent = fileContent.Replace(subsection.Snippet, new string('\n', lineCount));
            }
            return fileContent;
        }

        public CodeContext ParseCodebase(int runId, int runFileId, CodeComprehensionMetadata ccMetadata, string filePath)
        {
            if (filePath.Contains(".DS_Store")) return null;

            var tsqlPreprocessor = new TSqlPreprocessor();
            string relPath = filePath; 
            string actual = relPath;

            string fileContent = "";
            using (var sr = new StreamReader(filePath))
            {
                fileContent = sr.ReadToEnd();
                fileContent = fileContent.Replace("ï»¿", "");
                fileContent = fileContent.Replace("\r\n", "\n").Replace("\r", "\n");

                var match = Regex.Match(fileContent, @"CREATE TABLE \[dbo\]\.\[(\w+)\] \((.*?)\);", RegexOptions.Singleline);
                if (match.Success)
                {
                    string snippet = match.Value;
                    string tableName = match.Groups[1].Value;
                    string fieldsBlock = match.Groups[2].Value;
                    var fieldList = new List<string>();

                    var lines = fieldsBlock.Split('\n');
                    foreach (var line in lines)
                    {
                        string trimmed = line.Trim();
                        if (!string.IsNullOrEmpty(trimmed) && !trimmed.StartsWith("--") && !trimmed.StartsWith("GO"))
                        {
                            var matchField = Regex.Match(trimmed, @"\[(\w+)\]\s+([^,\s]+(?:\s*\([^)]*\))?)");
                            if (matchField.Success) fieldList.Add(matchField.Groups[1].Value);
                        }
                    }

                    if (fieldList.Count > 0)
                    {
                        CreateNodeDatabaseEntity(tableName, fieldList, fieldsBlock.TrimStart('\n'), ccMetadata.ProjectId.ToString(), runId, filePath, snippet, ccMetadata.Language);
                        CreateHasFieldRelationships(ccMetadata.ProjectId.ToString(), runId);
                        return new CodeContext();
                    }
                }
                else
                {
                    match = Regex.Match(fileContent, @"CREATE TABLE \[dbo\]\.\[([^\]]+)\]");
                    if (match.Success)
                    {
                        string snippet = fileContent;
                        string tableName = match.Groups[1].Value;
                        var fieldList = new List<string>();

                        int start = fileContent.IndexOf('(', fileContent.IndexOf("CREATE TABLE"));
                        if (start != -1)
                        {
                            int end = -1;
                            int constraintPos = fileContent.IndexOf("CONSTRAINT", start);
                            if (constraintPos != -1) end = constraintPos;
                            else
                            {
                                int count = 1;
                                int pos = start;
                                while (pos < fileContent.Length - 1 && count > 0)
                                {
                                    pos++;
                                    if (fileContent[pos] == '(') count++;
                                    else if (fileContent[pos] == ')') count--;
                                }
                                end = pos;
                            }
                            string fieldsBlock = fileContent.Substring(start + 1, end - start - 1);
                            foreach (var line in fieldsBlock.Split('\n'))
                            {
                                string trimmed = line.Trim();
                                if (trimmed.StartsWith("[") && trimmed.Contains("]"))
                                {
                                    var fieldMatch = Regex.Match(trimmed, @"\[([^\]]+)\]");
                                    if (fieldMatch.Success) fieldList.Add(fieldMatch.Groups[1].Value);
                                }
                            }
                            CreateNodeDatabaseEntity(tableName, fieldList, fieldsBlock.Trim('\n'), ccMetadata.ProjectId.ToString(), runId, filePath, snippet, ccMetadata.Language);
                            CreateHasFieldRelationships(ccMetadata.ProjectId.ToString(), runId);
                            return new CodeContext();
                        }
                    }
                }

                var (viewName, fields, tables, viewFieldsBlock) = ParseViewFile(fileContent);
                if (!string.IsNullOrEmpty(viewName))
                {
                    CreateNodeDatabaseEntity(viewName, fields, viewFieldsBlock.Trim('\n'), ccMetadata.ProjectId.ToString(), runId, filePath, fileContent, ccMetadata.Language, tables, "VIEW");
                    CreateHasFieldRelationships(ccMetadata.ProjectId.ToString(), runId);
                    return new CodeContext();
                }
            }

            string filename = Path.GetFileName(filePath);

            fileContent = tsqlPreprocessor.RemoveUnwantedLines(fileContent);
            string rawComments = tsqlPreprocessor.ExtractComments(fileContent);
            string commentsRemovedCode = tsqlPreprocessor.RemoveComments(fileContent);
            commentsRemovedCode = commentsRemovedCode.Replace("ï»¿", "");

            var inputStream = new AntlrInputStream(commentsRemovedCode);
            var lexer = new TSqlLexer(inputStream);
            var stream = new CommonTokenStream(lexer);

            var parser = new TSqlParser(stream);
            var tree = parser.tsql_file();

            var visitor = new CustomTSqlParser(ccMetadata, actual, fileContent, inputStream, stream);
            visitor.Visit(tree);

            foreach (var snippet in visitor.function_snippets)
            {
                snippet.Imports = visitor.imports;
                snippet.RawComments = rawComments;
            }

            return new CodeContext
            {
                FunctionSnippets = visitor.function_snippets,
                Callees = visitor.main_program_callees,
                ServiceCalls = visitor.main_program_service_calls,
                DbCalls = visitor.main_program_dbcalls,
                Calculations = visitor.main_program_calculations,
                DecisionNodes = visitor.main_program_decision_nodes,
                TransactionCommits = visitor.main_program_transaction_commits,
                TransactionRollbacks = visitor.main_program_transaction_rollbacks,
                LoopEntities = visitor.main_program_loop_entities,
                DatabaseEntities = visitor.main_program_database_entities,
                DatabaseFields = visitor.main_program_database_fields
            };
        }

        public (string, List<string>, List<string>, string) ParseViewFile(string content)
        {
            content = content.Replace("ï»¿", "");
            content = Regex.Replace(content, @"/\*.*?\*/", "", RegexOptions.Singleline);
            content = Regex.Replace(content, @"--.*", "");

            var viewNameMatch = Regex.Match(content, @"CREATE\s+VIEW\s+(?:\[[^\]]+\]\.)?(?:dbo\.)?\[?([^\s\[\]]+)\]?", RegexOptions.IgnoreCase);
            if (!viewNameMatch.Success) return (null, null, null, null);
            string viewName = viewNameMatch.Groups[1].Value;

            var selectFromMatch = Regex.Match(content, @"(?s)(?<=SELECT\s)(.*?)(?=\s+FROM\b)", RegexOptions.IgnoreCase);
            var selectFromMatch1 = Regex.Match(content, @"(?s)(?<=SELECT\s)(.*?)(?=\s+FROM\b)");
            string fieldsStr = "";
            var fields = new HashSet<string>();
            if (selectFromMatch.Success)
            {
                fieldsStr = selectFromMatch.Groups[1].Value;
                foreach (var f in GetViewFields(fieldsStr)) fields.Add(f);
            }
            if (selectFromMatch1.Success)
            {
                fieldsStr = selectFromMatch1.Groups[1].Value;
                foreach (var f in GetViewFields(fieldsStr)) fields.Add(f);
            }

            var tables = new HashSet<string>();
            var tablePattern = new Regex(@"(?:FROM|JOIN)\s+([^\s,]+)", RegexOptions.IgnoreCase);
            foreach (Match match in tablePattern.Matches(content))
            {
                string table = match.Groups[1].Value;
                table = Regex.Split(table, @"\s+AS\s+|\s+", RegexOptions.IgnoreCase)[0];
                string tableClean = table.Replace("[", "").Replace("]", "");
                string tableName = tableClean.Split('.').Last();
                if (!tableName.StartsWith("$")) tables.Add(tableName);
            }

            return (viewName, fields.ToList(), tables.ToList(), fieldsStr);
        }

        public HashSet<string> GetViewFields(string fieldsStr)
        {
            var fields = new HashSet<string>();
            string[] sqlReservedKeywords = { "SELECT", "FROM", "WHERE", "INSERT", "UPDATE", "DELETE", "JOIN", "INNER", "LEFT", "RIGHT", "FULL", "GROUP", "ORDER", "BY", "HAVING", "DISTINCT", "AS", "AND", "OR", "NOT", "NULL", "IS", "IN", "BETWEEN", "LIKE", "CREATE", "DROP", "ALTER", "TABLE", "DATABASE", "INDEX", "VIEW", "TRIGGER", "PRIMARY", "KEY", "FOREIGN", "CONSTRAINT", "DEFAULT", "CHECK", "EXISTS", "UNION", "ALL", "ANY", "CASE", "WHEN", "THEN", "ELSE", "END", "LIMIT", "OFFSET", "VALUES", "SET", "TOP", "ASC", "DESC", "TRANSACTION", "COMMIT", "ROLLBACK", "GRANT", "REVOKE", "WITH", "NOLOCK", "ON", "OUTER", "CROSS", "NATURAL", "USING", "INNER JOIN" };
            
            var rawFields = SplitFields(fieldsStr);
            foreach (var f in rawFields)
            {
                var aliasMatches = Regex.Matches(f, @"AS\s+([\w_]+)", RegexOptions.IgnoreCase);
                string fieldName = "";
                if (aliasMatches.Count > 0)
                {
                    fieldName = aliasMatches[aliasMatches.Count - 1].Groups[1].Value.Trim().Replace("[", "").Replace("]", "");
                    if (fieldName.ToLower() == "changed") fieldName = "Chng";
                    else if (fieldName.ToLower() == "chngdt") fieldName = "ChngDt";
                    else if (fieldName.ToLower() == "chngby") fieldName = "ChngBy";
                }
                else
                {
                    fieldName = f.Replace("[", "").Replace("]", "").Split('.').Last().Trim();
                }

                foreach (var sqlKey in sqlReservedKeywords)
                {
                    if (fieldName.ToUpper().Contains(sqlKey + " ") || fieldName.ToUpper().Contains(" " + sqlKey))
                    {
                        fieldName = Regex.Replace(fieldName, sqlKey, "", RegexOptions.IgnoreCase);
                    }
                }
                fields.Add(fieldName.Trim());
            }
            return fields;
        }

        public List<string> SplitFields(string s)
        {
            var fieldsList = new List<string>();
            string current = "";
            int parenCount = 0;
            int caseCount = 0;
            bool inQuote = false;
            for (int i = 0; i < s.Length; i++)
            {
                char c = s[i];
                if (c == '\'')
                {
                    inQuote = !inQuote;
                    current += c;
                }
                else if (c == '(' && !inQuote)
                {
                    parenCount++;
                    current += c;
                }
                else if (c == ')' && !inQuote)
                {
                    parenCount--;
                    current += c;
                }
                else if (i + 4 <= s.Length && s.Substring(i, 4).ToUpper() == "CASE" && !inQuote)
                {
                    caseCount++;
                    current += c;
                }
                else if (i + 3 <= s.Length && s.Substring(i, 3).ToUpper() == "END" && caseCount > 0 && !inQuote)
                {
                    caseCount--;
                    current += c;
                }
                else if (c == ',' && parenCount == 0 && caseCount == 0 && !inQuote)
                {
                    fieldsList.Add(current.Trim());
                    current = "";
                }
                else
                {
                    current += c;
                }
            }
            if (!string.IsNullOrWhiteSpace(current)) fieldsList.Add(current.Trim());
            return fieldsList;
        }
