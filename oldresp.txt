using System;
using System.Collections.Generic;
using System.Text.RegularExpressions;

namespace TsqlParserProject
{
    public class TSqlPreprocessor
    {
        private string single_line_comment_pattern = @"(?:^|\s)(?:---|--)[^\n]*";
        private string multi_line_comment_pattern = @"/\*[\s\S]*?\*/";

        private List<string> unwanted_line_patterns = new List<string>
        {
            @"^[ \t]*Text\s+--.*$",
            @"^[ \t]*Completion\stime:.*$"
        };

        public TSqlPreprocessor()
        {
        }

        public string RemoveUnwantedLines(string file_content)
        {
            if (file_content == null)
            {
                return null;
            }

            foreach (string pattern in unwanted_line_patterns)
            {
                file_content = Regex.Replace(
                    file_content,
                    pattern,
                    delegate (Match match)
                    {
                        int newlineCount = 0;
                        string value = match.Value;
                        for (int i = 0; i < value.Length; i++)
                        {
                            if (value[i] == '\n')
                            {
                                newlineCount++;
                            }
                        }

                        return new string('\n', newlineCount);
                    },
                    RegexOptions.Multiline
                );
            }

            return file_content;
        }

        public string ExtractComments(string file_content)
        {
            if (file_content == null)
            {
                return null;
            }

            List<string> comments = new List<string>();

            MatchCollection single_line_matches = Regex.Matches(
                file_content,
                single_line_comment_pattern,
                RegexOptions.Multiline
            );

            foreach (Match match in single_line_matches)
            {
                comments.Add(match.Value);
            }

            MatchCollection multi_line_matches = Regex.Matches(
                file_content,
                multi_line_comment_pattern,
                RegexOptions.Multiline
            );

            foreach (Match match in multi_line_matches)
            {
                comments.Add(match.Value);
            }

            return string.Join("\n", comments);
        }

        public string RemoveComments(string file_content)
        {
            if (file_content == null)
            {
                return null;
            }

            string text_without_comments = Regex.Replace(
                file_content,
                single_line_comment_pattern,
                "",
                RegexOptions.Multiline
            );

            text_without_comments = Regex.Replace(
                text_without_comments,
                multi_line_comment_pattern,
                delegate (Match match)
                {
                    int newlineCount = 0;
                    string value = match.Value;
                    for (int i = 0; i < value.Length; i++)
                    {
                        if (value[i] == '\n')
                        {
                            newlineCount++;
                        }
                    }

                    return new string('\n', newlineCount);
                },
                RegexOptions.Multiline
            );

            return text_without_comments;
        }
    }
}

using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Reflection;
using System.Text.RegularExpressions;
using ParserService.Entities.Entities;

namespace TsqlParserProject
{
    public class ConstantVariableExtractor
    {
        private HashSet<string> keywords = new();
        private string negative_lookahead;
        private string variable_pattern;
        private Regex variable_regex;
        private Regex keywords_as_vars;
        private string string_constant = @"'([^']*)'";
        private Regex string_constant_regex;
        private Regex numeric_constant_regex;
        private string calc_pattern;
        private Regex calc_regex;
        private string pattern_resultant = @"SET\s+(@\w+)\s+=";

        public ConstantVariableExtractor()
        {
            keywords = new HashSet<string>(StringComparer.OrdinalIgnoreCase);

            ReadKeywords();

            negative_lookahead = string.Join("|", keywords.Select(k => Regex.Escape(k)));

            variable_pattern = $@"(?<!@)(?!(?:{negative_lookahead})(?=[\s(]|$))\b[A-Z]+(?:[-._][A-Z0-9]+)*\b";
            variable_regex = new Regex(variable_pattern);

            keywords_as_vars = new Regex(@"(?:(?<=\s)|(?<=^))(@[A-Z][A-Z0-9_]*)(?=[\s(]|$)");
            string_constant_regex = new Regex(string_constant);
            numeric_constant_regex = new Regex(@"(?<=\s)\d+(?=\s)|(?<=\s)\d+$");

            calc_pattern =
                $@"(?!(?:T|CT|ECT|LECT|ELECT|{negative_lookahead})(?=\s|$))" +
                $@"(?:(?:SET|SELECT)?\s*(@[\w]+)?\s*(?:=\s*)?)?" +
                $@"\(*\s*@[\w.]+(?:\s*[+\-*/]\s*[@\w.()]+)+\s*\)*";

            calc_regex = new Regex(calc_pattern);
        }

        public void ReadKeywords()
        {
            try
            {
                string currentDir = Path.GetDirectoryName(Assembly.GetExecutingAssembly().Location);
                string targetFilePath = Path.GetFullPath(
                    Path.Combine(currentDir ?? "", "..", "..", "grammer", "TSqlLexer.g4"));

                string startMarker = "// Keywords Start Here";
                string endMarker = "// Keywords End Here";
                bool capture = false;

                if (!File.Exists(targetFilePath))
                    return;

                foreach (var lineRaw in File.ReadLines(targetFilePath))
                {
                    string line = lineRaw.Trim();

                    if (line == startMarker)
                    {
                        capture = true;
                        continue;
                    }
                    else if (line == endMarker)
                    {
                        capture = false;
                        break;
                    }

                    if (capture && !string.IsNullOrEmpty(line) && !line.StartsWith("//"))
                    {
                        var colonPart = Regex.Match(line, @":(.*?);");
                        if (colonPart.Success)
                        {
                            var afterColon = colonPart.Groups[1].Value;
                            var keywordMatches = Regex.Matches(afterColon, @"'[^']*'");

                            foreach (Match k in keywordMatches)
                            {
                                keywords.Add(k.Value.Trim(' ', '\''));
                            }
                        }
                    }
                }
            }
            catch (Exception)
            {
            }
        }

        public List<Dictionary<string, object>> ExtractMathematicalCalculations(string codeSnippet)
        {
            var matchedList = new List<Dictionary<string, object>>();

            if (string.IsNullOrEmpty(codeSnippet))
                return matchedList;

            try
            {
                foreach (Match match in calc_regex.Matches(codeSnippet))
                {
                    var dataDict = new Dictionary<string, object>
                    {
                        { "match_group", match.Value },
                        { "start", match.Index },
                        { "end", match.Index + match.Length }
                    };

                    matchedList.Add(dataDict);
                }
            }
            catch (Exception)
            {
            }

            return matchedList;
        }

        public List<Dictionary<string, object>> ExtractMathematicalCalculationsStartLine(string codeSnippet)
        {
            var matchedList = new List<Dictionary<string, object>>();

            if (string.IsNullOrEmpty(codeSnippet))
                return matchedList;

            var lines = codeSnippet.Split('\n');

            try
            {
                for (int i = 0; i < lines.Length; i++)
                {
                    foreach (Match match in calc_regex.Matches(lines[i]))
                    {
                        var dataDict = new Dictionary<string, object>
                        {
                            { "match_group", match.Value },
                            { "start", i + 1 }
                        };

                        matchedList.Add(dataDict);
                    }
                }
            }
            catch (Exception)
            {
            }

            return matchedList;
        }

        public List<DataDefinitionJSON> ExtractConstantsVariables(string codeSnippet, string filePath)
        {
            var variableDetails = new List<DataDefinitionJSON>();

            if (string.IsNullOrEmpty(codeSnippet))
                return variableDetails;

            var lines = codeSnippet.Split('\n');
            var variableDict = new Dictionary<string, DataDefinitionJSON>();

            foreach (var line in lines)
            {
                if (line.Trim().StartsWith("--"))
                    continue;

                foreach (Match match in string_constant_regex.Matches(line))
                {
                    string constant = match.Groups[1].Value;

                    if (!string.IsNullOrWhiteSpace(constant) &&
                        constant.Trim('\'').Split((char[])null, StringSplitOptions.RemoveEmptyEntries).Length == 1)
                    {
                        if (!variableDict.ContainsKey(constant))
                        {
                            variableDict[constant] = new DataDefinitionJSON
                            {
                                Type = "CONSTANT",
                                Term = constant,
                                UsageContext = new List<string> { line },
                                FilePath = filePath,
                                Meaning = ""
                            };
                        }
                        else
                        {
                            variableDict[constant].UsageContext.Add(line);
                        }
                    }
                }

                string upperLine = line.ToUpperInvariant();

                foreach (Match m in variable_regex.Matches(upperLine))
                {
                    string variable = line.Substring(m.Index, m.Length);

                    if (!string.IsNullOrWhiteSpace(variable) &&
                        variable.Trim('\'').Split((char[])null, StringSplitOptions.RemoveEmptyEntries).Length == 1)
                    {
                        if (!variableDict.ContainsKey(variable))
                        {
                            variableDict[variable] = new DataDefinitionJSON
                            {
                                Type = "DECLARATION",
                                Term = variable,
                                UsageContext = new List<string> { line },
                                FilePath = filePath,
                                Meaning = ""
                            };
                        }
                        else
                        {
                            variableDict[variable].UsageContext.Add(line);
                        }
                    }
                }

                foreach (Match m in keywords_as_vars.Matches(upperLine))
                {
                    string keyVar = line.Substring(m.Index, m.Length);

                    if (!string.IsNullOrWhiteSpace(keyVar) &&
                        keyVar.Trim('\'').Split((char[])null, StringSplitOptions.RemoveEmptyEntries).Length == 1)
                    {
                        if (!variableDict.ContainsKey(keyVar))
                        {
                            variableDict[keyVar] = new DataDefinitionJSON
                            {
                                Type = "DECLARATION",
                                Term = keyVar,
                                UsageContext = new List<string> { line },
                                FilePath = filePath,
                                Meaning = ""
                            };
                        }
                        else
                        {
                            variableDict[keyVar].UsageContext.Add(line);
                        }
                    }
                }
            }

            variableDetails.AddRange(variableDict.Values);

            return variableDetails;
        }

        public List<DataDefinitionJSON> ExtractNumericConstants(string codeSnippet, string filePath)
        {
            var variableDetails = new List<DataDefinitionJSON>();

            if (string.IsNullOrEmpty(codeSnippet))
                return variableDetails;

            var lines = codeSnippet.Split('\n');
            var variableDict = new Dictionary<string, DataDefinitionJSON>();

            foreach (var line in lines)
            {
                foreach (Match match in numeric_constant_regex.Matches(line))
                {
                    string constant = match.Value;

                    if (!string.IsNullOrWhiteSpace(constant) &&
                        constant.Trim('\'').Split((char[])null, StringSplitOptions.RemoveEmptyEntries).Length == 1)
                    {
                        if (!variableDict.ContainsKey(constant))
                        {
                            variableDict[constant] = new DataDefinitionJSON
                            {
                                Type = "CONSTANT",
                                Term = constant,
                                UsageContext = new List<string> { line },
                                FilePath = filePath,
                                Meaning = ""
                            };
                        }
                        else
                        {
                            variableDict[constant].UsageContext.Add(line);
                        }
                    }
                }
            }

            variableDetails.AddRange(variableDict.Values);

            return variableDetails;
        }
    }
