using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Text.RegularExpressions;
using Antlr4.Runtime;
using Antlr4.Runtime.Tree;
using ParserService.Entities.Schemas;
using ParserService.Grammar.Parsers;

namespace TsqlParserProject
{
    public class TSqlCodeParser
    {
        public static string RemoveSubsections(string fileContent, List<FunctionSnippetSchema> subsections)
        {
            foreach (var subsection in subsections)
            {
                var lineCount = subsection.Snippet
                    .Split(new[] { "\r\n", "\r", "\n" }, StringSplitOptions.None)
                    .Length;

                fileContent = fileContent.Replace(
                    subsection.Snippet,
                    string.Concat(Enumerable.Repeat("\n", lineCount))
                );
            }

            return fileContent;
        }

        public CodeContext ParseCodebase(
            int runId,
            int runFileId,
            CodeComprehensionMetadata ccMetadata,
            string filePath)
        {
            if (filePath.Contains(".DS_Store"))
                return null;

            var tsqlPreprocessor = new TSqlPreprocessor();

            // Mirror Python relpath logic
            string relPath = Path.GetRelativePath(Directory.GetCurrentDirectory(), filePath);
            string actual = relPath.TrimStart('.', '\\');

            string fileContent;

            using (var sr = new StreamReader(filePath))
            {
                fileContent = sr.ReadToEnd();
            }

            fileContent = fileContent.Replace("ï»¿", "");
            fileContent = fileContent.Replace("\r\n", "\n").Replace("\r", "\n");

            // -------- PREPROCESSING (EXACT ORDER AS PYTHON) --------

            fileContent = tsqlPreprocessor.RemoveUnwantedLines(fileContent);
            string rawComments = tsqlPreprocessor.ExtractComments(fileContent);
            string commentsRemovedCode = tsqlPreprocessor.RemoveComments(fileContent);
            commentsRemovedCode = commentsRemovedCode.Replace("ï»¿", "");

            // -------- ANTLR PIPELINE (MATCH PYTHON STYLE) --------

            var inputStream = new AntlrInputStream(commentsRemovedCode);
            var lexer = new TSqlLexer(inputStream);
            var stream = new CommonTokenStream(lexer);

            var parser = new TSqlParser(stream);

            // Python: tree = sa_tsql._cpp_parse(input_stream, "tsql_file")
            // Closest structural equivalent in C#
            var tree = parser.tsql_file();

            var visitor = new CustomTSqlParser(
                ccMetadata,
                actual,
                fileContent,
                inputStream,
                stream
            );

            visitor.Visit(tree);

            // -------- TOKEN RETRIEVAL (MATCH PYTHON) --------

            var tokens = stream.GetTokens(0, stream.Index);

            // -------- POST VISITOR PROCESSING --------

            foreach (var snippet in visitor.function_snippets)
            {
                snippet.Imports = visitor.imports;
                snippet.RawComments = rawComments;
            }

            // -------- MATCH PYTHON LIST EXTRACTION --------

            var functionSnippets = visitor.function_snippets;
            var comments = rawComments.Split('\n').ToList();
            var entityMetadata = visitor.entity_metadata;
            var entities = visitor.entities;
            var aliases = visitor.aliases;

            var inputInterfaces = new List<InputInterfaceSchema>();
            var miscellaneous = new List<object>();

            // -------- RETURN EXACT PYTHON STRUCTURE --------

            return new CodeContext
            {
                FunctionSnippets = functionSnippets,
                Comments = comments,
                EntityMetadata = entityMetadata,
                Entities = entities,
                Aliases = aliases,
                InputInterfaces = inputInterfaces,
                Miscellaneous = miscellaneous
            };
        }
    }
}
