    }     private string GetOriginalTextFromContext(ParserRuleContext ctx)
    {
        int startIdx = ctx.Start.StartIndex;
        int stopIdx = ctx.Stop.StopIndex;

        return InputStream.GetText(new Interval(startIdx, stopIdx));
    }

    private string GetSnippetTextFromContext(ParserRuleContext ctx)
    {
        int startIdx = ctx.Start.TokenIndex;
        int stopIdx = ctx.Stop.TokenIndex;

        var tokens = TokenStream.GetTokens(startIdx, stopIdx);
        return string.Concat(tokens
            .Where(t => !CommentTokens.Contains(t.Type))
            .Select(t => t.Text));
    }

    private string GetCurrentContextName(ParserRuleContext ctx)
    {
        while (ctx != null)
        {
            if (ctx is TSqlParser.Create_or_alter_functionContext funcCtx)
                return funcCtx.funcName.procedure().GetText().Trim('[', ']');

            if (ctx is TSqlParser.Create_or_alter_procedureContext procCtx)
                return procCtx.procName.procedure().GetText().Trim('[', ']');

            if (ctx is TSqlParser.Create_or_alter_dml_triggerContext dmlCtx)
                return dmlCtx.simple_name().name.GetText().Trim('[', ']');

            if (ctx is TSqlParser.Create_or_alter_ddl_triggerContext ddlCtx)
                return ddlCtx.simple_name().name.GetText().Trim('[', ']');

            ctx = ctx.Parent;
        }

        return MAIN_SNIPPET;
    }

    private List<VariableEntity> CreateVariableEntityList(IEnumerable<VariableEntity> variables)
    {
        var variableList = new List<VariableEntity>();

        foreach (var variable in variables)
        {
            variableList.Add(new VariableEntity
            {
                Term = variable.Term,
                Type = variable.Type,
                UsageContext = variable.UsageContext,
                Meaning = variable.Meaning,
                FilePath = variable.FilePath
            });
        }

        return variableList;
    }

    private (HashSet<string>, HashSet<string>, List<string>) ExtractTableSource(
        TSqlParser.Table_sourceContext tableSource)
    {
        var tables = new HashSet<string>();
        var aliases = new HashSet<string>();
        var filterConditions = new List<string>();

        if (tableSource != null)
        {
            var tsi = tableSource.table_source_item();

            if (tsi?.as_table_alias() != null)
                aliases.Add(GetOriginalTextFromContext(tsi.as_table_alias()));

            if (tsi?.full_table_name() != null)
                tables.Add(GetOriginalTextFromContext(tsi.full_table_name()));

            if (tableSource.join_part() != null)
            {
                foreach (var join in tableSource.join_part())
                {
                    if (join.join_on() != null)
                    {
                        var joinOn = join.join_on();

                        if (joinOn.search_condition() != null)
                            filterConditions.Add(GetOriginalTextFromContext(joinOn.search_condition()));

                        var result = ExtractTableSource(joinOn.table_source());
                        tables.UnionWith(result.Item1);
                        aliases.UnionWith(result.Item2);
                        filterConditions.AddRange(result.Item3);
                    }
                }
            }
        }

        return (tables, aliases, filterConditions);
    }

    private void CreateExtractedCalculationSnippet(
        string snippet,
        string functionName,
        int funcStartLine)
    {
        var extractData = Cve.ExtractMathematicalCalculations(snippet);
        var extractDataStartLine = Cve.ExtractMathematicalCalculationsStartline(snippet);

        string resultantData = "";
        string expressionData = "";

        for (int cnt = 0; cnt < extractData.Count; cnt++)
        {
            try
            {
                var lineDict = extractData[cnt];

                if (lineDict["match_group"].Contains("="))
                {
                    var matchData = lineDict["match_group"].Split('=');
                    resultantData = matchData[0];
                    expressionData = matchData[1];
                }
                else
                {
                    expressionData = lineDict["match_group"];
                }

                var strMatchedData = string.Join(" ", expressionData);
                var variablesDetails = Cve.ExtractConstantsVariables(strMatchedData, Path);
                var variableEntities = CreateVariableEntityList(variablesDetails);

                bool keyword = expressionData.Contains("SELECT");

                if (!keyword)
                {
                    var operands = GetOperands(expressionData.Trim());
                    var lineRange = FindTargetLines(
                        expressionData.Trim(),
                        extractDataStartLine[cnt]["start"]
                    );

                    var calculationEntity = new CalculationEntity
                    {
                        Name = functionName,
                        Tech = ProjectMetadata.Technology,
                        FileName = FileName,
                        FilePath = Path,
                        StartLineNumber = lineRange.Item1,
                        EndLineNumber = lineRange.Item2,
                        ProgramId = FileName,
                        Type = "",
                        Snippet = expressionData.Trim(),
                        Resultant = resultantData,
                        Operands = operands,
                        Variable = variableEntities
                    };

                    if (CurrentFunction != null)
                        CurrentFunction.Calculation.Add(calculationEntity);
                    else
                        MainProgramCalculations.Add(calculationEntity);
                }
            }
            catch
            {
                // Strict parity: swallow exception exactly as Python logger error behavior
            }
        }
    }      private FunctionSnippetSchema CreateFunctionSnippet(ParserRuleContext ctx)
    {
        ExecutionOrder = 0;
        CalleeExecutionOrder = 0;
        CurrentIfDepth = 0;
        CurrentCaseDepth = 0;
        SnippetExecutionOrder += 1;

        int startLineNumber = ctx.Start.Line;
        int endLineNumber = ctx.Stop.Line;

        string functionName = GetCurrentContextName(ctx);
        string className = FileName;

        if (ctx is TSqlParser.Create_or_alter_functionContext funcCtx)
            className = funcCtx.funcName.GetText();

        if (ctx is TSqlParser.Create_or_alter_procedureContext procCtx)
            className = procCtx.procName.GetText();

        if (ctx is TSqlParser.Create_or_alter_dml_triggerContext dmlCtx)
            className = dmlCtx.simple_name().GetText();

        if (ctx is TSqlParser.Create_or_alter_ddl_triggerContext ddlCtx)
            className = ddlCtx.simple_name().GetText();

        string snippetText = GetOriginalTextFromContext(ctx);

        var variablesDetails = Cve.ExtractConstantsVariables(snippetText, Path);
        var variableEntities = CreateVariableEntityList(variablesDetails);

        var functionSnippet = new FunctionSnippetSchema
        {
            FilePath = Path,
            FileName = FileName,
            FunctionName = functionName,
            ClassName = className,
            Snippet = snippetText,
            StartLineNumber = startLineNumber,
            EndLineNumber = endLineNumber,
            Callees = new List<FunctionCallee>(),
            Annotations = new List<object>(),
            ReturnType = "NA",
            DecisionNodeConstructs = new List<DecisionNodeSchema>(),
            DbCalls = new List<DbCallDataEntity>(),
            ServiceCalls = new List<ServiceCallDataEntity>(),
            Calculation = new List<CalculationEntity>(),
            ExecutionOrder = ExecutionOrder,
            SnippetExecutionOrder = SnippetExecutionOrder,
            Inputs = new List<InputInterfaceSchema>(),
            Variable = variableEntities
        };

        FunctionSnippets.Add(functionSnippet);

        return functionSnippet;
    }

    private void VisitFunctionSnippet(ParserRuleContext ctx)
    {
        var functionSnippet = CreateFunctionSnippet(ctx);
        int functionStartLine = ctx.Start.StartIndex;

        CurrentFunction = functionSnippet;
        FunctionSnippetStack.Push(functionSnippet);

        CreateExtractedCalculationSnippet(
            CurrentFunction.Snippet,
            CurrentFunction.FunctionName,
            functionStartLine
        );

        VisitChildren(ctx);

        CurrentFunction = FunctionSnippetStack.Count > 0
            ? FunctionSnippetStack.Pop()
            : null;
    }

    private DecisionNodeSchema CreateRuleSnippet(
        ParserRuleContext ctx,
        RuleConstructType constructType)
    {
        int startLineNumber = ctx.Start.Line;
        int endLineNumber = ctx.Stop.Line;

        Regex conditionalKeywords;

        if (constructType == RuleConstructType.SWITCH_CONSTRUCT)
            conditionalKeywords = new Regex(@"\b(WHEN|ELSE)\b", RegexOptions.IgnoreCase);
        else
            conditionalKeywords = new Regex(@"\b((?<!END-)IF|ELSE|(?<!END-)DECIDE)\b", RegexOptions.IgnoreCase);

        string snippetText = GetOriginalTextFromContext(ctx);

        var matches = conditionalKeywords.Matches(snippetText);
        int numConditions = matches.Count;

        string cleanedText = snippetText.Replace("\r", " ");

        string hostingFunctionName = GetCurrentContextName(ctx);

        var variablesDetails = Cve.ExtractConstantsVariables(cleanedText, Path);

        string numericExtractionText = snippetText;

        if (ctx is TSqlParser.If_statementContext ifCtx &&
            ifCtx.search_condition() != null)
        {
            numericExtractionText = GetOriginalTextFromContext(ifCtx.search_condition());
        }

        variablesDetails.AddRange(
            Cve.ExtractNumericConstants(
                numericExtractionText.Replace("\r", " "),
                Path
            )
        );

        var variableEntities = CreateVariableEntityList(variablesDetails);

        var decisionNode = new DecisionNodeSchema
        {
            Name = hostingFunctionName,
            ShortSummary = "",
            StartLineNumber = startLineNumber,
            EndLineNumber = endLineNumber,
            Snippet = snippetText,
            NumberOfConditionals = numConditions,
            Complexity = Complexity.CategorizeComplexity(numConditions),
            ConstructType = constructType,
            ExecutionOrder = ExecutionOrder,
            Variable = variableEntities
        };

        return decisionNode;
    }

    private DbCallDataEntity CreateDbCall(
        ParserRuleContext ctx,
        string query,
        List<string> filterConditions,
        HashSet<string> aliases,
        HashSet<string> tables,
        string operationType)
    {
        ExecutionOrder += 1;

        string fullText = GetOriginalTextFromContext(ctx);
        string cleanedText = fullText.Replace("\r", " ");

        int startLineNumber = ctx.Start.Line;
        int endLineNumber = ctx.Stop.Line;

        var variablesDetails = Cve.ExtractConstantsVariables(cleanedText, Path);
        var variableEntities = CreateVariableEntityList(variablesDetails);

        string hostingFunctionName = GetCurrentContextName(ctx);

        var dbCallEntity = new DbCallDataEntity
        {
            Name = hostingFunctionName,
            FilePath = Path,
            FileName = FileName,
            Tech = ProjectMetadata.Technology,
            ProgramId = FileName,
            StartLineNumber = startLineNumber,
            EndLineNumber = endLineNumber,
            Snippet = cleanedText,
            OperationType = operationType,
            ExecutionOrder = ExecutionOrder,
            Variable = variableEntities,
            Query = query,
            FilterConditions = filterConditions,
            Aliases = aliases,
            TableNames = tables
                .Select(t => t.Split('.').Last().Trim('[', ']').ToLower())
                .ToList()
        };

        if (CurrentFunction != null)
            CurrentFunction.DbCalls.Add(dbCallEntity);
        else
            MainProgramDbCalls.Add(dbCallEntity);

        if (DbCallStack.Count == 0)
        {
            DbCallExecutionOrder += 1;
            dbCallEntity.Level = DbCallLevel.PARENT;
            dbCallEntity.DbCallExecutionOrder = DbCallExecutionOrder.ToString();
            dbCallEntity.ChildCount = 0;
        }
        else
        {
            var parent = DbCallStack.Peek();
            parent.ChildCount += 1;

            dbCallEntity.Level = DbCallLevel.CHILD;
            dbCallEntity.DbCallExecutionOrder =
                parent.DbCallExecutionOrder + "." + parent.ChildCount;
            dbCallEntity.ChildCount = 0;
        }

        return dbCallEntity;
    }

    private Dictionary<string, object> GetCalleeConditionalHierarchy(ParserRuleContext ctx)
    {
        var hierarchy = new Dictionary<string, object>();
        var currentLevel = hierarchy;
        bool isElse = false;

        while (ctx != null)
        {
            var parentCtx = ctx.Parent;

            if (parentCtx == null)
                break;

            if (parentCtx is TSqlParser.If_statementContext ifCtx)
            {
                if (ifCtx.ELSE() != null)
                {
                    var sym = ifCtx.ELSE().Symbol;
                    if (sym.StopIndex < ctx.Stop.StartIndex)
                        isElse = true;
                }

                if (ifCtx.search_condition() != null)
                {
                    string conditionString =
                        GetOriginalTextFromContext(ifCtx.search_condition());

                    conditionString =
                        Regex.Replace(conditionString, @"\n\d{4}\s*$", "");

                    currentLevel["condition"] = conditionString;
                    currentLevel["start_line_number"] = ifCtx.Start.Line;
                    currentLevel["end_line_number"] = ifCtx.Stop.Line;
                    currentLevel["siblings"] = new List<string>();
                    currentLevel["is_else"] = isElse;
                    currentLevel["parent"] = new Dictionary<string, object>();

                    currentLevel = (Dictionary<string, object>)currentLevel["parent"];
                }

                isElse = false;
            }

            ctx = parentCtx;
        }

        return hierarchy;

    private void HandleBeginTransaction(ParserRuleContext ctx)
    {
        ExecutionOrder += 1;
        DbCallStack.Push(new DbCallDataEntity());
    }

    private void HandleCommit(ParserRuleContext ctx)
    {
        if (DbCallStack.Count > 0)
            DbCallStack.Pop();
    }

    private void HandleRollback(ParserRuleContext ctx)
    {
        if (DbCallStack.Count > 0)
            DbCallStack.Pop();
    }

    public override object VisitSelect_statement(
        TSqlParser.Select_statementContext context)
    {
        string query = GetOriginalTextFromContext(context);
        var tables = ExtractTables(query);
        var aliases = ExtractAliases(query);
        var filters = ExtractFilterConditions(query);

        var dbCall = CreateDbCall(
            context,
            query,
            filters,
            aliases,
            tables,
            "SELECT");

        DbCallStack.Push(dbCall);
        VisitChildren(context);
        DbCallStack.Pop();

        return null;
    }

    public override object VisitCase_expression(
        TSqlParser.Case_expressionContext context)
    {
        if (CurrentCaseDepth == 0)
        {
            ExecutionOrder += 1;

            var decisionNode = CreateRuleSnippet(
                context,
                RuleConstructType.SWITCH_CONSTRUCT);

            if (CurrentFunction != null)
                CurrentFunction.DecisionNodeConstructs.Add(decisionNode);
            else
                MainProgramDecisionNodes.Add(decisionNode);
        }

        CurrentCaseDepth += 1;
        VisitChildren(context);
        CurrentCaseDepth -= 1;

        return null;
    }

    public override object VisitExecute_statement(
        TSqlParser.Execute_statementContext context)
    {
        string targetFunctionName = null;

        if (context.execute_body()?.func_proc_name_server_database_schema() != null)
        {
            targetFunctionName =
                ParseFuncProcNameServerDatabaseSchema(
                    context.execute_body().func_proc_name_server_database_schema());
        }

        if (TargetFunctions.Contains(targetFunctionName))
            CreateServiceCall(context, targetFunctionName);
        else
            CreateCallee(context, CalleeConstructType.EXECUTE);

        return VisitChildren(context);
    }

    public override object VisitRowset_function(
        TSqlParser.Rowset_functionContext context)
    {
        string snippetText = GetOriginalTextFromContext(context);

        if (snippetText.ToUpper().Contains("OPENROWSET"))
            CreateServiceCall(context, "OPENROWSET");

        return VisitChildren(context);
    }

    public override object VisitTransaction_statement(
        TSqlParser.Transaction_statementContext context)

            CreateTransactionCommit(context);
        else if (context.ROLLBACK() != null)
            CreateTransactionRollback(context);

        return VisitChildren(context);
    }

    // ----------------------------
    // Operand extraction (exact regex)
    // ----------------------------

    private List<string> GetOperands(string data)
    {
        var operands = new List<string>();

        var matches = Regex.Matches(data, @"[+\-*/]");

        foreach (Match match in matches)
            operands.Add(match.Value);

        return operands;
    }

    // ----------------------------
    // Target line finder (strict parity)
    // ----------------------------

    private (int, int) FindTargetLines(string targetData, int startLine)
    {
        try
        {
            var targetDataLine = targetData.Split('\n');

            if (targetDataLine.Length > 1)
            {
                int endLine = startLine + targetDataLine.Length - 1;
                return (startLine, endLine);
            }

            if (targetDataLine.Length == 1)
                return (startLine, startLine);

            return (0, 0);
        }
        catch
        {
            return (0, 0);
        }
    }

    // ----------------------------
    // Entity Exists (Cypher exact parity)
    // ----------------------------

    private bool EntityExists(string name)
    {
        string query = @"
            MATCH (e:DatabaseEntity)
            WHERE toLower(e.name) = toLower($name)
              AND e.project_id = $project_id
            RETURN COUNT(e) > 0 AS exists
        ";

        var result = Neo4j.Driver.GraphDatabase.Driver(
                ProjectMetadata.Neo4jUri,
                AuthTokens.Basic(
                    ProjectMetadata.Neo4jUser,
                    ProjectMetadata.Neo4jPassword))
            .Session()
            .Run(query, new
            {
                name = name,
                project_id = ProjectId
            })
            .Single();

        return result["exists"].As<bool>();
    }
