using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using Newtonsoft.Json;
using ParserService.Entities.Entities;
using ClosedXML.Excel;

namespace TsqlParserProject
{
    public static class ParserUtils
    {
        public static void WriteFunctionSnippetsToExcelWithCounts(CodeContext output, string inputFilePath, string saveDirectory)
        {
            // Extract the base name of the input file (without extension)
            string baseFilename = Path.GetFileNameWithoutExtension(inputFilePath);

            // Generate a timestamp for the filename
            string timestamp = DateTime.Now.ToString("ddMMyyyy_HHmmss");

            // Construct the Excel file name
            string excelFilename = $"{baseFilename}_Parser_Results_{timestamp}.xlsx";
            string excelPath = Path.Combine(saveDirectory, excelFilename);

            // Create a new workbook
            var workbook = new XLWorkbook();

            // Define the fields and their additional properties
            var fields = new Dictionary<string, List<string>>
            {
                { "DecisionNodeConstructs", new List<string> { "FileName", "Name", "StartLineNumber", "EndLineNumber" } },
                { "LoopEntity", new List<string> { "FileName", "Name", "StartLineNumber", "EndLineNumber" } },
                { "DbCalls", new List<string> { "FileName", "FunctionName", "StartLineNumber", "EndLineNumber", "OperationType", "Aliases", "TableNames", "FilterConditions" } },
                { "DatabaseEntities", new List<string> { "Name", "FileName", "StartLineNumber", "EndLineNumber" } },
                { "Calculation", new List<string> { "FileName", "Name", "StartLineNumber", "EndLineNumber", "Snippet", "Operands", "Variable" } },
                { "Callees", new List<string> { "CalleeFileName", "CallerFunction", "Snippet", "StartLineNumber", "EndLineNumber", "CalleeFunction", "Condition", "ConditionalHierarchy", "IsConditional", "Parameters" } },
                { "ServiceCalls", new List<string> { "FileName", "Name", "StartLineNumber", "EndLineNumber", "Snippet", "Target", "Parameters" } },
                { "TransactionCommits", new List<string> { "FileName", "Name", "StartLineNumber", "EndLineNumber", "Snippet" } },
                { "TransactionRollbacks", new List<string> { "FileName", "Name", "StartLineNumber", "EndLineNumber", "Snippet" } }
            };

            // Write each field's data to separate tabs
            foreach (var fieldEntry in fields)
            {
                string fieldName = fieldEntry.Key;
                var properties = fieldEntry.Value;

                var sheet = workbook.Worksheets.Add(fieldName);

                // Write header row
                for (int i = 0; i < properties.Count; i++)
                {
                    sheet.Cell(1, i + 1).Value = properties[i];
                }

                int rowIndex = 2;
                foreach (var functionSnippet in output.FunctionSnippets)
                {
                    var fieldObjects = GetPropertyValue(functionSnippet, fieldName) as System.Collections.IEnumerable;
                    if (fieldObjects != null)
                    {
                        foreach (var obj in fieldObjects)
                        {
                            for (int colIndex = 0; colIndex < properties.Count; colIndex++)
                            {
                                string prop = properties[colIndex];
                                var value = GetPropertyValue(obj, prop);

                                // Explicit handling for ConditionalHierarchy
                                if (prop == "ConditionalHierarchy" && value != null)
                                {
                                    sheet.Cell(rowIndex, colIndex + 1).Value = JsonConvert.SerializeObject(value, Formatting.Indented);
                                }
                                else
                                {
                                    sheet.Cell(rowIndex, colIndex + 1).Value = value?.ToString() ?? "";
                                }
                            }
                            rowIndex++;
                        }
                    }
                }
            }

            // Create a summary sheet for counts
            var summarySheet = workbook.Worksheets.Add("Summary");
            var summaryHeaders = new List<string>
            {
                "FileName", "FunctionName", "StartLineNumber", "EndLineNumber",
                "DecisionNodeConstructsCount", "LoopEntityCount", "DbCallsCount",
                "CalculationCount", "CalleesCount"
            };

            // Write summary headers
            for (int i = 0; i < summaryHeaders.Count; i++)
            {
                summarySheet.Cell(1, i + 1).Value = summaryHeaders[i];
            }

            int summaryRowIndex = 2;
            foreach (var functionSnippet in output.FunctionSnippets)
            {
                string fileName = GetPropertyValue(functionSnippet, "FileName")?.ToString();
                string functionName = GetPropertyValue(functionSnippet, "FunctionName")?.ToString();
                var startLineNumber = GetPropertyValue(functionSnippet, "StartLineNumber");
                var endLineNumber = GetPropertyValue(functionSnippet, "EndLineNumber");

                var decisionNodes = GetPropertyValue(functionSnippet, "DecisionNodeConstructs") as System.Collections.IEnumerable;
                var loopEntities = GetPropertyValue(functionSnippet, "LoopEntity") as System.Collections.IEnumerable;
                var dbCalls = GetPropertyValue(functionSnippet, "DbCalls") as System.Collections.IEnumerable;
                var calculations = GetPropertyValue(functionSnippet, "Calculation") as System.Collections.IEnumerable;
                var callees = GetPropertyValue(functionSnippet, "Callees") as System.Collections.IEnumerable;

                int decisionCount = decisionNodes?.Cast<object>().Count() ?? 0;
                int loopCount = loopEntities?.Cast<object>().Count() ?? 0;
                int dbCallCount = dbCalls?.Cast<object>().Count() ?? 0;
                int calcCount = calculations?.Cast<object>().Count() ?? 0;
                int calleeCount = callees?.Cast<object>().Count() ?? 0;

                summarySheet.Cell(summaryRowIndex, 1).Value = fileName;
                summarySheet.Cell(summaryRowIndex, 2).Value = functionName;
                summarySheet.Cell(summaryRowIndex, 3).Value = startLineNumber?.ToString() ?? "";
                summarySheet.Cell(summaryRowIndex, 4).Value = endLineNumber?.ToString() ?? "";
                summarySheet.Cell(summaryRowIndex, 5).Value = decisionCount;
                summarySheet.Cell(summaryRowIndex, 6).Value = loopCount;
                summarySheet.Cell(summaryRowIndex, 7).Value = dbCallCount;
                summarySheet.Cell(summaryRowIndex, 8).Value = calcCount;
                summarySheet.Cell(summaryRowIndex, 9).Value = calleeCount;

                summaryRowIndex++;
            }

            // Remove the default sheet created by ClosedXML (if it exists and is empty)
            var sheet1 = workbook.Worksheet("Sheet1");
            if (sheet1 != null)
            {
                sheet1.Delete();
            }

            // Save the workbook to the specified path
            workbook.SaveAs(excelPath);
        }

        private static object GetPropertyValue(object obj, string propertyName)
        {
            if (obj == null) return null;
            var property = obj.GetType().GetProperty(propertyName);
            if (property != null)
            {
                return property.GetValue(obj);
            }
            var field = obj.GetType().GetField(propertyName);
            if (field != null)
            {
                return field.GetValue(obj);
            }
            return null;
        }
    }
}
