using System;
using System.Collections.Generic;
using System.Linq;
using System.Text.RegularExpressions;

namespace TsqlParserProject
{
    public class TSqlPreprocessor
    {
        private string singleLineCommentPattern = @"(?:^|\s)(?:---|--)[^\n]*";
        private string multiLineCommentPattern = @"/\*[\s\S]*?\*/";
        private List<string> unwantedLinePatterns = new()
        {
            @"^[ \t]*Text\s+--.*$",
            @"^[ \t]*Completion\stime:.*$"
        };

        public string RemoveUnwantedLines(string fileContent)
        {
            foreach (var pattern in unwantedLinePatterns)
            {
                fileContent = Regex.Replace(fileContent, pattern, match =>
                {
                    int lineCount = match.Value.Count(c => c == '\n');
                    return new string('\n', lineCount);
                }, RegexOptions.Multiline);
            }
            return fileContent;
        }

        public string ExtractComments(string fileContent)
        {
            var singleLineComments = Regex.Matches(fileContent, singleLineCommentPattern, RegexOptions.Multiline)
                                          .Cast<Match>().Select(m => m.Value);
            var multiLineComments = Regex.Matches(fileContent, multiLineCommentPattern, RegexOptions.Multiline)
                                         .Cast<Match>().Select(m => m.Value);
            
            return string.Join("\n", singleLineComments.Concat(multiLineComments));
        }

        public string RemoveComments(string fileContent)
        {
            string textWithoutComments = Regex.Replace(fileContent, singleLineCommentPattern, "", RegexOptions.Multiline);
            
            textWithoutComments = Regex.Replace(textWithoutComments, multiLineCommentPattern, match =>
            {
                int lineCount = match.Value.Count(c => c == '\n');
                return new string('\n', lineCount);
            }, RegexOptions.Multiline);

            return textWithoutComments;
        }
    }
}
