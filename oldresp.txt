#nullable enable
using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Text.RegularExpressions;

namespace ParserService.Grammar.LanguageParser.tsql.visitor.extractors
{
    /// <summary>
    /// Extracts constants, variables, and mathematical calculations from T-SQL code.
    /// Optimized C# port with caching, case-insensitive deduplication, and safe null handling.
    /// </summary>
    public class ConstantVariableExtractor
    {
        // ---- Optional external error listener ----
        public static Action<string, Exception>? ErrorListener { get; set; }

        // ---- Static cached resources (initialized once) ----
        private static readonly HashSet<string> keywords = new(StringComparer.OrdinalIgnoreCase);
        private static string? negative_lookahead;
        private static string? variable_pattern;
        private static Regex? variable_regex;
        private static Regex? keywords_as_vars;
        private static Regex? string_constant_regex;
        private static Regex? numeric_constant_regex;
        private static Regex? calc_regex;
        private static string? pattern_resultant;
        private static readonly object keywordsLock = new();
        private static bool initialized = false;

        static ConstantVariableExtractor()
        {
            lock (keywordsLock)
            {
                if (!initialized)
                {
                    try
                    {
                        LoadKeywordsAndCompileRegexes();
                        initialized = true;
                    }
                    catch (Exception ex)
                    {
                        ErrorListener?.Invoke("Static initialization failed", ex);
                    }
                }
            }
        }

        private static void LoadKeywordsAndCompileRegexes()
        {
            try
            {
                // ---- Read lexer keywords once ----
                string? currentDir = Path.GetDirectoryName(System.Reflection.Assembly.GetExecutingAssembly().Location);
                string targetFilePath = Path.Combine(currentDir ?? ".", "..", "..", "..", "grammer", "TSqlLexer.g4");
                targetFilePath = Path.GetFullPath(targetFilePath);

                string startMarker = "// Keywords Start Here";
                string endMarker = "// Keywords End Here";
                bool capture = false;

                if (File.Exists(targetFilePath))
                {
                    foreach (string rawLine in File.ReadLines(targetFilePath))
                    {
                        string line = rawLine.Trim();
                        if (line == startMarker) { capture = true; continue; }
                        if (line == endMarker) { capture = false; break; }

                        if (capture && line.Length > 0 && !line.StartsWith("//"))
                        {
                            Match colonMatch = Regex.Match(line, @":(.*?);");
                            if (colonMatch.Success)
                            {
                                string afterColon = colonMatch.Groups[1].Value;
                                foreach (Match keywordMatch in Regex.Matches(afterColon, @"'[^']*'"))
                                {
                                    string keyword = keywordMatch.Value.Trim('\'', ' ');
                                    if (!string.IsNullOrWhiteSpace(keyword))
                                        keywords.Add(keyword);
                                }
                            }
                        }
                    }
                }

                // ---- Build and compile regex patterns ----
                negative_lookahead = string.Join("|", keywords.Select(Regex.Escape));
                variable_pattern = $@"(?<!@)(?!(?:{negative_lookahead})(?=[\s(]|$))\b[A-Z]+(?:[-._][A-Z0-9]+)*\b";
                variable_regex = new Regex(variable_pattern, RegexOptions.IgnoreCase | RegexOptions.Compiled);

                keywords_as_vars = new Regex(@"(?:(?<=\s)|(?<=^))(@[A-Z][A-Z0-9_]*)(?=[\s(]|$)",
                    RegexOptions.IgnoreCase | RegexOptions.Compiled);

                string_constant_regex = new Regex(@"'([^']*)'", RegexOptions.Compiled);
                numeric_constant_regex = new Regex(@"(?<=\s)\d+(?=\s)|(?<=\s)\d+$", RegexOptions.Compiled);

                string calcPattern =
                    $@"(?!(?:T|CT|ECT|LECT|ELECT|{negative_lookahead})(?=\s|$))" +
                    @"(?:(?:SET|SELECT)?\s*(@[\w]+)?\s*(?:=\s*)?)?\(*\s*@[\w.]+(?:\s*[+\-*/]\s*[@\w.()]+)+\s*\)*";
                calc_regex = new Regex(calcPattern, RegexOptions.IgnoreCase | RegexOptions.Compiled);

                pattern_resultant = @"SET\s+(@\w+)\s+=";
            }
            catch (Exception ex)
            {
                ErrorListener?.Invoke("LoadKeywordsAndCompileRegexes", ex);
            }
        }

        // ---- Extract mathematical calculations ----
        public List<Dictionary<string, object>> extract_mathematical_calculations(string code_snippet)
        {
            var list = new List<Dictionary<string, object>>();
            try
            {
                foreach (Match match in calc_regex!.Matches(code_snippet))
                {
                    list.Add(new Dictionary<string, object>
                    {
                        { "match_group", match.Value },
                        { "start", match.Index },
                        { "end", match.Index + match.Length }
                    });
                }
            }
            catch (Exception ex)
            {
                ErrorListener?.Invoke(nameof(extract_mathematical_calculations), ex);
            }
            return list;
        }

        // ---- Extract calculations with line numbers ----
        public List<Dictionary<string, object>> extract_mathematical_calculations_startline(string code_snippet)
        {
            var list = new List<Dictionary<string, object>>();
            try
            {
                string[] lines = code_snippet.Split('\n');
                for (int i = 0; i < lines.Length; i++)
                {
                    foreach (Match match in calc_regex!.Matches(lines[i]))
                    {
                        list.Add(new Dictionary<string, object>
                        {
                            { "match_group", match.Value },
                            { "start", i + 1 }
                        });
                    }
                }
            }
            catch (Exception ex)
            {
                ErrorListener?.Invoke(nameof(extract_mathematical_calculations_startline), ex);
            }
            return list;
        }

        // ---- Extract constants & variables ----
        public List<DataDefinitionJSON> extract_constants_variables(string code_snippet, string file_path)
        {
            var result = new List<DataDefinitionJSON>();
            var dict = new Dictionary<string, DataDefinitionJSON>(StringComparer.OrdinalIgnoreCase);

            try
            {
                foreach (string rawLine in code_snippet.Split('\n'))
                {
                    string line = rawLine ?? string.Empty;
                    if (line.TrimStart().StartsWith("--")) continue;

                    // string constants
                    foreach (Match match in string_constant_regex!.Matches(line))
                    {
                        string constant = match.Groups[1].Value;
                        if (constant.Trim('\'').Split(' ').Length == 1)
                            AddOrAppend(dict, constant, "CONSTANT", file_path, line);
                    }

                    // variables
                    foreach (Match match in variable_regex!.Matches(line))
                    {
                        string variable = match.Value;
                        if (variable.Trim('\'').Split(' ').Length == 1)
                            AddOrAppend(dict, variable, "DECLARATION", file_path, line);
                    }

                    // keyword variables (@var)
                    foreach (Match match in keywords_as_vars!.Matches(line))
                    {
                        string varName = match.Value;
                        if (varName.Trim('\'').Split(' ').Length == 1)
                            AddOrAppend(dict, varName, "DECLARATION", file_path, line);
                    }
                }
            }
            catch (Exception ex)
            {
                ErrorListener?.Invoke(nameof(extract_constants_variables), ex);
            }

            result.AddRange(dict.Values);
            return result;
        }

        // ---- Extract numeric constants ----
        public List<DataDefinitionJSON> extract_numeric_constants(string code_snippet, string file_path)
        {
            var result = new List<DataDefinitionJSON>();
            var dict = new Dictionary<string, DataDefinitionJSON>(StringComparer.OrdinalIgnoreCase);

            try
            {
                foreach (string line in code_snippet.Split('\n'))
                {
                    foreach (Match match in numeric_constant_regex!.Matches(line))
                    {
                        string constant = match.Value;
                        if (constant.Trim('\'').Split(' ').Length == 1)
                            AddOrAppend(dict, constant, "CONSTANT", file_path, line);
                    }
                }
            }
            catch (Exception ex)
            {
                ErrorListener?.Invoke(nameof(extract_numeric_constants), ex);
            }

            result.AddRange(dict.Values);
            return result;
        }

        // ---- Helper: Add or append usage context ----
        private static void AddOrAppend(Dictionary<string, DataDefinitionJSON> dict, string key, string type, string file, string line)
        {
            if (!dict.TryGetValue(key, out var existing))
            {
                dict[key] = new DataDefinitionJSON(type, key, new List<string> { line }, file, "");
            }
            else
            {
                existing.UsageContext.Add(line);
            }
        }

        // ---- Legacy wrappers ----
        public List<Dictionary<string, object>> ExtractMathematicalCalculations(string codeSnippet) =>
            extract_mathematical_calculations(codeSnippet);
        public List<Dictionary<string, object>> ExtractMathematicalCalculationsStartline(string codeSnippet) =>
            extract_mathematical_calculations_startline(codeSnippet);
        public List<DataDefinitionJSON> ExtractConstantsVariables(string codeSnippet, string filePath) =>
            extract_constants_variables(codeSnippet, filePath);
        public List<DataDefinitionJSON> ExtractNumericConstants(string codeSnippet, string filePath) =>
            extract_numeric_constants(codeSnippet, filePath);
    }

    // ---- POCO data model ----
    public class DataDefinitionJSON
    {
        public string Type { get; set; }
        public string Term { get; set; }
        public List<string> UsageContext { get; set; }
        public string FilePath { get; set; }
        public string Meaning { get; set; }

        public DataDefinitionJSON(string type, string term, List<string> usage_context, string file_path, string meaning)
        {
            Type = type;
            Term = term;
            UsageContext = usage_context ?? new();
            FilePath = file_path;
            Meaning = meaning;
        }

        // Lowercase property aliases for backward compatibility
        public string type { get => Type; set => Type = value; }
        public string term { get => Term; set => Term = value; }
        public List<string> usage_context { get => UsageContext; set => UsageContext = value; }
        public string file_path { get => FilePath; set => FilePath = value; }
        public string meaning { get => Meaning; set => Meaning = value; }
    }
}
